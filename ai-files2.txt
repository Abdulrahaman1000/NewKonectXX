/**
 * Builds the assistant's knowledge from the live database: active combos,
 * gifts in stock, published FAQs, and the shop's real rules. Cached for 60s.
 *
 * Nothing here is invented: every price, stock count and rule comes from your
 * database or from code that checkout itself uses.
 */

import { Combo } from "../models/Combo";
import { Gift } from "../models/Gift";
import { FAQ } from "../models/FAQ";
import { SiteSettings } from "../models/SiteSettings";
import { GIFT_THRESHOLD, GIFT_WINDOW_DAYS } from "../services/giftProgress";
import type { KDoc } from "./retrieve";

const TTL_MS = 60_000;
let cache: { at: number; docs: KDoc[] } | null = null;

const naira = (n: number) => "₦" + Math.round(n).toLocaleString("en-NG");
const cut = (s: unknown, max = 500) => String(s ?? "").replace(/\s+/g, " ").trim().slice(0, max);

export async function getKnowledge(): Promise<KDoc[]> {
  if (cache && Date.now() - cache.at < TTL_MS) return cache.docs;

  const [combos, gifts, faqs, settings] = await Promise.all([
    Combo.find({ isActive: true }).sort({ isFeatured: -1, createdAt: -1 }).limit(60).lean(),
    Gift.find({ isActive: { $ne: false }, stock: { $gt: 0 } }).limit(30).lean(),
    FAQ.find({ isPublished: true }).sort({ order: 1 }).limit(40).lean(),
    SiteSettings.findOne().lean(),
  ]);

  const docs: KDoc[] = [];

  for (const c of combos as any[]) {
    const inStock = (c.stockLeft ?? 0) > 0;
    const save = Math.max(0, (c.originalPrice ?? 0) - (c.totalPrice ?? 0));
    const items = (c.items ?? [])
      .map((it: any) => {
        const alts = (it.alternatives ?? []).map((a: any) => a.name).filter(Boolean);
        const colors = (it.colors ?? []).map((k: any) => k.name).filter(Boolean);
        return (
          cut(it.name, 80) +
          (it.individualPrice ? ` (${naira(it.individualPrice)} on its own)` : "") +
          (alts.length ? ` — alternatives: ${alts.slice(0, 5).join(", ")}` : "") +
          (colors.length ? ` — colours: ${colors.slice(0, 8).join(", ")}` : "")
        );
      })
      .join("; ");
    docs.push({
      id: `combo:${c.slug}`,
      type: "combo",
      title: cut(c.name, 80),
      price: c.totalPrice,
      inStock,
      link: `/combos/${c.slug}`,
      text: cut(
        `${c.tagline ? c.tagline + ". " : ""}Combo price ${naira(c.totalPrice)}` +
          (save > 0 ? ` (normally ${naira(c.originalPrice)}, you save ${naira(save)})` : "") +
          `. ${inStock ? `In stock, ${c.stockLeft} left` : "Out of stock"}.` +
          (items ? ` Includes: ${items}.` : ""),
        900,
      ),
    });
  }

  for (const f of faqs as any[]) {
    docs.push({
      id: `faq:${f._id}`,
      type: "faq",
      title: cut(f.question, 120),
      text: cut(f.answer, 700),
      link: "/faq",
    });
  }

  // ---- Rules that checkout really applies ----
  const ship = (settings as any)?.shipping;
  const contact = (settings as any)?.contact;
  const cities: string[] = (ship?.codCities ?? []).filter(Boolean);

  if (ship) {
    docs.push({
      id: "policy:delivery",
      type: "policy",
      title: "Delivery fee and cash on delivery",
      link: "/shipping",
      text:
        `The standard delivery fee is ${naira(ship.standardFee ?? 0)}. ` +
        (cities.length
          ? `If the customer pays cash on delivery in ${cities.join(", ")}, the delivery fee is ${naira(0)}. `
          : "") +
        "The exact fee is always shown at checkout before paying.",
    });
  }

  docs.push({
    id: "policy:gift",
    type: "policy",
    title: "How to qualify for a free gift",
    text:
      `A customer unlocks the free gift when what they spent in the last ${GIFT_WINDOW_DAYS} days plus the current order reaches ${naira(GIFT_THRESHOLD)}. ` +
      "Spending is added up by phone number. An order counts once it is paid; cash-on-delivery orders count once delivered. Cancelled and refunded orders never count. " +
      "After a gift is given, the total starts again from zero." +
      (gifts.length ? ` Gifts available now: ${(gifts as any[]).map((g) => cut(g.name, 60)).join(", ")}.` : " No gifts are in stock right now."),
  });

  docs.push({
    id: "policy:combo",
    type: "policy",
    title: "How combos work",
    link: "/products",
    text:
      "A combo bundles several items at one combo price, which is lower than buying the items separately; each combo page shows how much you save. " +
      "Some items let you choose an alternative product or a colour at the same combo price. " +
      "Customers can also build their own combo at /products?mode=combo-builder.",
  });

  docs.push({
    id: "policy:payment",
    type: "policy",
    title: "Payment",
    text:
      "Payment options are shown at checkout; they include online payment through Paystack, bank transfer, and cash on delivery in selected cities. " +
      "Bank transfer details are shown at checkout.",
  });

  if (contact) {
    docs.push({
      id: "policy:contact",
      type: "policy",
      title: "Contact us",
      link: "/contact",
      text:
        [
          contact.phone ? `Phone: ${contact.phone}` : "",
          contact.whatsappNumber ? `WhatsApp: ${contact.whatsappNumber}` : "",
          contact.email ? `Email: ${contact.email}` : "",
          contact.address ? `Address: ${contact.address}` : "",
        ]
          .filter(Boolean)
          .join(". ") || "Use the chat to reach the team.",
    });
  }

  cache = { at: Date.now(), docs };
  return docs;
}

/**
 * Retrieval for the Smart Combo assistant (pure functions, no database here).
 *
 * Simple, honest keyword retrieval: we score each knowledge document against
 * the customer's question and keep the best few. Only those documents are shown
 * to the AI, so it can only answer from real shop data.
 */

export type DocType = "combo" | "gift" | "faq" | "policy";

export interface KDoc {
  id: string;
  type: DocType;
  title: string;
  text: string;
  link?: string;
  inStock?: boolean;
  price?: number;
}

const STOP = new Set(
  "a an the is are was were be been do does did can could would should will i you your we our my me it its of to for on in at by with and or but if so this that these those there what which who how when where why any some please tell about get have has not no yes just want need like very much more than then from as into up out".split(
    " ",
  ),
);

export function stem(w: string): string {
  if (w.length > 4 && w.endsWith("ies")) return w.slice(0, -3) + "y";
  if (w.length > 4 && w.endsWith("es")) return w.slice(0, -2);
  if (w.length > 3 && w.endsWith("s")) return w.slice(0, -1);
  return w;
}

export function tokenize(s: string): string[] {
  return s
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, " ")
    .split(/\s+/)
    .filter((w) => w.length > 1 && !STOP.has(w))
    .map(stem);
}

/** Finds a naira budget in the question: "₦50,000", "50k", "under 50000". */
export function parseBudget(q: string): number | null {
  const text = q.toLowerCase().replace(/,/g, "");
  const scale = (n: string, suffix?: string) => {
    let v = parseFloat(n);
    if (suffix === "k") v *= 1_000;
    if (suffix === "m") v *= 1_000_000;
    return v;
  };

  const patterns: RegExp[] = [
    /(?:₦|\bngn\b\s*|\bn(?=\d))\s*(\d+(?:\.\d+)?)\s*(k|m)?\b/,
    /\b(?:under|below|within|less than|up to|max(?:imum)?|budget(?: of)?|around|about)\s*(?:₦|ngn\s*|n(?=\d))?\s*(\d+(?:\.\d+)?)\s*(k|m)?\b/,
    /\b(\d+(?:\.\d+)?)\s*(k)\b/,
  ];
  for (const re of patterns) {
    const m = text.match(re);
    if (m) {
      const v = scale(m[1], m[2]);
      if (v >= 1000) return Math.round(v);
    }
  }
  return null;
}

const INTENTS: { id: string; words: string[] }[] = [
  { id: "policy:gift", words: ["gift", "free", "reward", "qualify", "unlock", "threshold"] },
  { id: "policy:delivery", words: ["deliver", "delivery", "shipping", "ship", "fee", "arrive", "cod", "cash", "dispatch", "transport"] },
  { id: "policy:combo", words: ["combo", "bundle", "offer", "save", "saving", "discount", "custom", "build"] },
  { id: "policy:payment", words: ["pay", "payment", "bank", "transfer", "paystack", "card", "flutterwave"] },
  { id: "policy:contact", words: ["contact", "whatsapp", "phone", "call", "address", "location", "email", "reach"] },
];

const BROWSE = /\b(available|in stock|what do you (have|sell)|products?|catalog(ue)?|list|options?|show me|everything)\b/i;
const GREETING = /^\s*(hi|hello|hey|good (morning|afternoon|evening)|howdy|yo|sup)\b[\s!.?]*$/i;

export function isGreeting(q: string): boolean {
  return GREETING.test(q);
}

export interface Retrieval {
  docs: KDoc[];
  budget: number | null;
  matched: boolean;
}

const fmt = (n: number) => "₦" + Math.round(n).toLocaleString("en-NG");

export function retrieve(question: string, all: KDoc[], limit = 6): Retrieval {
  const q = question.trim();
  const qTokens = tokenize(q);
  const qSet = new Set(qTokens);
  const lower = q.toLowerCase();
  const budget = parseBudget(q);

  const scored = new Map<string, { doc: KDoc; score: number }>();
  const bump = (doc: KDoc, s: number) => {
    const cur = scored.get(doc.id);
    if (cur) cur.score += s;
    else scored.set(doc.id, { doc, score: s });
  };

  for (const doc of all) {
    const titleTokens = new Set(tokenize(doc.title));
    const bodyTokens = new Set(tokenize(doc.text));
    let s = 0;
    for (const t of qSet) {
      if (titleTokens.has(t)) s += 3;
      else if (bodyTokens.has(t)) s += 1;
    }
    if (doc.title.length > 3 && lower.includes(doc.title.toLowerCase())) s += 6;
    if (s > 0) bump(doc, s);
  }

  // Intent boosts: policy documents the question is clearly about
  for (const intent of INTENTS) {
    if (intent.words.some((w) => qSet.has(stem(w)) || lower.includes(w))) {
      const doc = all.find((d) => d.id === intent.id);
      if (doc) bump(doc, 5);
    }
  }

  const extra: KDoc[] = [];

  // Budget question: build an exact list from the real catalogue
  if (budget) {
    const combos = all.filter((d) => d.type === "combo" && typeof d.price === "number");
    const within = combos.filter((d) => (d.price as number) <= budget);
    const inStock = within.filter((d) => d.inStock).sort((a, b) => (a.price as number) - (b.price as number));
    const out = within.filter((d) => !d.inStock);
    const lines = inStock.map((d) => `- ${d.title}: ${fmt(d.price as number)} (in stock) [${d.id}]`);
    const outLines = out.map((d) => `- ${d.title}: ${fmt(d.price as number)} (out of stock) [${d.id}]`);
    extra.push({
      id: "list:budget",
      type: "policy",
      title: `Combos within ${fmt(budget)}`,
      text: lines.length || outLines.length
        ? [...lines, ...outLines].join("\n")
        : `No active combo costs ${fmt(budget)} or less.`,
    });
  }

  // Browsing question: short overview of what's on sale
  if (BROWSE.test(q) && !budget) {
    const combos = all.filter((d) => d.type === "combo");
    extra.push({
      id: "list:catalogue",
      type: "policy",
      title: "Current combos",
      text: combos.length
        ? combos
            .slice(0, 15)
            .map((d) => `- ${d.title}: ${fmt(d.price ?? 0)} (${d.inStock ? "in stock" : "out of stock"}) [${d.id}]`)
            .join("\n")
        : "There are no active combos right now.",
    });
  }

  const ranked = [...scored.values()]
    .filter((x) => x.score >= 2)
    .sort((a, b) => b.score - a.score)
    .slice(0, limit)
    .map((x) => x.doc);

  const docs = [...extra, ...ranked.filter((d) => !extra.some((e) => e.id === d.id))].slice(0, limit + extra.length);
  return { docs, budget, matched: docs.length > 0 };
}

/**
 * AI assistant API. Nothing is stored on the server; the browser sends the
 * last few messages with each question.
 */

import { apiFetch } from "./client";

export interface AISource {
  id: string;
  title: string;
  link: string | null;
}

export interface AIReply {
  answer: string;
  sources: AISource[];
  needsHuman: boolean;
}

export interface AITurn {
  role: "user" | "assistant";
  text: string;
}

export async function askAI(messages: AITurn[]): Promise<AIReply> {
  const res = await apiFetch<{ data: AIReply }>("/api/ai/chat", {
    method: "POST",
    body: JSON.stringify({ messages }),
  });
  return res.data;
}

function loadAI(): AIMsg[] {
  try {
    const raw = JSON.parse(localStorage.getItem(AI_KEY) || '[]');
    return Array.isArray(raw) ? raw.slice(-MAX_KEPT) : [];
  } catch {
    return [];
  }
}
function saveAI(m: AIMsg[]) {
  try {
    localStorage.setItem(AI_KEY, JSON.stringify(m.slice(-MAX_KEPT)));
  } catch {
    /* private mode: chat still works for this visit */
  }
}

function AIChat({ onHuman, whatsappLink }: { onHuman: (q?: string) => void; whatsappLink: string }) {
  const [messages, setMessages] = useState<AIMsg[]>(loadAI);
  const [text, setText] = useState('');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  const bottomRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => saveAI(messages), [messages]);
  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
  }, [messages.length, busy, error]);
  useEffect(() => {
    if (window.matchMedia('(min-width: 768px)').matches) inputRef.current?.focus();
  }, []);

  const lastQuestion = [...messages].reverse().find((m) => m.role === 'user')?.text;

  const ask = useCallback(
    async (question: string, base: AIMsg[]) => {
      const next = [...base, { id: uid(), role: 'user' as const, text: question }];
      setMessages(next);
      setBusy(true);
      setError('');
      try {
        const turns = next.slice(-10).map((m) => ({ role: m.role, text: m.text }));
        const r = await askAI(turns);
        setMessages((cur) => [
          ...cur,
          { id: uid(), role: 'assistant', text: r.answer, sources: r.sources, needsHuman: r.needsHuman },
        ]);
      } catch (err) {
        const msg =
          err instanceof ApiError && err.status === 429
            ? err.message
            : "Our assistant isn't available right now. You can talk to our team instead.";
        setError(msg);
      } finally {
        setBusy(false);
      }
    },
    [],
  );

  const submit = (e: React.FormEvent) => {
    e.preventDefault();
    const q = text.trim();
    if (!q || busy) return;
    setText('');
    void ask(q, messages);
  };

  const retry = () => {
    if (!lastQuestion || busy) return;
    // drop the unanswered last question, then ask it again
    const base = messages.slice(0, messages.length - 1);
    void ask(lastQuestion, base);
  };

  const close = () => {
    if (window.innerWidth < 768) chatUI.close();
  };

  return (
    <>
      <div className="flex-1 overflow-y-auto px-3 py-4 space-y-3">

            </div>

              {AI_STARTERS.map((s) => (
                <button
                  key={s}

                </button>
              ))}
            </div>
          </div>
        )}

        {messages.map((m) => {
          const mine = m.role === 'user';
          return (
            <div key={m.id} className={`flex flex-col ${mine ? 'items-end' : 'items-start'}`}>
              <div
                className={`max-w-[85%] rounded-2xl px-3.5 py-2.5 ${
                  mine
                    ? 'bg-primary text-primary-foreground rounded-br-md'
                    : 'bg-muted text-foreground rounded-bl-md'

              {!mine && m.sources && m.sources.length > 0 && (

                  <button
                    type="button"
                    onClick={() => onHuman(lastQuestion)}
                    className="px-3 h-9 rounded-full bg-primary text-primary-foreground text-sm font-semibold"
                  >
                    Talk to a person
                  </button>
                  {whatsappLink && (
                    <a
                      href={whatsappLink}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="px-3 h-9 rounded-full border border-emerald-500/50 text-emerald-500 text-sm font-semibold flex items-center"
                    >
                      WhatsApp
                    </a>
                  )}
                </div>
              )}
            </div>
          );
        })}

        {busy && (
          <div className="flex justify-start">
            <div className="rounded-2xl rounded-bl-md bg-muted px-4 py-3 flex gap-1" aria-label="Assistant is typing">
              {[0, 150, 300].map((d) => (
                <span
                  key={d}
                  className="w-2 h-2 rounded-full bg-muted-foreground/60 animate-bounce"
                  style={{ animationDelay: `${d}ms` }}
                />
              ))}
            </div>
          </div>
        )}

        {error && (
          <div className="rounded-2xl border border-destructive/40 bg-destructive/10 p-3">
            <p className="text-sm">{error}</p>
            <div className="mt-2 flex flex-wrap gap-2">
              <button
                type="button"
                onClick={retry}
                className="px-3 h-9 rounded-full border border-border text-sm font-semibold hover:bg-foreground/5"
              >
                Try again
              </button>
              <button
                type="button"
                onClick={() => onHuman(lastQuestion)}
                className="px-3 h-9 rounded-full bg-primary text-primary-foreground text-sm font-semibold"
              >
                Talk to a person
              </button>
            </div>
          </div>
        )}
        <div ref={bottomRef} />
      </div>

      <div className="shrink-0 border-t border-border bg-background">
        {messages.length > 0 && (
          <div className="px-4 pt-1.5 flex justify-end">
            <button
              type="button"
              onClick={() => {
                setMessages([]);
                setError('');
              }}
              className="text-[11px] text-muted-foreground hover:text-foreground underline"
            >
              Clear chat
            </button>
          </div>
        )}
        <form
          onSubmit={submit}
          className="flex items-center gap-2 px-3 pt-2"
          style={{ paddingBottom: 'calc(env(safe-area-inset-bottom, 0px) + 0.25rem)' }}
        >
          <input
            ref={inputRef}
            value={text}
            onChange={(e) => setText(e.target.value)}
            maxLength={500}
            enterKeyHint="send"
            placeholder="Ask about a product, price or delivery"
            className="flex-1 min-w-0 h-12 rounded-full border border-border bg-muted/50 px-4 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
          />
          <button
            type="submit"
            disabled={busy || !text.trim()}
            aria-label="Send"
            className="w-12 h-12 shrink-0 rounded-full bg-primary text-primary-foreground flex items-center justify-center active:scale-95 transition disabled:opacity-50"
          >
            {busy ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
          </button>
        </form>
        <p
          className="px-4 pb-2 text-[10px] text-muted-foreground text-center"
          style={{ paddingBottom: 'calc(env(safe-area-inset-bottom, 0px) + 0.5rem)' }}
        >
          AI can make mistakes. Always confirm the final price at checkout.
        </p>
      </div>
    </>
  );
}

/* Human chat tab (the original chat, same behaviour)                  */
/* ------------------------------------------------------------------ */

function HumanChat({ prefill }: { prefill: string }) {
  const [token, setToken] = useState<string | null>(getChatToken());
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [loading, setLoading] = useState<boolean>(!!getChatToken());
  const [sending, setSending] = useState(false);
  const [error, setError] = useState('');

  const [name, setName] = useState('');
  const [phone, setPhone] = useState('');
  const [firstMessage, setFirstMessage] = useState(prefill);
  const [text, setText] = useState('');

  const bottomRef = useRef<HTMLDivElement>(null);

  const load = useCallback(async (t: string) => {
    try {
      setMessages(await fetchChat(t));
    } catch (err) {
      if (err instanceof ApiError && err.status === 404) {
        // The chat was removed on the server: start fresh
        clearChatToken();
        setToken(null);
        setMessages([]);
      }
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    if (!token) return;
    load(token);
    const id = setInterval(() => {
      if (document.visibilityState === 'visible') load(token);
    }, 4000);
    return () => clearInterval(id);
  }, [token, load]);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages.length]);

  const begin = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!name.trim() || !firstMessage.trim() || sending) return;
    setSending(true);
    setError('');
    try {
      const r = await startChat({
        name: name.trim(),
        phone: phone.trim() || undefined,
        message: firstMessage.trim(),
      });
      saveChatToken(r.token);
      setToken(r.token);
      setMessages(r.messages);
    } catch (err: any) {
      setError(err?.message || 'Could not start the chat. Please try again.');
    } finally {
      setSending(false);
    }
  };

  const send = async (e: React.FormEvent) => {
    e.preventDefault();
    const t = text.trim();
    if (!t || !token || sending) return;
    setSending(true);
    setError('');
    try {
      await sendChatMessage(token, t);
      setText('');
      await load(token);
    } catch (err: any) {
      setError(err?.message || 'Message not sent. Please try again.');
    } finally {
      setSending(false);
    }
  };

  if (!token) {
    return (
      <form onSubmit={begin} className="flex-1 overflow-y-auto px-4 py-5 space-y-4">
        <div>
          <h2 className="text-xl font-black">Talk to our team</h2>
          <p className="text-sm text-muted-foreground mt-1">
            A real person will reply here. No account needed.
          </p>
        </div>

        <label className="block">
          <span className="block text-[11px] font-bold text-muted-foreground uppercase tracking-wider mb-1.5">
            Your name *
          </span>
          <input
            value={name}
            onChange={(e) => setName(e.target.value)}
            maxLength={60}
            autoComplete="name"
            className="w-full h-12 rounded-xl border border-border bg-transparent px-3.5 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
            placeholder="e.g. Ade"
          />
        </label>

        <label className="block">
          <span className="block text-[11px] font-bold text-muted-foreground uppercase tracking-wider mb-1.5">
            Phone (optional)
          </span>
          <input
            value={phone}
            onChange={(e) => setPhone(e.target.value)}
            maxLength={30}
            inputMode="tel"
            autoComplete="tel"
            className="w-full h-12 rounded-xl border border-border bg-transparent px-3.5 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
            placeholder="So we can call you back"
          />
        </label>

        <div>
          <span className="block text-[11px] font-bold text-muted-foreground uppercase tracking-wider mb-1.5">
            Your message *
          </span>
          <div className="flex flex-wrap gap-2 mb-2">
            {QUICK.map((q) => (
              <button
                key={q}
                type="button"
                onClick={() => setFirstMessage(q)}
                className="px-3 h-9 rounded-full border border-emerald-500/50 text-emerald-500 text-sm hover:bg-emerald-500/10"
              >
                {q}
              </button>
            ))}
          </div>
          <textarea
            value={firstMessage}
            onChange={(e) => setFirstMessage(e.target.value)}
            maxLength={1000}
            rows={4}
            className="w-full rounded-xl border border-border bg-transparent px-3.5 py-3 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
            placeholder="Type your message here"
          />
        </div>

        {error && <p className="text-sm text-destructive">{error}</p>}

        <button
          type="submit"
          disabled={sending || !name.trim() || !firstMessage.trim()}
          className="btn-primary w-full h-12 font-bold text-sm flex items-center justify-center gap-2 disabled:opacity-50"
        >
          {sending ? <Loader2 className="w-4 h-4 animate-spin" /> : <Send className="w-4 h-4" />}
          Start chat
        </button>
      </form>
    );
  }

  return (
    <>
      <div className="flex-1 overflow-y-auto px-3 py-4 space-y-2.5">
        {loading ? (
          <div className="py-16 flex justify-center">
            <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
          </div>
        ) : (
          messages.map((m) => {
            const mine = m.sender === 'customer';
            return (
              <div key={m.id} className={`flex ${mine ? 'justify-end' : 'justify-start'}`}>
                <div
                  className={`max-w-[80%] rounded-2xl px-3.5 py-2 ${
                    mine
                      ? 'bg-primary text-primary-foreground rounded-br-md'
                      : 'bg-muted text-foreground rounded-bl-md'
                  }`}
                >
                  <p className="text-[15px] leading-snug whitespace-pre-wrap break-words">{m.text}</p>
                  <p className={`text-[10px] mt-1 text-right ${mine ? 'opacity-60' : 'text-muted-foreground'}`}>
                    {timeOf(m.createdAt)}
                  </p>
                </div>
              </div>
            );
          })
        )}
        <div ref={bottomRef} />
      </div>

      {error && <p className="px-4 pb-1 text-xs text-destructive">{error}</p>}

      <form
        onSubmit={send}
        className="shrink-0 flex items-center gap-2 px-3 pt-2 border-t border-border bg-background"
        style={{ paddingBottom: 'calc(env(safe-area-inset-bottom, 0px) + 0.5rem)' }}
      >
        <input
          value={text}
          onChange={(e) => setText(e.target.value)}
          maxLength={1000}
          enterKeyHint="send"
          placeholder="Type a message"
          className="flex-1 min-w-0 h-12 rounded-full border border-border bg-muted/50 px-4 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
        />
        <button
          type="submit"
          disabled={sending || !text.trim()}
          aria-label="Send"
          className="w-12 h-12 shrink-0 rounded-full bg-primary text-primary-foreground flex items-center justify-center active:scale-95 transition disabled:opacity-50"
        >
          {sending ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
        </button>
      </form>
    </>
  );
}
