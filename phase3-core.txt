export class OrderError extends Error {
  status: number;
  code: string;
  constructor(status: number, code: string, message: string) {
    super(message);
    this.status = status;
    this.code = code;

async function generateOrderNumber(): Promise<string> {
  const year = new Date().getFullYear();
  const seq = await nextSeq(`order:${year}`);
  return `SC-${year}-${String(seq).padStart(6, "0")}`;
}

/** Flat delivery fee for customers in Ilorin. Everywhere else Jumia sets the price and the customer pays Jumia directly. */
export const LOCAL_DELIVERY_FEE = 1000;

async function calculateShippingFee(
  _paymentMethod: PaymentMethod,
  city: string,
): Promise<number> {
  const settings = await SiteSettings.findOne().lean();
  const localCities = (settings?.shipping?.codCities ?? ["ilorin"]).map((c: string) => c.toLowerCase());
  const cityLower = city.trim().toLowerCase();
  return localCities.includes(cityLower) ? LOCAL_DELIVERY_FEE : 0;
}

/**
 * Builds a human-readable summary of what the customer picked for each slot.
 * Example: "Casio Watch Â· Bluetooth Audio Glasses Â· Premium Bracelet"
 */
function buildVariantSummary(
  combo: any,
  selectedVariants: Record<string, string> | undefined,
): string {
  return (combo.items ?? [])
    .map((item: any) => {
      const chosenId = selectedVariants?.[item.id];
      const alt = chosenId ? (item.alternatives ?? []).find((a: any) => a.id === chosenId) : undefined;
      const name = alt ? alt.name : item.name; // default when nothing was chosen
      // include the chosen color so you know exactly what to pack
      const colorId = selectedVariants?.[`${item.id}__color`];
      const color = colorId ? (item.colors ?? []).find((c: any) => c.id === colorId) : undefined;
      return color ? `${name} (${color.name})` : name;
    })
    .join(" Â· ");
}

async function buildItemSnapshots(
  items: CartItem[],
  priorSpend = 0,
): Promise<OrderItemSnapshot[]> {
  if (!items?.length) {
    throw new OrderError(400, "EMPTY_CART", "Order has no items");
  }

  // Pass 1: Calculate subtotal from paid items only
  let paidSubtotal = 0;
  const freeGiftItems = items.filter((it) => it.isFreeGift);

  if (freeGiftItems.length > 1) {
    throw new OrderError(400, "INVALID_GIFT", "Only 1 free gift is allowed per order");
  }

  for (const cartItem of items) {
    if (!cartItem.isFreeGift) {
      if (!cartItem.comboId) {
        throw new OrderError(400, "INVALID_ITEM", "Item missing combo reference");
      }

      const combo = await Combo.findById(cartItem.comboId);
      if (combo && combo.isActive) {
        paidSubtotal += unitPriceFor(combo, cartItem.quantity) * cartItem.quantity;
      }
    }
  }

  // Validate threshold if a free gift is attached
  // Earlier paid spending by the same phone number counts too.
  const giftUnlocked = paidSubtotal + priorSpend >= GIFT_THRESHOLD;
  if (freeGiftItems.length === 1 && !giftUnlocked) {
    throw new OrderError(
      400,
      "GIFT_THRESHOLD_NOT_MET",
      "Your spending (this order plus the last 30 days) must reach â‚¦40,000 to qualify for a free gift",
    );
  }

  const snapshots: OrderItemSnapshot[] = [];

  // Pass 2: Build snapshot objects
  for (const cartItem of items) {
    if (!cartItem.comboId) {
      throw new OrderError(400, "INVALID_ITEM", "Item missing combo reference");
    }
    if (!Number.isInteger(cartItem.quantity) || cartItem.quantity < 1 || cartItem.quantity > BULK_MAX_QTY) {
      throw new OrderError(400, "INVALID_ITEM", "Item quantity must be a whole number of 1 or more");
    }

    // Free gifts normally come from the Gift Bank
    if (cartItem.isFreeGift) {
      const gift: any = await Gift.findById(cartItem.comboId).catch(() => null);
      if (gift) {
        if (gift.isActive === false || (gift.stock ?? 0) < 1) {
          throw new OrderError(409, "INSUFFICIENT_STOCK", `${gift.name} is no longer available`);
        }
        snapshots.push({
          comboId: gift._id,
          slug: `gift-${gift._id}`,
          name: `[FREE GIFT] ${gift.name}`,
          tagline: gift.description ?? "",
          thumbnailUrl: gift.images?.[0] || gift.image || "",
          unitPrice: 0,
          quantity: 1,
          subtotal: 0,
          selectedVariants: {},
          variantSummary: "",
          isFreeGift: true,
        });
        continue;
      }
    }

    const combo = await Combo.findById(cartItem.comboId);
    if (!combo) {
      throw new OrderError(404, "COMBO_NOT_FOUND", `Combo not found: ${cartItem.comboId}`);
    }
    if (!combo.isActive) {
      throw new OrderError(400, "COMBO_INACTIVE", `${combo.name} is no longer available`);
    }
    if (cartItem.quantity > maxQtyFor(combo)) {
      throw new OrderError(
        400,
        "INVALID_ITEM",
        `You can order up to ${maxQtyFor(combo)} of ${combo.name} at a time`,
      );
    }
    // Pre-order: the switch is on and there is no stock. Nothing is taken from stock; you buy it after payment.
    const isPre = Boolean((combo as any).preorder) && combo.stockLeft === 0;
    if (!isPre && combo.stockLeft < cartItem.quantity) {
      throw new OrderError(
        409,
        "INSUFFICIENT_STOCK",
        `Only ${combo.stockLeft} of ${combo.name} available`,
      );
    }

    const isFree = Boolean(cartItem.isFreeGift && giftUnlocked);

    // Ensure the combo itself is eligible to be a free gift candidate
    if (isFree && !combo.isFreeGiftCandidate && combo.totalPrice > 5000) {
      throw new OrderError(
        400,
        "INELIGIBLE_FREE_GIFT",
        `${combo.name} is not eligible as a free gift`,
      );
    }

    const unitPrice = isFree ? 0 : unitPriceFor(combo, cartItem.quantity);
    const subtotal = unitPrice * cartItem.quantity;

    const thumbnailUrl = combo.items?.[0]?.images?.[0]?.url ?? "";
    const variantSummary = buildVariantSummary(combo, cartItem.selectedVariants);

    snapshots.push({
      comboId: combo._id,
      slug: combo.slug,
      name: isFree ? `[FREE GIFT] ${combo.name}` : combo.name,
      tagline: combo.tagline,
      thumbnailUrl,
      unitPrice,
      quantity: isFree ? 1 : cartItem.quantity,
      subtotal,
      selectedVariants: cartItem.selectedVariants ?? {},
      variantSummary,
      isFreeGift: isFree,
      ...(isPre
        ? {
            preorder: true,
            leadDaysMin: Number((combo as any).leadDaysMin) || 5,
            leadDaysMax: Number((combo as any).leadDaysMax) || 10,
          }
        : {}),
    });
  }

  return snapshots;
}

/** Takes one item's stock. Says where it came from, or throws if sold out. */
async function takeStockForItem(item: OrderItemSnapshot): Promise<"gift" | "combo"> {
  // Pre-order line: there is no stock to take
  if (item.preorder) return "combo";
  // Gift Bank gifts have their own stock
  if (item.isFreeGift) {
    const gift = await Gift.findOneAndUpdate(
      { _id: item.comboId, stock: { $gte: 1 } },
      { $inc: { stock: -1 } },
    );
    if (gift) return "gift";
  }
  const updated = await Combo.findOneAndUpdate(
    { _id: item.comboId, stockLeft: { $gte: item.quantity } },
    { $inc: { stockLeft: -item.quantity } },
  );
  if (!updated) {
    throw new OrderError(409, "INSUFFICIENT_STOCK", `${item.name} sold out before checkout`);
  }
  return "combo";
}

async function giveStockBackForItem(item: OrderItemSnapshot, kind?: "gift" | "combo"): Promise<void> {
  // Pre-order line: nothing was taken, so nothing goes back
  if (item.preorder) return;
  if (kind === "combo") {
    await Combo.updateOne({ _id: item.comboId }, { $inc: { stockLeft: item.quantity } });
    return;
  }
  if (kind === "gift" || item.isFreeGift) {
    const r = await Gift.updateOne({ _id: item.comboId }, { $inc: { stock: 1 } });
    if (r.matchedCount > 0) return;
  }
  await Combo.updateOne({ _id: item.comboId }, { $inc: { stockLeft: item.quantity } });
}

/** All-or-nothing: if any item is sold out, everything already taken is put back. */
async function decrementStock(items: OrderItemSnapshot[]): Promise<void> {
  const taken: Array<{ item: OrderItemSnapshot; kind: "gift" | "combo" }> = [];
  try {
    for (const item of items) {
      const kind = await takeStockForItem(item);
      taken.push({ item, kind });
    }
  } catch (err) {
    for (const t of taken) {
      await giveStockBackForItem(t.item, t.kind).catch((e) =>
        console.error("Stock rollback failed:", e),
      );
    }
    throw err;
  }
}

function buildEmailContext(order: any) {
  return {
    orderNumber: order.orderNumber,
    customerName: order.shipping.fullName,
    customerEmail: order.shipping.email,
    total: order.total,
    items: order.items.map((it: any) => ({
      name: it.variantSummary || it.name,
      quantity: it.quantity,
      subtotal: it.subtotal,
    })),
    shippingAddress: {
      fullName: order.shipping.fullName,
      street: order.shipping.street,
      city: order.shipping.city,
      state: order.shipping.state,
      phone: order.shipping.phone,
    },
    paymentMethod: order.paymentMethod,
    trackingUrl: `${config.siteUrl}/order-tracking?order=${encodeURIComponent(order.orderNumber)}`,
  };
}

/** Extra options when the owner records a sale by hand (WhatsApp, walk-in...). */
export interface ManualSaleOptions {
  source: "whatsapp" | "walkin" | "social" | "other";
  /** payment already received */
  paid: boolean;
  /** customer collects the goods: no delivery fee */
  pickup: boolean;
}

export async function createOrder(input: CreateOrderInput, manual?: ManualSaleOptions) {
  const phoneKey = normalizePhone(input.shipping.phone);
  const priorSpend = await getPriorSpend(phoneKey);
  const items = await buildItemSnapshots(input.items, priorSpend);

  const subtotal = items.reduce((sum, it) => sum + it.subtotal, 0);
  // Bulk orders: one flat Ilorin delivery fee for the whole order, and they must be paid online.
  const bulk = await orderHasBulk(input.items);
  if (bulk && !manual && input.paymentMethod !== "paystack") {
    throw new OrderError(400, "BULK_PAY_ONLINE", "Bulk orders must be paid online. Please choose Pay online.");
  }
  const baseFee = manual?.pickup
    ? 0
    : await calculateShippingFee(
        input.paymentMethod,
        input.shipping.city,
      );
  const shippingFee = bulk && baseFee > 0 ? BULK_DELIVERY_FEE : baseFee;

  // Pre-order items: the customer must tick the box and pay online (you buy the goods after payment).
  const hasPre = items.some((i) => i.preorder);
  if (hasPre && !manual) {
    if (!input.ackPreorder) {
      throw new OrderError(400, "PREORDER_ACK", "Please tick the box to say you understand some items are pre-order.");
    }
    if (input.paymentMethod !== "paystack") {
      throw new OrderError(400, "PREORDER_PAY_ONLINE", "Pre-order items must be paid online. Please choose Pay online.");
    }
  }
  const total = subtotal + shippingFee;

  const orderNumber = await generateOrderNumber();

  await decrementStock(items);

  const initialStatus: OrderStatus = manual
    ? manual.paid
      ? "paid"
      : "pending"
    : input.paymentMethod === "cod"
      ? "processing"
      : "pending";

  let order;
  try {
    order = await Order.create({
      orderNumber,
      items,
      subtotal,
      shippingFee,
      total,
      isBulk: bulk,
      hasPreorder: hasPre,
      awaitingGoods: hasPre,
      status: initialStatus,
      paymentMethod: input.paymentMethod,
      shipping: input.shipping,
      phoneKey,
      notes: input.notes,
      ...(manual
        ? {
            source: manual.source,
            manual: true,
            ...(manual.paid ? { paidAt: new Date(), paymentReference: `manual-${input.paymentMethod}` } : {}),
          }
        : {}),
    });
  } catch (err) {
    // The order could not be saved: put the stock back so nothing is lost.
    for (const item of items) {
      await giveStockBackForItem(item).catch((e) => console.error("Stock rollback failed:", e));
    }
    throw err;
  }

  // Remember what these goods cost you right now, so profit stays correct if prices change later.
  await recordOrderCost(order);

  // A sale recorded by hand needs no "order placed" email: you are already talking to the customer.
  if (!manual) {
    const ctx = buildEmailContext(order);
    sendOrderPlaced(ctx).catch(() => {});
    sendNewOrderAdminAlert(ctx).catch(() => {});
  }

  return order;
}

/**
 * A Paystack payment only counts for an order when the reference was started for THAT order,
 * the payment succeeded, and the amount paid equals the order total (in kobo).
 * Without this check a customer could pay a small amount and have a big order marked paid.
 */
export function paymentMatchesOrder(
  order: { orderNumber: string; total: number },
  v: { status: string; amount: number; currency?: string },
  reference: string,
): boolean {
  return (
    v.status === "success" &&
    v.amount === Math.round(order.total * 100) &&
    (!v.currency || v.currency === "NGN") &&
    reference.startsWith(`${order.orderNumber}-`)
  );
}

export async function markOrderPaid(
  orderNumber: string,
  paymentReference: string,
): Promise<any> {
  const number = orderNumber.trim().toUpperCase();

  // Atomic claim: only ONE caller (webhook, return-redirect or admin) can flip
  // pending -> paid, so emails are never doubled.
  const claimed = await Order.findOneAndUpdate(
    { orderNumber: number, status: "pending" },
    { $set: { status: "paid", paidAt: new Date(), paymentReference } },
    { new: true },
  );
  if (claimed) {
    sendPaymentConfirmed(buildEmailContext(claimed)).catch(() => {});
    return claimed;
  }

  const order = await Order.findOne({ orderNumber: number });
  if (!order) return null;

  // Customer paid AFTER the order was auto-cancelled: take the stock again if it is still there.
  if (order.status === "cancelled" && order.stockRestored) {
    const revived = await Order.findOneAndUpdate(
      { orderNumber: number, status: "cancelled", stockRestored: true },
      { $set: { status: "paid", paidAt: new Date(), paymentReference, stockRestored: false } },
      { new: true },
    );
    if (revived) {
      try {
        await decrementStock(revived.items as any);
      } catch {
        revived.stockRestored = true;
        revived.adminNotes =
          `${revived.adminNotes ? revived.adminNotes + "\n" : ""}LATE PAYMENT: customer paid after the order was cancelled and an item is now out of stock. Ship when restocked or refund on Paystack.`;
        await revived.save();
      }
      sendPaymentConfirmed(buildEmailContext(revived)).catch(() => {});
      return revived;
    }
  }

  // Already paid / processing / shipped / delivered / refunded: nothing to redo
  if (!order.paymentReference && order.status !== "cancelled") {
    order.paymentReference = paymentReference;
    await order.save();
  }
  return order;
}

/**
 * Cancels or refunds an order and gives its stock back, exactly once.
 *  - cancelled: always puts stock back (if not already).
 *  - refunded : puts stock back only if the order had not shipped yet
 *               (shipped/delivered refunds mean goods come back - adjust stock by hand after checking them).
 */
export async function cancelOrRefundOrder(
  orderNumber: string,
  newStatus: "cancelled" | "refunded",
  note?: string,
): Promise<any> {
  const number = orderNumber.trim().toUpperCase();
  const current = await Order.findOne({ orderNumber: number });
  if (!current) return null;

  const wasShipped = current.status === "shipped" || current.status === "delivered";
  const shouldRestock = newStatus === "cancelled" || !wasShipped;
  const noteLine = note ? `${current.adminNotes ? current.adminNotes + "\n" : ""}${note}` : undefined;

  if (shouldRestock) {
    const claimed = await Order.findOneAndUpdate(
      { orderNumber: number, stockRestored: { $ne: true }, status: { $nin: ["cancelled", "refunded"] } },
      { $set: { status: newStatus, stockRestored: true, ...(noteLine ? { adminNotes: noteLine } : {}) } },
      { new: true },
    );
    if (claimed) {
      for (const item of claimed.items as any[]) {
        await giveStockBackForItem(item).catch((e) => console.error("Restock failed:", e));
      }
      return claimed;
    }
  }

  // Already cancelled/refunded (or no restock wanted): only the label changes
  return Order.findOneAndUpdate(
    { orderNumber: number },
    { $set: { status: newStatus, ...(noteLine ? { adminNotes: noteLine } : {}) } },
    { new: true },
  );
}

/**
 * Cancels Paystack orders that were never paid so abandoned checkouts do not
 * lock stock. Before cancelling, it asks Paystack whether the customer actually paid.
 */
export async function expireStalePendingOrders(maxAgeMinutes = 180): Promise<number> {
  const cutoff = new Date(Date.now() - maxAgeMinutes * 60 * 1000);
  const stale = await Order.find({
    status: "pending",
    paymentMethod: { $in: ["paystack", "flutterwave"] },
    createdAt: { $lt: cutoff },
  })
    .sort({ createdAt: 1 })
    .limit(50);

  let cancelled = 0;
  for (const order of stale) {
    try {
      const refs: string[] = (order as any).paystackRefs ?? [];
      let verdict: "paid" | "unpaid" | "unknown" = "unknown";
      if (refs.length === 0) {
        verdict = "unpaid"; // customer never reached the Paystack page
      } else {
        let paid = false;
        let unsure = false;
        for (const ref of refs) {
          try {
            const v = await verifyTransaction(ref);
            if (paymentMatchesOrder(order, v, ref)) {
              await markOrderPaid(order.orderNumber, ref);
              paid = true;
              break;
            }
            if (v.status === "success") {
              console.error(`[expire] ${order.orderNumber}: paid amount ${v.amount} does not match total, left pending for review`);
              unsure = true;
            }
          } catch (err) {
            // 404 = Paystack has no such payment (never started); anything else = could not check, so do not cancel
            if (!(err instanceof PaystackError && err.status === 404)) unsure = true;
          }
        }
        verdict = paid ? "paid" : unsure ? "unknown" : "unpaid";
      }

      if (verdict === "unpaid") {
        await cancelOrRefundOrder(
          order.orderNumber,
          "cancelled",
          "Auto-cancelled: payment was not completed in time. Stock returned.",
        );
        cancelled++;
      }
    } catch (err) {
      console.error(`[expire] ${order.orderNumber} failed:`, err);
    }
  }
  return cancelled;
}

export async function markOrderShipped(
  orderNumber: string,
  trackingNumber?: string,
  trackingProviderUrl?: string,
): Promise<any> {
  const order = await Order.findOne({
    orderNumber: orderNumber.trim().toUpperCase(),
  });
  if (!order) return null;

  order.status = "shipped";
  order.shippedAt = new Date();
  if (trackingNumber) order.trackingNumber = trackingNumber;
  if (trackingProviderUrl) order.trackingProviderUrl = trackingProviderUrl;
  await order.save();

  const ctx = buildEmailContext(order);
  sendOrderShipped({ ...ctx, trackingNumber, trackingProviderUrl }).catch(() => {});

  return order;
}

export async function markOrderDelivered(orderNumber: string): Promise<any> {
  const order = await Order.findOne({
    orderNumber: orderNumber.trim().toUpperCase(),
  });
  if (!order) return null;

  order.status = "delivered";
  order.deliveredAt = new Date();
  await order.save();

  sendOrderDelivered(buildEmailContext(order)).catch(() => {});

  return order;
}

/**
 * Order model.
 *
 * Items now snapshot the customer's variant selections so admin knows
 * exactly what to ship.
 *
 * phoneKey is the customer's phone number cleaned to its last 10 digits. It is
 * how spending is added up for the free gift (customers do not have accounts).
 */

import { Schema, model, Document, Types } from "mongoose";

export type OrderStatus =
  | "pending"
  | "paid"
  | "processing"
  | "shipped"
  | "delivered"
  | "cancelled"
  | "refunded";

export type OrderSource = "website" | "whatsapp" | "walkin" | "social" | "other";

export type PaymentMethod =
  | "paystack"
  | "flutterwave"
  | "bank_transfer"
  | "cod"
  | "whatsapp"
  | "cash"
  | "pos";

export interface OrderItemSnapshot {
  comboId: Types.ObjectId;
  slug: string;
  name: string;
  tagline?: string;
  thumbnailUrl?: string;
  unitPrice: number;
  quantity: number;
  subtotal: number;
  selectedVariants?: Record<string, string>;   // itemId â†’ alternativeId
  variantSummary?: string;                     // human-readable
  isFreeGift?: boolean;                        // identifies free gift items
  preorder?: boolean;                          // pre-order line: nothing was taken from stock
  leadDaysMin?: number;
  leadDaysMax?: number;
}

const OrderItemSnapshotSchema = new Schema<OrderItemSnapshot>(
  {
    comboId: { type: Schema.Types.ObjectId, ref: "Combo", required: true },
    slug: { type: String, required: true },
    name: { type: String, required: true },
    tagline: { type: String },
    thumbnailUrl: { type: String },
    unitPrice: { type: Number, required: true, min: 0 },
    quantity: { type: Number, required: true, min: 1 },
    subtotal: { type: Number, required: true, min: 0 },
    selectedVariants: { type: Schema.Types.Mixed, default: {} },
    variantSummary: { type: String, default: "" },
    isFreeGift: { type: Boolean, default: false },

import { Router, Request, Response } from "express";
import {
  createOrder,
  findByNumberAndPhone,

import { attachReferral } from "../services/referralService";

  GIFT_THRESHOLD,
  GIFT_WINDOW_DAYS,
  getPriorSpend,
  normalizePhone,

const router = Router();

const ALLOWED_METHODS = ["paystack", "flutterwave", "bank_transfer", "cod", "whatsapp"];

// Order creation limiter: 10 orders per minute per visitor, so nobody can drain stock by spamming.
const orderHits = new Map<string, { count: number; resetAt: number }>();
function orderLimited(ip: string): boolean {
  const now = Date.now();
  const hit = orderHits.get(ip);
  if (!hit || hit.resetAt < now) {
    orderHits.set(ip, { count: 1, resetAt: now + 60_000 });
    return false;
  }
  hit.count += 1;
  return hit.count > 10;
}

// Small in-memory limiter for gift-progress: 30 lookups per minute per IP.
const progressHits = new Map<string, { count: number; resetAt: number }>();
function progressLimited(ip: string): boolean {
  const now = Date.now();
  const hit = progressHits.get(ip);
  if (!hit || hit.resetAt < now) {
    progressHits.set(ip, { count: 1, resetAt: now + 60_000 });
    return false;
  }
  hit.count += 1;
  return hit.count > 30;
}

router.get("/gift-progress", async (req: Request, res: Response) => {
  try {
    if (progressLimited(req.ip ?? "unknown")) {
      return res.status(429).json({
        error: { code: "RATE_LIMITED", message: "Too many requests. Try again in a minute." },
      });
    }
    const phoneKey = normalizePhone(req.query.phone);
    const spent = await getPriorSpend(phoneKey);
    // Only the total is returned, never order details.
    return res.json({
      data: {
        spent,
        threshold: GIFT_THRESHOLD,
        remaining: Math.max(0, GIFT_THRESHOLD - spent),
        windowDays: GIFT_WINDOW_DAYS,
      },
    });
  } catch (err) {
    console.error("GET /api/orders/gift-progress failed:", err);
    return res.status(500).json({
      error: { code: "SERVER_ERROR", message: "Could not check gift progress" },
    });
  }
});

router.post("/", async (req: Request, res: Response) => {
  try {
    if (orderLimited(req.ip ?? "unknown")) {
      return res.status(429).json({
        error: { code: "RATE_LIMITED", message: "Too many orders. Please wait a minute and try again." },
      });
    }

    const { items, shipping, paymentMethod, notes } = req.body;

    if (!Array.isArray(items) || items.length === 0 || items.length > 20) {
      return res.status(400).json({
        error: { code: "BAD_REQUEST", message: "Your cart is empty or has too many items" },
      });
    }

    // Basic input shape validation
    if (!shipping || typeof shipping !== "object") {
      return res.status(400).json({
        error: { code: "BAD_REQUEST", message: "Missing shipping details" },
      });
    }
    const required: (keyof typeof shipping)[] = [
      "fullName",
      "email",
      "phone",
      "state",
      "city",
      "street",
    ];
    for (const field of required) {
      if (!shipping[field] || typeof shipping[field] !== "string") {
        return res.status(400).json({
          error: { code: "BAD_REQUEST", message: `Missing field: ${field}` },
        });
      }
    }

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(shipping.email.trim()) || shipping.email.length > 200) {
      return res.status(400).json({
        error: { code: "BAD_REQUEST", message: "Please enter a valid email address" },
      });
    }

    if (!paymentMethod || typeof paymentMethod !== "string" || !ALLOWED_METHODS.includes(paymentMethod)) {
      return res.status(400).json({
        error: { code: "BAD_REQUEST", message: "Choose a valid payment method" },
      });
    }

    const order = await createOrder({ items, shipping, paymentMethod, notes, ackPreorder: req.body.ackPreorder === true });
    await attachReferral(order, req.body.ref);

    return res.status(201).json({ data: order });
  } catch (err: any) {
    if (err instanceof OrderError) {
      return res.status(err.status).json({
        error: { code: err.code, message: err.message },
      });
    }
    console.error("POST /api/orders failed:", err);
    return res.status(500).json({
      error: { code: "SERVER_ERROR", message: "Failed to create order" },
    });
  }
});

/**
 * "To get": pre-order orders that are paid but whose goods you still have to buy.
 *
 * GET  /api/admin/to-get              what to buy (totals per product) + the orders waiting
 * POST /api/admin/to-get/:id/got-it   the goods arrived: the order moves to Fulfilment > To pack
 *
 * Until you press Got it, the order is kept out of the packing queue.
 */
import { Router, Request, Response } from "express";
import { Order } from "../../models/Order";

const router = Router();

/** Adds business days (Monday to Friday) to a date. */
function addBusinessDays(from: Date, days: number): Date {
  const d = new Date(from);
  let left = Math.max(0, Math.floor(days));
  while (left > 0) {
    d.setDate(d.getDate() + 1);
    const wd = d.getDay();
    if (wd !== 0 && wd !== 6) left--;
  }
  return d;
}

const WAITING = { awaitingGoods: true, status: { $in: ["paid", "processing"] } };

router.get("/", async (_req: Request, res: Response) => {
  try {
    const orders = await Order.find(WAITING).sort({ paidAt: 1, createdAt: 1 }).limit(300).lean();
    const totals = new Map<string, { name: string; quantity: number; orders: number }>();
    const now = Date.now();
    const shaped = orders.map((o: any) => {
      const lines = (o.items ?? []).filter((i: any) => i.preorder);
      const maxDays = Math.max(0, ...lines.map((i: any) => Number(i.leadDaysMax) || 0));
      const paid = new Date(o.paidAt || o.createdAt);
      const promisedBy = addBusinessDays(paid, maxDays);
      for (const i of lines) {
        const key = `${i.comboId}`;
        const t = totals.get(key) ?? { name: i.name, quantity: 0, orders: 0 };
        t.quantity += i.quantity;
        t.orders += 1;
        totals.set(key, t);
      }
      return {
        _id: String(o._id),
        orderNumber: o.orderNumber,
        customer: o.shipping?.fullName ?? "",
        phone: o.shipping?.phone ?? "",
        paidAt: paid,
        promisedBy,
        late: promisedBy.getTime() < now,
        items: lines.map((i: any) => ({
          name: i.name,
          variantSummary: i.variantSummary ?? "",
          quantity: i.quantity,
          leadDaysMin: Number(i.leadDaysMin) || 0,
          leadDaysMax: Number(i.leadDaysMax) || 0,
        })),
      };
    });
    res.json({ data: { list: [...totals.values()].sort((a, b) => b.quantity - a.quantity), orders: shaped } });
  } catch (err) {
    console.error("GET /api/admin/to-get failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to load the list" } });
  }
});

router.post("/:id/got-it", async (req: Request, res: Response) => {
  try {
    const o = await Order.findOneAndUpdate(
      { _id: req.params.id, ...WAITING },
      { $set: { awaitingGoods: false } },
      { new: true },
    ).lean();
    if (!o) {
      return res.status(409).json({
        error: { code: "BAD_REQUEST", message: "This order is not waiting for goods (already done, unpaid, or closed)." },
      });
    }
    res.json({ data: { _id: String(o._id), orderNumber: o.orderNumber } });
  } catch (err: any) {
    if (err.name === "CastError") return res.status(404).json({ error: { code: "NOT_FOUND", message: "Order not found" } });
    console.error("to-get got-it failed:", err);

export interface CustomComboGroup {
  id: string;
  /** Flat â‚¦ amount off, applied once for this whole group â€” not per item. */
  discount: number;
  /** When the group was created, for display ordering / debugging. */
  createdAt: number;

/** Minimal shape needed to add ANY free gift as a cart line item â€” decoupled

export interface FreeGiftSource {
  id: string;
  name: string;
  slug?: string;
  image: string;
  /** Estimated/original value, shown struck-through â€” not charged. */

export const GIFT_THRESHOLD = 40000;

interface CartStore {
  items: CartItem[];
  customComboGroups: CustomComboGroup[];
  isOpen: boolean;
  /** Earlier paid spending that counts toward the gift. Set by Checkout, not saved. */
  giftCredit: number;
  setGiftCredit: (amount: number) => void;

  addItem: (
    combo: Combo,
    quantity?: number,
    selectedVariants?: Record<string, string>,
  ) => void;

  /** Adds 2-3 products as a linked "Build Your Own Combo" group. */
  addCustomCombo: (products: Combo[], discount: number) => void;

  /** Free Gift Actions */
  selectFreeGift: (gift: FreeGiftSource) => void;
  removeFreeGift: () => void;
  selectedFreeGift: () => CartItem | undefined;
  qualifiesForFreeGift: () => boolean;

  removeItem: (comboId: string, comboGroupId?: string) => void;
  updateQuantity: (comboId: string, quantity: number, comboGroupId?: string) => void;
  clear: () => void;

  openCart: () => void;
  closeCart: () => void;
  toggleCart: () => void;

  itemCount: () => number;
  /** Sum of unitPrice * quantity across all items, before any custom-combo discount. */
  itemsSubtotal: () => number;
  /** Total of all active custom-combo group discounts. */
  customComboDiscountTotal: () => number;
  /** itemsSubtotal() minus customComboDiscountTotal() â€” what the customer actually pays for items. */
  subtotal: () => number;
  originalTotal: () => number;
  savings: () => number;
}

function makeGroupId(): string {
  return `combo-group-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
}

export const useCart = create<CartStore>()(
  persist(
    (set, get) => ({
      items: [],
      customComboGroups: [],
      isOpen: false,
      giftCredit: 0,

      setGiftCredit: (amount) => {
        const credit = Math.max(0, Math.floor(amount) || 0);
        const stillQualifies = get().subtotal() + credit >= GIFT_THRESHOLD;
        set({
          giftCredit: credit,
          items: stillQualifies ? get().items : get().items.filter((i) => !i.isFreeGift),
        });
      },

      addItem: (combo, quantity = 1, selectedVariants) => {
        const existing = get().items.find(
          (i) => i.comboId === combo.id && !i.comboGroupId && !i.isFreeGift,
        );
        if (existing) {
          set({
            items: get().items.map((i) =>
              i.comboId === combo.id && !i.comboGroupId && !i.isFreeGift
                ? {
                    ...i,
                    quantity: i.quantity + quantity,
                    unitPrice: priceForQty(combo.totalPrice, tiersOf(combo), i.quantity + quantity),
                    ...({ bulkTiers: tiersOf(combo), basePrice: combo.totalPrice, ...preorderFlags(combo) } as object),
                    selectedVariants: selectedVariants ?? i.selectedVariants,
                  }
                : i,
            ),
          });
        } else {
          set({
            items: [
              ...get().items,
              {
                comboId: combo.id,
                comboName: combo.name,
                comboSlug: combo.slug,
                unitPrice: priceForQty(combo.totalPrice, tiersOf(combo), quantity),
                originalPrice: combo.originalPrice,
                quantity,
                ...({ bulkTiers: tiersOf(combo), basePrice: combo.totalPrice, ...preorderFlags(combo) } as object),
                image: combo.items[0]?.images[0]?.url ?? '',
                selectedVariants,
              },
            ],
          });
        }
      },

      addCustomCombo: (products, discount) => {
        if (products.length < 2) return;
        const groupId = makeGroupId();

        const newItems: CartItem[] = products.map((combo) => ({
          comboId: combo.id,
          comboName: combo.name,
          comboSlug: combo.slug,
          unitPrice: combo.totalPrice,
          originalPrice: combo.originalPrice,
          quantity: 1,
          image: combo.items[0]?.images[0]?.url ?? '',
          comboGroupId: groupId,
        }));

        set({
          items: [...get().items, ...newItems],
          customComboGroups: [
            ...get().customComboGroups,
            { id: groupId, discount, createdAt: Date.now() },
          ],
        });
      },

      selectFreeGift: (gift) => {
        if (!get().qualifiesForFreeGift()) return;

        const giftItem: CartItem = {
          comboId: gift.id,
          comboName: gift.name,
          comboSlug: gift.slug ?? '',
          unitPrice: 0,
          originalPrice: gift.price ?? 0,
          quantity: 1,
          image: gift.image,
          isFreeGift: true,
        };

        // Remove any existing free gift line item before adding the new selection
        const cleanedItems = get().items.filter((i) => !i.isFreeGift);
        set({ items: [...cleanedItems, giftItem] });
      },

      removeFreeGift: () => {
        set({ items: get().items.filter((i) => !i.isFreeGift) });
      },

      selectedFreeGift: () => {
        return get().items.find((i) => i.isFreeGift);
      },

      qualifiesForFreeGift: () => {
        return get().subtotal() + get().giftCredit >= GIFT_THRESHOLD;
      },

      removeItem: (comboId, comboGroupId) => {
        const remaining = get().items.filter((i) =>
          comboGroupId ? !(i.comboId === comboId && i.comboGroupId === comboGroupId) : i.comboId !== comboId,
        );

        let groups = get().customComboGroups;
        if (comboGroupId) {
          const groupStillHasItems = remaining.some((i) => i.comboGroupId === comboGroupId);
          if (!groupStillHasItems) {
            groups = groups.filter((g) => g.id !== comboGroupId);
          }
        }

        // Enforce guardrail: if removal causes subtotal to drop below 40k, remove free gift automatically
        const nonGiftSubtotal = remaining.reduce((sum, i) => {
          if (i.isFreeGift) return sum;
          return sum + i.unitPrice * i.quantity;
        }, 0) - groups.reduce((sum, g) => sum + g.discount, 0) + get().giftCredit;

        const finalItems = nonGiftSubtotal < GIFT_THRESHOLD ? remaining.filter((i) => !i.isFreeGift) : remaining;

        set({ items: finalItems, customComboGroups: groups });
      },

      updateQuantity: (comboId, quantity, comboGroupId) => {
        if (quantity <= 0) {
          get().removeItem(comboId, comboGroupId);
          return;
        }

        const updatedItems = get().items.map((i) =>
          i.comboId === comboId && (comboGroupId ? i.comboGroupId === comboGroupId : !i.comboGroupId) && !i.isFreeGift
            ? { ...i, quantity, unitPrice: repriceLine(i, quantity) }
            : i,
        );

        const nonGiftSubtotal = updatedItems.reduce((sum, i) => {
          if (i.isFreeGift) return sum;
          return sum + i.unitPrice * i.quantity;
        }, 0) - get().customComboDiscountTotal() + get().giftCredit;

        const finalItems = nonGiftSubtotal < GIFT_THRESHOLD ? updatedItems.filter((i) => !i.isFreeGift) : updatedItems;

        set({ items: finalItems });
      },

      clear: () => set({ items: [], customComboGroups: [], giftCredit: 0 }),

      openCart: () => set({ isOpen: true }),
      closeCart: () => set({ isOpen: false }),
      toggleCart: () => set({ isOpen: !get().isOpen }),

      itemCount: () => get().items.reduce((sum, i) => sum + i.quantity, 0),

      itemsSubtotal: () =>
        get().items.reduce((sum, i) => (i.isFreeGift ? sum : sum + i.unitPrice * i.quantity), 0),

      customComboDiscountTotal: () =>
        get().customComboGroups.reduce((sum, g) => sum + g.discount, 0),

      subtotal: () => get().itemsSubtotal() - get().customComboDiscountTotal(),

      originalTotal: () =>
        get().items.reduce((sum, i) => sum + (i.originalPrice || i.unitPrice) * i.quantity, 0),

      savings: () => get().originalTotal() - get().subtotal(),
    }),
    {
      name: 'smart-combo-cart',
      partialize: (state) => ({ items: state.items, customComboGroups: state.customComboGroups }),
    },
  ),
);

/**
 * Orders API - frontend client.
 *
 * Cart items include selectedVariants and the free-gift flag when sent to the backend.
 */

import { apiFetch } from "./client";
import type { Order, ShippingAddress, PaymentMethod, CartItem } from "@/types/order";
import { getReferral } from "@/lib/referral";

interface ApiResponse<T> {
  data: T;
}

interface BackendOrder extends Omit<Order, "id"> {
  _id: string;
}

function normalize(raw: BackendOrder): Order {
  const { _id, ...rest } = raw;
  return { ...rest, id: _id };
}

export interface CreateOrderInput {
  items: CartItem[];
  shipping: ShippingAddress;
  paymentMethod: PaymentMethod;
  notes?: string;
  ackPreorder?: boolean;
}

export async function createOrder(input: CreateOrderInput): Promise<Order> {
  // Send each cart item with its selectedVariants and free-gift flag.
  // The server needs isFreeGift to know the item comes from the Gift Bank.
  const payload = {
    items: input.items.map((it) => ({
      comboId: it.comboId,
      quantity: it.quantity,
      selectedVariants: it.selectedVariants ?? {},
      isFreeGift: Boolean(it.isFreeGift),
    })),
    shipping: input.shipping,
    paymentMethod: input.paymentMethod,
    ref: getReferral(),
    notes: input.notes,
    ackPreorder: Boolean(input.ackPreorder),
  };

  const res = await apiFetch<ApiResponse<BackendOrder>>("/api/orders", {
    method: "POST",
    body: JSON.stringify(payload),
  });
  return normalize(res.data);
}

export async function trackOrder(orderNumber: string, phone: string): Promise<Order | null> {
  try {
    const res = await apiFetch<ApiResponse<BackendOrder>>(
      `/api/orders/track?orderNumber=${encodeURIComponent(orderNumber)}&phone=${encodeURIComponent(phone)}`,
    );
    return normalize(res.data);
  } catch (err: any) {
    if (err.status === 404) return null;
    throw err;
  }
}

    else chatUI.open();
  };

  return (
    <>
      <div
        className={`fixed bottom-6 right-6 z-[60] transition-all duration-500 ${
          visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-4'
        }`}
      >
        {!open && !tipSeen && (
          <div className="absolute right-full mr-3 top-1/2 -translate-y-1/2 pointer-events-none">
            <div className="relative whitespace-nowrap rounded-lg border border-border bg-card px-3 py-2 text-xs font-semibold text-card-foreground shadow-lg">
              Ask our AI assistant
              <span className="absolute left-full top-1/2 -translate-y-1/2 border-y-[5px] border-l-[6px] border-y-transparent border-l-border" />
            </div>
          </div>
        )}

        <button
          type="button"
          onClick={toggle}
          aria-label={open ? 'Close chat' : 'Open chat assistant'}
          aria-expanded={open}
          className="relative flex h-14 w-14 items-center justify-center rounded-full bg-gradient-to-br from-primary to-amber-500 text-primary-foreground shadow-2xl transition-transform hover:scale-110 active:scale-95"
          style={{ boxShadow: '0 8px 32px hsl(var(--primary) / 0.45)' }}
        >
          {!open && (
            <span className="absolute inset-0 rounded-full bg-primary/50 animate-ping [animation-duration:2.5s]" />
          )}
          {open ? <X className="relative h-6 w-6" /> : <Sparkles className="relative h-6 w-6" />}
        </button>
      </div>

      <ChatSheet />
    </>
  );
}

/**
 * "Buy more, pay less" box for a product page.
 * Shows the price levels, a quantity box, the total, and a button to add that many pieces.
 * It shows nothing for products that have no bulk prices.
 *
 * Plain words on purpose: one job (pick how many), big buttons, the total always visible.
 */
import { useState } from 'react';
import { useParams } from 'react-router-dom';
import { useQuery } from '@tanstack/react-query';
import { Minus, Plus, ShoppingBag, MessageCircle, Truck } from 'lucide-react';
import { formatNaira } from '@/lib/format';
import { fetchComboBySlug } from '@/api/combos';
import { useSettings } from '@/contexts/SettingsContext';
import { BULK_DELIVERY_FEE, priceForQty, tiersOf } from '@/lib/bulk';
import { preorderInfo } from '@/lib/preorder';
import type { Combo } from '@/types/combo';

const MAX_QTY = 1000;

export function BulkBox({ combo, onAdd }: { combo?: Combo; onAdd: (qty: number) => void }) {
  const { slug } = useParams<{ slug: string }>();
  const { settings } = useSettings();
  const { data: fetched } = useQuery({
    queryKey: ['combo', slug],
    queryFn: () => fetchComboBySlug(slug!),
    enabled: !combo && !!slug,
  });
  const c: any = combo ?? fetched;
  const tiers = tiersOf(c);
  const [qtyText, setQtyText] = useState('');
  const [touched, setTouched] = useState(false);

  if (!c || tiers.length === 0 || c.isActive === false) return null;

  const firstMin = tiers[0].minQty;
  const qty = !touched ? firstMin : Math.min(MAX_QTY, Math.max(1, parseInt(qtyText, 10) || 1));
  const unit = priceForQty(c.totalPrice, tiers, qty);
  const total = unit * qty;
  const stock = Number(c.stockLeft) || 0;
  const pre = preorderInfo(c);
  const tooMany = !pre.active && qty > stock;
  const isBulk = qty >= firstMin;

  const setQty = (n: number) => {
    setTouched(true);
    setQtyText(String(Math.min(MAX_QTY, Math.max(1, n))));
  };

  const waBase = settings?.contact?.whatsappLink ?? '#';
  const waUrl = `${waBase}${waBase.includes('?') ? '&' : '?'}text=${encodeURIComponent(
    `Hello Smart Combo, I want ${qty} pieces of ${c.name}.`,
  )}`;

  // Rows: the normal price, then each price level
  const rows = [
    { label: firstMin - 1 === 1 ? '1 piece' : `1 to ${firstMin - 1} pieces`, price: c.totalPrice, from: 1 },
    ...tiers.map((t) => ({ label: `${t.minQty}+ pieces`, price: t.unitPrice, from: t.minQty })),
  ];
  const activeFrom = [...rows].reverse().find((r) => qty >= r.from)?.from ?? 1;

  return (
    <div
      className="mt-6 rounded-2xl border border-primary/25 p-4 md:p-5"
      style={{ background: 'rgba(255,255,255,0.03)' }}
      data-testid="bulk-box"
    >
      <div className="flex items-baseline justify-between gap-2 flex-wrap">
        <h3 className="text-base font-black text-white">Buy more, pay less</h3>
        <span className="text-xs text-white/50">Price for each piece:</span>
      </div>

      <div className="mt-3 divide-y divide-white/10 rounded-xl border border-white/10 overflow-hidden">
        {rows.map((r) => (
          <div
            key={r.from}
            className={`flex items-center justify-between px-3 py-2.5 text-[15px] ${
              r.from === activeFrom ? 'bg-primary/10 text-white font-bold' : 'text-white/75'
            }`}
          >
            <span>{r.label}</span>
            <span className={r.from === activeFrom ? 'text-primary font-black' : 'font-semibold'}>
              {formatNaira(r.price)}
            </span>
          </div>
        ))}
      </div>

      <div className="mt-4 flex items-center justify-between gap-3">
        <span className="text-[15px] text-white/80 font-semibold">How many?</span>
        <div className="flex items-center rounded-xl border border-white/15">
          <button
            type="button"
            onClick={() => setQty(qty - 1)}
            aria-label="Fewer"
            className="w-12 h-12 flex items-center justify-center rounded-l-xl hover:bg-white/10"
          >
            <Minus className="w-5 h-5" />
          </button>
          <input
            type="number"
            inputMode="numeric"
            min={1}
            max={MAX_QTY}
            value={touched ? qtyText : String(firstMin)}
            onChange={(e) => {
              setTouched(true);
              setQtyText(e.target.value);
            }}
            onBlur={() => setQtyText(String(qty))}
            aria-label="Number of pieces"
            className="w-16 h-12 bg-transparent text-center text-lg font-black text-white focus:outline-none [appearance:textfield] [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"
          />
          <button
            type="button"
            onClick={() => setQty(qty + 1)}
            aria-label="More"
            className="w-12 h-12 flex items-center justify-center rounded-r-xl hover:bg-white/10"
          >
            <Plus className="w-5 h-5" />
          </button>
        </div>
      </div>

      <div className="mt-3 flex items-center justify-between rounded-xl bg-black/20 px-3 py-3">
        <span className="text-sm text-white/70">
          {qty} {qty === 1 ? 'piece' : 'pieces'} at {formatNaira(unit)}
        </span>
        <span className="text-xl font-black text-primary" data-testid="bulk-total">
          {formatNaira(total)}
        </span>
      </div>

      {isBulk && (
        <div className="mt-3 flex items-start gap-2 text-[13px] leading-relaxed text-white/60">
          <Truck className="w-4 h-4 shrink-0 mt-0.5 text-primary" />
          <p>
            Delivery in Ilorin: {formatNaira(BULK_DELIVERY_FEE)} for your whole bulk order. Elsewhere we send through
            Jumia. You pay Jumia when it arrives. We add no fee. Bulk orders are paid online.
          </p>
        </div>
      )}

      {pre.active && (
        <p className="mt-3 text-[13px] leading-relaxed text-primary/90">
          Pre-order: we get these for you after you pay. They arrive in {pre.min} to {pre.max} business days. If we cannot
          get them, you get your money back.
        </p>
      )}

      {tooMany && (
        <p className="mt-3 text-[13px] text-amber-300">
          {stock > 0
            ? `We have ${stock} ready. Please choose ${stock} or fewer, or get a price on WhatsApp.`
            : 'This item is out of stock right now.'}
        </p>
      )}

      <button
        type="button"
        onClick={() => onAdd(qty)}
        disabled={tooMany || (stock === 0 && !pre.active)}
        className="btn-primary mt-4 w-full min-h-[48px] flex items-center justify-center gap-2 text-[15px] font-bold disabled:opacity-50 disabled:cursor-not-allowed"
      >
        <ShoppingBag className="w-4 h-4" />
        {pre.active ? 'Pre-order' : 'Add'} {qty} {qty === 1 ? 'piece' : 'pieces'}{pre.active ? '' : ' to cart'}
      </button>

      <a
        href={waUrl}
        target="_blank"
        rel="noopener noreferrer"
        className="btn-secondary mt-2 w-full min-h-[48px] flex items-center justify-center gap-2 text-[15px] font-bold"
      >
        <MessageCircle className="w-4 h-4" />
        Get a price on WhatsApp
      </a>
    </div>
  );
}

===== client\lib\preorder.ts =====
/**
 * Pre-order helpers for the browser. A product is a pre-order item while its switch is on AND it has no stock.
 * The server decides for real when an order is placed.
 */
export interface PreorderInfo {
  active: boolean;
  min: number;
  max: number;
}

export function preorderInfo(x: any): PreorderInfo {
  const active = Boolean(x?.preorder) && (Number(x?.stockLeft) || 0) === 0;
  return {
    active,
    min: Math.max(1, Math.floor(Number(x?.leadDaysMin)) || 5),
    max: Math.max(1, Math.floor(Number(x?.leadDaysMax)) || 10),
  };
}

/** Flags saved on a cart line when it is added. */
export function preorderFlags(combo: any): { preorder: boolean; leadDaysMin: number; leadDaysMax: number } {
  const p = preorderInfo(combo);
  return { preorder: p.active, leadDaysMin: p.min, leadDaysMax: p.max };
}

export function lineIsPreorder(line: any): boolean {
  return !line?.isFreeGift && Boolean(line?.preorder);
}

/** "5 to 10" across all pre-order lines of a cart. */
export function cartPreorderWindow(lines: any[]): { min: number; max: number } {
  const pre = lines.filter(lineIsPreorder);
  if (pre.length === 0) return { min: 0, max: 0 };
  return {
    min: Math.min(...pre.map((l) => Number(l.leadDaysMin) || 5)),
    max: Math.max(...pre.map((l) => Number(l.leadDaysMax) || 10)),
  };
}
