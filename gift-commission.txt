import { Schema, model, Document } from 'mongoose';

export interface IGift extends Document {
  name: string;
  description?: string;
  image: string;       // Primary cover image
  images: string[];    // Array of Cloudinary URLs for the carousel slider
  price: number;
  stock: number;
  isActive: boolean;
  createdAt: Date;
  updatedAt: Date;
}

const giftSchema = new Schema<IGift>(
  {
    name: { type: String, required: true },
    description: { type: String, default: '' },
    image: { type: String, required: true },
    images: { type: [String], default: [] },
    price: { type: Number, default: 0 },
    stock: { type: Number, default: 10 },
    isActive: { type: Boolean, default: true },
  },
  { timestamps: true }
);

export const Gift = model<IGift>('Gift', giftSchema);
/**
 * Commission records + reseller program settings.
 *
 * A Commission row is created once, when an order that came through a reseller's
 * referral link is placed. It stores the numbers that were true at that moment
 * (base, rate, holdDays) so changing the program settings later never rewrites history.
 *
 * Status is worked out from the order and paidAt every time it is needed:
 *   order cancelled / refunded                      -> cancelled
 *   paidAt is set                                   -> paid
 *   order delivered and holdDays have passed        -> approved
 *   anything else                                   -> pending
 * (names match the Partner Program plan: Pending, Approved, Paid, Cancelled)
 * That way cancelling or refunding an order anywhere in the app needs no extra code here.
 *
 * Destination: server/models/Commission.ts
 */

import mongoose, { Schema, Types } from "mongoose";

export interface ICommission {
  _id: Types.ObjectId;
  orderId: Types.ObjectId;
  orderNumber: string;
  resellerId: Types.ObjectId;
  resellerCode: string;
  /** Amount the percentage was applied to (products only, shipping excluded), in naira */
  base: number;
  /** Blended percent actually applied (amount / base), for display, e.g. 5 or 7.5 */
  rate: number;
  /** How the amount was built: money and percent for combos vs single products */
  comboBase: number;
  comboRate: number;
  singleBase: number;
  singleRate: number;
  /** Set when the weekly payout that included this commission is sent */
  paidAt?: Date;
  /** Set while the commission is part of a withdrawal request (locks it so it cannot be requested twice) */
  withdrawalId?: Types.ObjectId | null;
  /** Days after delivery before the commission can be withdrawn */
  holdDays: number;
  /** Commission in naira */
  amount: number;
  createdAt: Date;
  updatedAt: Date;
}

const commissionSchema = new Schema<ICommission>(
  {
    // unique: one order can only ever produce one commission, even if the code runs twice
    orderId: { type: Schema.Types.ObjectId, ref: "Order", required: true, unique: true },
    orderNumber: { type: String, required: true, index: true },
    resellerId: { type: Schema.Types.ObjectId, ref: "Reseller", required: true, index: true },
    resellerCode: { type: String, required: true, index: true },
    base: { type: Number, required: true, min: 0 },
    rate: { type: Number, required: true, min: 0 },
    comboBase: { type: Number, default: 0, min: 0 },
    comboRate: { type: Number, default: 0, min: 0 },
    singleBase: { type: Number, default: 0, min: 0 },
    singleRate: { type: Number, default: 0, min: 0 },
    paidAt: { type: Date },
    withdrawalId: { type: Schema.Types.ObjectId, ref: "Withdrawal", default: null, index: true },
    holdDays: { type: Number, required: true, min: 0 },
    amount: { type: Number, required: true, min: 0 },
  },
  { timestamps: true },
);

export const Commission =
  (mongoose.models.Commission as mongoose.Model<ICommission>) ??
  mongoose.model<ICommission>("Commission", commissionSchema);

/** Single settings document (_id "program"). Created on first read with the defaults. */
export interface IResellerProgram {
  _id: string;
  /** Percent on combos (2+ items). Plan: 5 */
  comboPercent: number;
  /** Percent on single products (1 item). Plan: up to 10 */
  singlePercent: number;
  holdDays: number;
  /** Smallest balance a partner can withdraw, in naira. Plan: 2000 */
  minWithdrawal: number;
}

const programSchema = new Schema<IResellerProgram>(
  {
    _id: { type: String },
    comboPercent: { type: Number, default: 5, min: 0, max: 50 },
    singlePercent: { type: Number, default: 10, min: 0, max: 50 },
    holdDays: { type: Number, default: 7, min: 0, max: 90 },
    minWithdrawal: { type: Number, default: 2000, min: 0 },
  },
  { timestamps: true },
);

export const ResellerProgram =
  (mongoose.models.ResellerProgram as mongoose.Model<IResellerProgram>) ??
  mongoose.model<IResellerProgram>("ResellerProgram", programSchema);

export async function getProgramSettings(): Promise<IResellerProgram> {
  const doc = await ResellerProgram.findOneAndUpdate(
    { _id: "program" },
    { $setOnInsert: { comboPercent: 5, singlePercent: 10, holdDays: 7, minWithdrawal: 2000 } },
    { new: true, upsert: true },
  ).lean();
  return doc as IResellerProgram;
}
import React, { useState, useEffect } from 'react';
import { Gift, Trash2, Plus, Upload, Loader2, X, ArrowLeft } from 'lucide-react';
import { Link } from 'react-router-dom';
import { toast } from 'sonner';
import { apiFetch } from '@/api/client';
import { uploadToCloudinary } from '@/lib/cloudinary';
import { useDialogs } from '@/components/shared/ConfirmProvider';

export interface IGift {
  _id: string;
  name: string;
  description?: string;
  image?: string;
  images?: string[];
  price: number;
  stock: number;
  isActive: boolean;
}

export function AdminGiftBank() {
  const { confirm } = useDialogs();
  const [gifts, setGifts] = useState<IGift[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [isSaving, setIsSaving] = useState(false);

  // Form states
  const [name, setName] = useState('');
  const [description, setDescription] = useState('');
  const [price, setPrice] = useState('');
  const [stock, setStock] = useState('10');

  // Multi-image selection state (Cloudinary links)
  const [images, setImages] = useState<string[]>([]);
  const [isUploading, setIsUploading] = useState(false);

  useEffect(() => {
    fetchGifts();
  }, []);

  // apiFetch sends the admin login with every request
  const fetchGifts = async () => {
    setIsLoading(true);
    try {
      const data = await apiFetch<IGift[]>('/api/admin/gifts');
      setGifts(Array.isArray(data) ? data : []);
    } catch (err: any) {
      toast.error(err?.message || 'Failed to load gifts');
    } finally {
      setIsLoading(false);
    }
  };

  // Upload the chosen photos to Cloudinary and keep only the links
  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = e.target.files;
    if (!files || files.length === 0) return;
    const fileArray = Array.from(files);
    e.target.value = ''; // so choosing the same file again still works

    setIsUploading(true);
    try {
      const results = await Promise.all(fileArray.map((f) => uploadToCloudinary(f)));
      setImages((prev) => [...prev, ...results.map((r) => r.url)]);
    } catch (err: any) {
      toast.error(err?.message || 'Photo upload failed');
    } finally {
      setIsUploading(false);
    }
  };

  const handleRemoveImage = (index: number) => {
    setImages((prev) => prev.filter((_, i) => i !== index));
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (images.length === 0) {
      toast.error('Please select at least one image.');
      return;
    }

    setIsSaving(true);

    // Save both 'image' (for backwards compatibility) and 'images' (for multi-image)
    const payload = {
      name,
      description,
      price: Number(price) || 0,
      stock: Number(stock) || 0,
      image: images[0],
      images,
      isActive: true,
    };

    try {
      await apiFetch('/api/admin/gifts', { method: 'POST', body: JSON.stringify(payload) });
      toast.success('Gift added');
      setName('');
      setDescription('');
      setPrice('');
      setStock('10');
      setImages([]);
      fetchGifts();
    } catch (err: any) {
      toast.error(err?.message || 'Failed to add the gift');
    } finally {
      setIsSaving(false);
    }
  };

  const handleDelete = async (gift: IGift) => {
    const ok = await confirm({
      title: 'Delete this gift?',
      message: `"${gift.name}" will be removed from the Free Gift Bank.`,
      confirmText: 'Delete',
      danger: true,
    });
    if (!ok) return;

    try {
      await apiFetch(`/api/admin/gifts/${gift._id}`, { method: 'DELETE' });
      setGifts((prev) => prev.filter((g) => g._id !== gift._id));
      toast.success('Gift deleted');
    } catch (err: any) {
      toast.error(err?.message || 'Failed to delete the gift');
    }
  };

  return (
    <div className="min-h-screen bg-[#0a0b0e] text-white p-4 sm:p-10">
      <div className="max-w-5xl mx-auto space-y-6">

        <Link
          to="/admin"
          className="inline-flex items-center gap-2 text-xs font-semibold text-white/50 hover:text-white transition-colors"
        >
          <ArrowLeft className="w-4 h-4" />
          Back to Dashboard
        </Link>

        <div className="flex items-center gap-3">
          <div className="p-3 bg-emerald-500/10 border border-emerald-500/20 rounded-2xl text-emerald-400">
            <Gift className="w-6 h-6" />
          </div>
          <div>
            <h1 className="text-xl font-bold">Free Gift Bank Manager</h1>
            <p className="text-xs text-white/50">
              Manage free gifts, estimated prices, and available stock inventory.
            </p>
          </div>
        </div>

        <div className="bg-[#12141c] border border-white/10 rounded-2xl p-4 sm:p-6 space-y-5">
          <div className="flex items-center gap-2 text-xs font-bold text-emerald-400 tracking-wider uppercase">
            <Plus className="w-4 h-4" />
            <span>Add New Free Gift Item</span>
          </div>

          <form onSubmit={handleSubmit} className="space-y-4">
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
              <div>
                <label className="block text-xs font-semibold text-white/70 mb-1.5">
                  Gift Title
                </label>
                <input
                  type="text"
                  required
                  placeholder="e.g. Rolex Submariner"
                  value={name}
                  onChange={(e) => setName(e.target.value)}
                  className="w-full bg-[#0a0b0e] border border-white/10 rounded-xl px-4 py-2.5 text-sm text-white outline-none focus:border-emerald-500 transition-colors"
                />
              </div>

              {/* Multi Image Upload Box */}
              <div>
                <label className="block text-xs font-semibold text-white/70 mb-1.5">
                  Product Images ({images.length} Selected)
                </label>

                <div className="flex flex-wrap gap-2 items-center">
                  {images.map((url, idx) => (
                    <div
                      key={idx}
                      className="relative w-14 h-14 rounded-xl overflow-hidden border border-white/20 group bg-black/40"
                    >
                      <img src={url} alt={`Upload ${idx + 1}`} className="w-full h-full object-cover" />

                      {idx === 0 && (
                        <span className="absolute bottom-0 inset-x-0 bg-emerald-500 text-black font-black text-[7px] text-center py-0.5 uppercase">
                          Cover
                        </span>
                      )}

                      <button
                        type="button"
                        onClick={() => handleRemoveImage(idx)}
                        className="absolute inset-0 bg-black/70 flex items-center justify-center text-red-400 opacity-0 group-hover:opacity-100 transition-opacity"
                      >
                        <X className="w-4 h-4" />
                      </button>
                    </div>
                  ))}

                  <label className="w-14 h-14 border-2 border-dashed border-white/20 hover:border-emerald-500 rounded-xl flex flex-col items-center justify-center cursor-pointer text-white/40 hover:text-emerald-400 transition-colors bg-white/5">
                    {isUploading ? (
                      <Loader2 className="w-4 h-4 animate-spin text-emerald-400" />
                    ) : (
                      <>
                        <Upload className="w-4 h-4" />
                        <span className="text-[8px] font-bold mt-1">
                          {images.length > 0 ? '+ More' : '+ Add'}
                        </span>
                      </>
                    )}
                    <input
                      type="file"
                      multiple
                      accept="image/*"
                      onChange={handleFileUpload}
                      disabled={isUploading}
                      className="hidden"
                    />
                  </label>
                </div>
              </div>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
              <div>
                <label className="block text-xs font-semibold text-white/70 mb-1.5">
                  Estimated Price / Value (₦)
                </label>
                <input
                  type="number"
                  placeholder="e.g. 23000"
                  value={price}
                  onChange={(e) => setPrice(e.target.value)}
                  className="w-full bg-[#0a0b0e] border border-white/10 rounded-xl px-4 py-2.5 text-sm text-white outline-none focus:border-emerald-500 transition-colors"
                />
              </div>

              <div>
                <label className="block text-xs font-semibold text-white/70 mb-1.5">
                  Quantity / Stock Available
                </label>
                <input
                  type="number"
                  value={stock}
                  onChange={(e) => setStock(e.target.value)}
                  className="w-full bg-[#0a0b0e] border border-white/10 rounded-xl px-4 py-2.5 text-sm text-white outline-none focus:border-emerald-500 transition-colors"
                />
              </div>
            </div>

            <div>
              <label className="block text-xs font-semibold text-white/70 mb-1.5">
                Description (Optional)
              </label>
              <input
                type="text"
                placeholder="e.g. Complimentary gift for orders over ₦40,000"
                value={description}
                onChange={(e) => setDescription(e.target.value)}
                className="w-full bg-[#0a0b0e] border border-white/10 rounded-xl px-4 py-2.5 text-sm text-white outline-none focus:border-emerald-500 transition-colors"
              />
            </div>

            <button
              type="submit"
              disabled={isSaving || isUploading}
              className="py-2.5 px-5 bg-emerald-500 hover:bg-emerald-400 text-black font-bold text-xs rounded-xl transition-colors disabled:opacity-50 flex items-center gap-2"
            >
              {isSaving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Plus className="w-4 h-4" />}
              <span>Add Gift to Bank</span>
            </button>
          </form>
        </div>

        {/* Inventory Listing */}
        <div className="space-y-3">
          <h2 className="text-sm font-bold text-white/70">
            Active Gift Inventory ({gifts.length})
          </h2>

          {isLoading ? (
            <div className="flex items-center justify-center py-12 text-white/40">
              <Loader2 className="w-6 h-6 animate-spin" />
            </div>
          ) : gifts.length === 0 ? (
            <div className="text-center py-10 bg-[#12141c] border border-white/10 rounded-xl text-white/40 text-xs">
              No gifts added yet.
            </div>
          ) : (
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
              {gifts.map((gift) => {
                // Fallback check to support old single-image items & new multi-image items
                const displayImage = gift.image || (gift.images && gift.images[0]) || '';

                return (
                  <div
                    key={gift._id}
                    className="bg-[#12141c] border border-white/10 rounded-xl p-3 flex items-center gap-3 relative group"
                  >
                    {displayImage ? (
                      <img
                        src={displayImage}
                        alt={gift.name}
                        className="w-12 h-12 rounded-lg object-cover bg-black/40"
                      />
                    ) : (
                      <div className="w-12 h-12 rounded-lg bg-white/5 flex items-center justify-center text-white/30 text-xs">
                        No img
                      </div>
                    )}

                    <div className="flex-1 min-w-0">
                      <h3 className="text-xs font-bold truncate">{gift.name}</h3>
                      <p className="text-[10px] text-white/50">
                        Value:{' '}
                        <span className="line-through">₦{gift.price?.toLocaleString()}</span>{' '}
                        <span className="text-emerald-400 font-bold">FREE</span>
                      </p>
                      <span className="inline-block mt-1 text-[9px] bg-emerald-500/10 text-emerald-400 font-bold px-1.5 py-0.5 rounded">
                        {gift.stock} Remaining
                      </span>
                    </div>

                    <button
                      onClick={() => handleDelete(gift)}
                      className="text-white/30 hover:text-red-400 p-1.5 transition-colors"
                      aria-label={`Delete ${gift.name}`}
                    >
                      <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                );
              })}
            </div>
          )}
        </div>

      </div>
    </div>
  );
}

export default AdminGiftBank;
