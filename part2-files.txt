/**
 * Admin combos CRUD routes.
 * Handles `alternatives` and `colors` arrays on each item.
 */

import { Router, Request, Response } from "express";
import { Combo } from "../../models/Combo";
import { COST_SELECT } from "../../services/costService";

const router = Router();

function slugify(s: string): string {
  return (s ?? "")
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/(^-|-$)/g, "")
    .slice(0, 80) || "item";
}

function ensureAlternatives(alts: any[]): any[] {
  if (!Array.isArray(alts)) return [];
  return alts.map((alt, i) => ({
    id: alt.id || slugify(alt.name || `alt-${i + 1}`),
    name: String(alt.name || `Alternative ${i + 1}`).trim(),
    badge: alt.badge || "",
    images: Array.isArray(alt.images) ? alt.images : [],
    description: alt.description || "",
    costPrice: Math.max(0, Number(alt.costPrice) || 0),
  }));
}

function ensureColors(colors: any[]): any[] {
  if (!Array.isArray(colors)) return [];
  return colors.map((c, i) => ({
    id: c.id || slugify(c.name || `color-${i + 1}`),
    name: String(c.name || `Color ${i + 1}`).trim(),
    hexCode: typeof c.hexCode === "string" ? c.hexCode.trim() : "",
    imageUrl: typeof c.imageUrl === "string" ? c.imageUrl.trim() : "",
  }));
}

function ensureItemIds(items: any[]): any[] {
  if (!Array.isArray(items)) return [];
  return items.map((item, i) => ({
    ...item,
    id: item.id || slugify(item.name || `item-${i + 1}`),
    images: Array.isArray(item.images) ? item.images : [],
    individualPrice: Number(item.individualPrice) || 0,
    costPrice: Math.max(0, Number(item.costPrice) || 0),
    alternatives: ensureAlternatives(item.alternatives),
    colors: ensureColors(item.colors),
  }));
}

router.get("/", async (_req: Request, res: Response) => {
  try {
    const combos = await Combo.find({}).select(COST_SELECT).sort({ createdAt: -1 }).lean();
    res.json({ data: combos });
  } catch (err) {
    console.error("GET /api/admin/combos failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to fetch combos" } });
  }
});

router.get("/:id", async (req: Request, res: Response) => {
  try {
    const combo = await Combo.findById(req.params.id).select(COST_SELECT).lean();
    if (!combo) {
      return res.status(404).json({ error: { code: "NOT_FOUND", message: "Combo not found" } });
    }
    res.json({ data: combo });
  } catch (err: any) {
    if (err.name === "CastError") {
      return res.status(404).json({ error: { code: "NOT_FOUND", message: "Combo not found" } });
    }
    console.error(`GET /api/admin/combos/${req.params.id} failed:`, err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to fetch combo" } });
  }
});

router.post("/", async (req: Request, res: Response) => {
  try {
    const body = req.body || {};
    if (!body.name) {
      return res.status(400).json({ error: { code: "BAD_REQUEST", message: "Name is required" } });
    }
    const slug = body.slug ? slugify(body.slug) : slugify(body.name);
    const existing = await Combo.findOne({ slug });
    if (existing) {
      return res.status(409).json({
        error: { code: "DUPLICATE", message: `A combo with slug "${slug}" already exists.` },
      });
    }
    if (body.isFeatured) {
      await Combo.updateMany({ isFeatured: true }, { $set: { isFeatured: false } });
    }
    const combo = await Combo.create({
      slug,
      name: String(body.name).trim(),
      tagline: body.tagline ?? "",
      totalPrice: Number(body.totalPrice) || 0,
      costPrice: Math.max(0, Number(body.costPrice) || 0),
      originalPrice: Number(body.originalPrice) || 0,
      badge: body.badge ?? "",
      stockLeft: Number(body.stockLeft) || 0,
      isFeatured: Boolean(body.isFeatured),
      isActive: body.isActive !== false,
      customComboEligible: Boolean(body.customComboEligible),
      items: ensureItemIds(body.items),
      heroImage: body.heroImage ?? "",
      categorySlugs: Array.isArray(body.categorySlugs) ? body.categorySlugs : [],
    });
    res.status(201).json({ data: combo });
  } catch (err: any) {
    if (err.name === "ValidationError") {
      return res.status(400).json({ error: { code: "VALIDATION_ERROR", message: err.message } });
    }
    console.error("POST /api/admin/combos failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to create combo" } });
  }
});

router.patch("/:id", async (req: Request, res: Response) => {
  try {
    const body = req.body || {};
    const update: any = {};

    if (body.name !== undefined) update.name = String(body.name).trim();
    if (body.tagline !== undefined) update.tagline = body.tagline;
    if (body.totalPrice !== undefined) update.totalPrice = Number(body.totalPrice) || 0;
    if (body.costPrice !== undefined) update.costPrice = Math.max(0, Number(body.costPrice) || 0);
    if (body.originalPrice !== undefined) update.originalPrice = Number(body.originalPrice) || 0;
    if (body.badge !== undefined) update.badge = body.badge;
    if (body.stockLeft !== undefined) update.stockLeft = Number(body.stockLeft) || 0;
    if (body.isActive !== undefined) update.isActive = Boolean(body.isActive);
    if (body.customComboEligible !== undefined) update.customComboEligible = Boolean(body.customComboEligible);
    if (body.heroImage !== undefined) update.heroImage = body.heroImage;
    if (Array.isArray(body.categorySlugs)) update.categorySlugs = body.categorySlugs;
    if (Array.isArray(body.items)) update.items = ensureItemIds(body.items);

    if (body.slug !== undefined) {
      const newSlug = slugify(body.slug);
      const existing = await Combo.findOne({ slug: newSlug, _id: { $ne: req.params.id } });
      if (existing) {
        return res.status(409).json({
          error: { code: "DUPLICATE", message: `A different combo already uses slug "${newSlug}".` },
        });
      }
      update.slug = newSlug;
    }

    if (body.isFeatured === true) {
      await Combo.updateMany(
        { _id: { $ne: req.params.id }, isFeatured: true },
        { $set: { isFeatured: false } },
      );
      update.isFeatured = true;
    } else if (body.isFeatured === false) {
      update.isFeatured = false;
    }

    const combo = await Combo.findByIdAndUpdate(req.params.id, update, {
      new: true,
      runValidators: true,
    }).select(COST_SELECT).lean();

    if (!combo) {
      return res.status(404).json({ error: { code: "NOT_FOUND", message: "Combo not found" } });
    }
    res.json({ data: combo });
  } catch (err: any) {
    if (err.name === "CastError") return res.status(404).json({ error: { code: "NOT_FOUND", message: "Combo not found" } });
    if (err.name === "ValidationError") return res.status(400).json({ error: { code: "VALIDATION_ERROR", message: err.message } });
    console.error(`PATCH /api/admin/combos/${req.params.id} failed:`, err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to update combo" } });
  }
});

router.delete("/:id", async (req: Request, res: Response) => {
  try {
    const combo = await Combo.findByIdAndDelete(req.params.id).lean();
    if (!combo) return res.status(404).json({ error: { code: "NOT_FOUND", message: "Combo not found" } });
    res.json({ data: { id: req.params.id, deleted: true } });
  } catch (err: any) {
    if (err.name === "CastError") return res.status(404).json({ error: { code: "NOT_FOUND", message: "Combo not found" } });
    console.error(`DELETE /api/admin/combos/${req.params.id} failed:`, err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to delete combo" } });
  }
});

export default router;

/**
 * Stock levels, low-stock alerts and the restock log.
 *
 * GET  /api/admin/stock              all products with stock + status (out / low / ok), counts
 * GET  /api/admin/stock/log          latest 60 stock changes
 * POST /api/admin/stock/:id/restock  { quantity, note? }  add units
 * POST /api/admin/stock/:id/set      { stockLeft, note? } set the exact count (stock take / correction)
 */
import { Router, Request, Response } from "express";
import { Combo } from "../../models/Combo";
import { StockLog } from "../../models/StockLog";

const router = Router();
export const LOW_STOCK_AT = 3;

const statusOf = (n: number) => (n <= 0 ? "out" : n <= LOW_STOCK_AT ? "low" : "ok");
const err = (res: Response, code: number, message: string) =>
  res.status(code).json({ error: { code: code === 404 ? "NOT_FOUND" : "BAD_REQUEST", message } });

router.get("/", async (_req: Request, res: Response) => {
  try {
    const combos = await Combo.find({}).select("name images stockLeft isActive").sort({ stockLeft: 1, name: 1 }).lean();
    const items = combos.map((c: any) => ({
      id: String(c._id),
      name: c.name,
      image: c.images?.[0]?.url ?? "",
      stockLeft: c.stockLeft ?? 0,
      isActive: c.isActive !== false,
      status: statusOf(c.stockLeft ?? 0),
    }));
    const live = items.filter((i) => i.isActive);
    res.json({
      data: {
        items,
        lowAt: LOW_STOCK_AT,
        counts: { out: live.filter((i) => i.status === "out").length, low: live.filter((i) => i.status === "low").length },
      },
    });
  } catch (e) {
    console.error("GET /api/admin/stock failed:", e);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to load stock" } });
  }
});

router.get("/log", async (_req: Request, res: Response) => {
  try {
    const rows = await StockLog.find({}).sort({ createdAt: -1 }).limit(60).lean();
    res.json({ data: rows.map((r: any) => ({ id: String(r._id), comboName: r.comboName, kind: r.kind, delta: r.delta, before: r.before, after: r.after, note: r.note ?? "", createdAt: r.createdAt })) });
  } catch (e) {
    console.error("GET /api/admin/stock/log failed:", e);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to load the log" } });
  }
});

router.post("/:id/restock", async (req: Request, res: Response) => {
  try {
    const q = Math.floor(Number(req.body?.quantity));
    if (!Number.isFinite(q) || q < 1 || q > 100000) return err(res, 400, "Enter a quantity between 1 and 100000");
    const note = typeof req.body?.note === "string" ? req.body.note.trim().slice(0, 200) : "";
    // atomic: $inc cannot lose a sale that happens at the same moment
    const after: any = await Combo.findByIdAndUpdate(req.params.id, { $inc: { stockLeft: q } }, { new: true }).select("name stockLeft").lean();
    if (!after) return err(res, 404, "Product not found");
    await StockLog.create({ comboId: after._id, comboName: after.name, kind: "restock", delta: q, before: after.stockLeft - q, after: after.stockLeft, note });
    res.json({ data: { id: String(after._id), stockLeft: after.stockLeft, status: statusOf(after.stockLeft) } });
  } catch (e: any) {
    if (e.name === "CastError") return err(res, 404, "Product not found");
    console.error("restock failed:", e);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to add stock" } });
  }
});

router.post("/:id/set", async (req: Request, res: Response) => {
  try {
    const n = Math.floor(Number(req.body?.stockLeft));
    if (!Number.isFinite(n) || n < 0 || n > 100000) return err(res, 400, "Enter a count between 0 and 100000");
    const note = typeof req.body?.note === "string" ? req.body.note.trim().slice(0, 200) : "";
    const before: any = await Combo.findByIdAndUpdate(req.params.id, { stockLeft: n }, { new: false }).select("name stockLeft").lean();
    if (!before) return err(res, 404, "Product not found");
    const was = before.stockLeft ?? 0;
    await StockLog.create({ comboId: before._id, comboName: before.name, kind: "set", delta: n - was, before: was, after: n, note });
    res.json({ data: { id: String(before._id), stockLeft: n, status: statusOf(n) } });
  } catch (e: any) {
    if (e.name === "CastError") return err(res, 404, "Product not found");
    console.error("set stock failed:", e);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to set stock" } });
  }
});

export default router;

/**
 * Fulfilment queue: what to pack, what is on its way.
 *
 * GET  /api/admin/fulfilment              orders to pack + orders in transit
 * POST /api/admin/fulfilment/:id/packed   paid/COD order -> "processing" (packed, waiting for pickup/Jumia)
 * POST /api/admin/fulfilment/:id/ship     { trackingNumber?, trackingProviderUrl? } -> shipped (+ customer email if they have one)
 * POST /api/admin/fulfilment/:id/delivered -> delivered
 */
import { Router, Request, Response } from "express";
import { Order } from "../../models/Order";
import { markOrderShipped, markOrderDelivered } from "../../services/orderService";

const router = Router();

type Route = "pickup" | "local" | "jumia";
function routeOf(o: any): Route {
  const street = String(o.shipping?.street ?? "");
  if (street.startsWith("Pickup")) return "pickup";
  return String(o.shipping?.city ?? "").trim().toLowerCase() === "ilorin" ? "local" : "jumia";
}

// COD orders sit in "pending" until delivery, but still have to be packed.
const TO_PACK_FILTER = {
  $or: [
    { status: { $in: ["paid", "processing"] } },
    { status: "pending", paymentMethod: "cod" },
  ],
};

function shape(o: any) {
  return {
    _id: String(o._id),
    orderNumber: o.orderNumber,
    status: o.status,
    paymentMethod: o.paymentMethod,
    source: o.source ?? "website",
    total: o.total,
    shippingFee: o.shippingFee,
    createdAt: o.createdAt,
    paidAt: o.paidAt,
    shippedAt: o.shippedAt,
    trackingNumber: o.trackingNumber ?? "",
    trackingProviderUrl: o.trackingProviderUrl ?? "",
    notes: o.notes ?? "",
    adminNotes: o.adminNotes ?? "",
    route: routeOf(o),
    shipping: o.shipping,
    items: (o.items ?? []).map((it: any) => ({
      name: it.name,
      variantSummary: it.variantSummary ?? "",
      quantity: it.quantity,
      isFreeGift: !!it.isFreeGift,
    })),
  };
}

router.get("/", async (_req: Request, res: Response) => {
  try {
    const [toPack, inTransit] = await Promise.all([
      Order.find(TO_PACK_FILTER).sort({ createdAt: 1 }).limit(200).lean(),
      Order.find({ status: "shipped" }).sort({ shippedAt: 1 }).limit(200).lean(),
    ]);
    res.json({ data: { toPack: toPack.map(shape), inTransit: inTransit.map(shape) } });
  } catch (err) {
    console.error("GET /api/admin/fulfilment failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to load the queue" } });
  }
});

function bad(res: Response, status: number, message: string) {
  return res.status(status).json({ error: { code: status === 404 ? "NOT_FOUND" : "BAD_REQUEST", message } });
}

router.post("/:id/packed", async (req: Request, res: Response) => {
  try {
    const o = await Order.findOneAndUpdate(
      { _id: req.params.id, ...TO_PACK_FILTER, status: { $in: ["paid", "pending"] } },
      { status: "processing" },
      { new: true },
    ).lean();
    if (!o) return bad(res, 409, "This order cannot be marked as packed (already packed, unpaid, or closed).");
    res.json({ data: shape(o) });
  } catch (err: any) {
    if (err.name === "CastError") return bad(res, 404, "Order not found");
    console.error("fulfilment packed failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to update order" } });
  }
});

router.post("/:id/ship", async (req: Request, res: Response) => {
  try {
    const existing = await Order.findById(req.params.id).select("orderNumber status paymentMethod").lean();
    if (!existing) return bad(res, 404, "Order not found");
    const ok =
      existing.status === "paid" ||
      existing.status === "processing" ||
      (existing.status === "pending" && existing.paymentMethod === "cod");
    if (!ok) return bad(res, 409, `An order that is ${existing.status} cannot be shipped.`);
    const tn = typeof req.body?.trackingNumber === "string" ? req.body.trackingNumber.trim().slice(0, 80) : "";
    const tu = typeof req.body?.trackingProviderUrl === "string" ? req.body.trackingProviderUrl.trim().slice(0, 300) : "";
    if (tu && !/^https?:\/\//i.test(tu)) return bad(res, 400, "The tracking link must start with http:// or https://");
    const updated = await markOrderShipped(existing.orderNumber, tn || undefined, tu || undefined);
    if (!updated) return bad(res, 404, "Order not found");
    res.json({ data: shape(updated.toObject ? updated.toObject() : updated) });
  } catch (err: any) {
    if (err.name === "CastError") return bad(res, 404, "Order not found");
    console.error("fulfilment ship failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to mark as shipped" } });
  }
});

router.post("/:id/delivered", async (req: Request, res: Response) => {
  try {
    const existing = await Order.findById(req.params.id).select("orderNumber status paymentMethod").lean();
    if (!existing) return bad(res, 404, "Order not found");
    const ok =
      ["shipped", "paid", "processing"].includes(existing.status) ||
      (existing.status === "pending" && existing.paymentMethod === "cod");
    if (!ok) return bad(res, 409, `An order that is ${existing.status} cannot be marked delivered.`);
    const updated = await markOrderDelivered(existing.orderNumber);
    if (!updated) return bad(res, 404, "Order not found");
    res.json({ data: shape(updated.toObject ? updated.toObject() : updated) });
  } catch (err: any) {
    if (err.name === "CastError") return bad(res, 404, "Order not found");
    console.error("fulfilment delivered failed:", err);
    res.status(500).json({ error: { code: "SERVER_ERROR", message: "Failed to mark as delivered" } });
  }
});

export default router;

/**
 * Admin dashboard stats — aggregates for the homepage of /admin.
 *
 * GET /api/admin/dashboard/stats
 *
 * Returns:
 *   - ordersToday: count of orders created today
 *   - revenueToday: sum of totals from PAID orders today (excludes pending)
 *   - pendingOrders: count of orders with status "pending"
 *   - customers: distinct unique customers (by phone) ever
 *   - recentOrders: last 5 orders for quick preview
 */

import { Router, Request, Response } from "express";
import { Order } from "../../models/Order";

const router = Router();

router.get("/stats", async (_req: Request, res: Response) => {
  try {
    const startOfToday = new Date();
    startOfToday.setHours(0, 0, 0, 0);

    const [ordersToday, revenueAggResult, pendingOrders, customerAgg, recentOrders] =
      await Promise.all([
        Order.countDocuments({ createdAt: { $gte: startOfToday } }),

        // Revenue today = sum of totals for orders that are 'paid' OR beyond
        // (paid, processing, shipped, delivered) created today.
        // We exclude 'pending' so revenue reflects actual confirmed money.
        Order.aggregate([
          {
            $match: {
              createdAt: { $gte: startOfToday },
              status: { $in: ["paid", "processing", "shipped", "delivered"] },
            },
          },
          { $group: { _id: null, total: { $sum: "$total" } } },
        ]),

        Order.countDocuments({ status: "pending" }),

        // Unique customers ever (by phone)
        Order.aggregate([
          { $group: { _id: "$shipping.phone" } },
          { $count: "uniqueCustomers" },
        ]),

        // Recent 5 orders, lightweight
        Order.find({})
          .sort({ createdAt: -1 })
          .limit(5)
          .select("orderNumber status total shipping.fullName createdAt")
          .lean(),
      ]);

    const revenueToday = revenueAggResult[0]?.total ?? 0;
    const customers = customerAgg[0]?.uniqueCustomers ?? 0;

    res.json({
      data: {
        ordersToday,
        revenueToday,
        pendingOrders,
        customers,
        recentOrders,
      },
    });
  } catch (err) {
    console.error("GET /api/admin/dashboard/stats failed:", err);
    res.status(500).json({
      error: { code: "SERVER_ERROR", message: "Failed to fetch stats" },
    });
  }
});

export default router;

interface Line { key: number; comboId: string; quantity: number; variants: Record<string, string> }

/** Same rule the server uses: a typed combo cost wins, otherwise the item costs (a variant with its own cost overrides). */
function unitCost(combo: any, variants: Record<string, string>): number {
  const own = Number(combo?.costPrice) || 0;
  if (own > 0) return own;
  let total = 0;
  for (const item of combo?.items ?? []) {
    let cost = Number(item.costPrice) || 0;
    const alt = (item.alternatives ?? []).find((a: any) => a.id === variants[item.id]);
    if (alt && Number(alt.costPrice) > 0) cost = Number(alt.costPrice);
    total += cost;
  }
  return total;
}

function slugify(s: string): string {
  return (s ?? '').toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '').slice(0, 80);
}

function fromCombo(combo: Combo): FormState {
  return {
    slug: combo.slug,
    name: combo.name,
    tagline: combo.tagline ?? '',
    totalPrice: combo.totalPrice,
    originalPrice: combo.originalPrice,
    badge: combo.badge ?? '',
    stockLeft: combo.stockLeft,
    costPrice: (combo as any).costPrice ?? 0,
    isFeatured: combo.isFeatured,
    isActive: combo.isActive,
    customComboEligible: combo.customComboEligible ?? false,
    heroImage: combo.heroImage ?? '',
    categorySlugs: combo.categorySlugs ?? [],
    items: (combo.items ?? []).map((item) => ({
      id: item.id,
      name: item.name,
      badge: item.badge ?? '',
      individualPrice: item.individualPrice,
      costPrice: (item as any).costPrice ?? 0,
      description: item.description ?? '',
      images: item.images ?? [],
      alternatives: (item.alternatives ?? []).map((alt) => ({
        id: alt.id,
        name: alt.name,
        badge: alt.badge ?? '',
        description: alt.description ?? '',
        costPrice: (alt as any).costPrice ?? 0,
        images: alt.images ?? [],
      })),
      colors: (item.colors ?? []).map((c) => ({
        id: c.id,
        name: c.name,
        hexCode: c.hexCode ?? '',
        imageUrl: c.imageUrl ?? '',
      })),
    })),
  };
}

/** Shows cost and profit for the combo, using the cost prices entered on its items (worst case when variants differ). */
function CostSummary({ form }: { form: FormState }) {
  if (!form.totalPrice) return null;
  const itemsCost = form.items.reduce((sum, it) => {
    const alts = it.alternatives.map((a) => a.costPrice || it.costPrice || 0);
    return sum + Math.max(it.costPrice || 0, ...alts);
  }, 0);
  const cost = form.costPrice > 0 ? form.costPrice : itemsCost;
  if (!cost) {
    return (
      <p className="mt-3 text-[11px] text-white/40">
        Enter your cost price (what you paid) to see your profit on each sale.
      </p>
    );
  }
  const profit = form.totalPrice - cost;
  const pct = Math.round((profit / form.totalPrice) * 100);
  const loss = profit < 0;
  return (
    <div className={`mt-3 rounded-lg border px-3 py-2 text-xs ${loss ? 'border-red-500/40 bg-red-500/10 text-red-200' : 'border-emerald-500/25 bg-emerald-500/5 text-emerald-200'}`}>
      Cost ₦{cost.toLocaleString('en-NG')} · Profit per sale ₦{profit.toLocaleString('en-NG')} ({pct}%)
      {loss && ' — selling below cost!'}
    </div>
  );
}

function Section({ title, children }: { title: string; children: React.ReactNode }) {
  return (
    <div className="rounded-2xl border border-white/10 p-5" style={{ background: 'rgba(255,255,255,0.02)' }}>
      <h2 className="text-sm font-black text-white mb-4">{title}</h2>
      <div className="space-y-3">{children}</div>
    </div>
  );
}

function Field({ label, required, hint, children }: { label: string; required?: boolean; hint?: string; children: React.ReactNode }) {
  return (
    <label className="block">
      <span className="block text-[11px] font-bold text-white/60 uppercase tracking-wider mb-1.5">
        {label} {required && <span className="text-primary/70">*</span>}
      </span>
      {children}
      {hint && <p className="text-[10px] text-white/40 mt-1">{hint}</p>}
    </label>
  );
}

const inputCls = 'w-full px-3 py-2 rounded-lg bg-black/30 border border-white/10 focus:border-primary/40 focus:outline-none text-sm text-white placeholder-white/30 transition-colors';

function Toggle({ label, hint, checked, onChange }: { label: string; hint?: string; checked: boolean; onChange: (v: boolean) => void }) {
  return (
    <label className="flex items-start gap-3 cursor-pointer p-2 -m-2 rounded-lg hover:bg-white/[0.02]">
      <input type="checkbox" checked={checked} onChange={(e) => onChange(e.target.checked)} className="w-4 h-4 mt-0.5 rounded" />
      <div className="flex-1">
        <p className="text-sm text-white">{label}</p>
        {hint && <p className="text-[11px] text-white/40">{hint}</p>}
      </div>
    </label>
  );
}

  </style></head><body>${orders.map(slipHtml).join('')}</body></html>`);
  w.document.close();
  w.focus();

}

function age(iso?: string) {
  if (!iso) return '';
  const h = Math.floor((Date.now() - new Date(iso).getTime()) / 3600000);
  if (h < 1) return 'just now';
  if (h < 24) return `${h}h ago`;
  return `${Math.floor(h / 24)}d ago`;
}

/** Stock levels, low-stock alerts and restock log. */
import { useState } from 'react';
import { Link } from 'react-router-dom';
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { ArrowLeft, Loader2, Search } from 'lucide-react';

import { getStock, getStockLog, restock, setStock, type StockItem } from '@/api/adminStock';

const STATUS_STYLE = {
  out: 'bg-red-500/15 text-red-300 border-red-500/25',
  low: 'bg-amber-500/15 text-amber-300 border-amber-500/25',
  ok: 'bg-emerald-500/15 text-emerald-300 border-emerald-500/25',
} as const;
const STATUS_LABEL = { out: 'Out of stock', low: 'Running low', ok: 'In stock' } as const;

type Filter = 'all' | 'attention' | 'ok' | 'log';

export default function AdminStock() {
  const qc = useQueryClient();
  const [filter, setFilter] = useState<Filter>('attention');
  const [search, setSearch] = useState('');
  const { data, isLoading, error } = useQuery({ queryKey: ['admin-stock'], queryFn: getStock });
  const { data: log = [] } = useQuery({ queryKey: ['admin-stock-log'], queryFn: getStockLog, enabled: filter === 'log' });

  const refresh = () => {
    qc.invalidateQueries({ queryKey: ['admin-stock'] });
    qc.invalidateQueries({ queryKey: ['admin-stock-log'] });
    qc.invalidateQueries({ queryKey: ['admin-stock-alerts'] });
    qc.invalidateQueries({ queryKey: ['admin-combos'] });
  };
  const onErr = (e: any) => toast.error(e?.message || 'Something went wrong');
  const add = useMutation({
    mutationFn: (v: { id: string; q: number; note: string }) => restock(v.id, v.q, v.note),
    onSuccess: () => { toast.success('Stock added'); refresh(); }, onError: onErr,
  });
  const set = useMutation({
    mutationFn: (v: { id: string; q: number; note: string }) => setStock(v.id, v.q, v.note),
    onSuccess: () => { toast.success('Stock count saved'); refresh(); }, onError: onErr,
  });

  const items = (data?.items ?? []).filter((i) => {
    if (search.trim() && !i.name.toLowerCase().includes(search.trim().toLowerCase())) return false;
    if (filter === 'attention') return i.isActive && i.status !== 'ok';
    if (filter === 'ok') return i.status === 'ok';
    return true;
  });
  const attention = (data?.counts.out ?? 0) + (data?.counts.low ?? 0);

  return (
    <AdminFrame>
      <SEO title="Stock" />
      <Link to="/admin" className="inline-flex items-center gap-2 text-xs text-white/50 hover:text-white/80 mb-5">

      </Link>
      <h1 className="text-2xl md:text-3xl font-black text-white mb-1">Stock</h1>

      <div className="flex flex-wrap items-center gap-2 mb-6">

                <span className="text-white/40"> ({r.before} → {r.after})</span>

            </div>
          ))}

      ) : (

function StockRow({ item, busy, onAdd, onSet }: { item: StockItem; busy: boolean; onAdd: (q: number, note: string) => void; onSet: (q: number, note: string) => void }) {
  const [qty, setQty] = useState('');
  const [note, setNote] = useState('');
  const n = Math.floor(Number(qty));
  const valid = qty.trim() !== '' && Number.isFinite(n);
  const input = 'px-3 py-2 rounded-lg bg-white/5 border border-white/10 text-xs text-white outline-none focus:border-primary/50';
  return (
    <article className="rounded-2xl border border-white/10 p-4 flex flex-wrap items-center gap-4" style={{ background: 'rgba(255,255,255,0.02)', opacity: item.isActive ? 1 : 0.6 }}>
      <div className="flex items-center gap-3 min-w-[220px] flex-1">
        {item.image ? <img src={item.image} alt="" className="w-12 h-12 rounded-lg object-cover bg-white/5" /> : <div className="w-12 h-12 rounded-lg bg-white/5" />}
        <div>
          <p className="text-sm font-bold text-white">{item.name}{!item.isActive && <span className="ml-2 text-[10px] text-white/40">(hidden)</span>}</p>
          <span className={`inline-block mt-1 px-2 py-0.5 rounded-full border text-[10px] font-bold ${STATUS_STYLE[item.status]}`}>{STATUS_LABEL[item.status]}</span>
        </div>
      </div>
      <p className="text-3xl font-black text-white tabular-nums w-16 text-center">{item.stockLeft}</p>
      <div className="flex flex-wrap items-center gap-2">
        <input className={`${input} w-20`} inputMode="numeric" value={qty} onChange={(e) => setQty(e.target.value.replace(/[^0-9]/g, ''))} placeholder="Qty" />
        <input className={`${input} w-36`} value={note} onChange={(e) => setNote(e.target.value)} placeholder="Note (optional)" maxLength={200} />
        <button type="button" disabled={busy || !valid || n < 1} onClick={() => { onAdd(n, note); setQty(''); setNote(''); }}
          className="px-3 py-2 rounded-lg bg-primary text-black text-xs font-black disabled:opacity-40">Add stock</button>
        <button type="button" disabled={busy || !valid} onClick={() => { onSet(n, note); setQty(''); setNote(''); }}
          className="px-3 py-2 rounded-lg border border-white/15 text-xs font-bold text-white/80 hover:bg-white/5 disabled:opacity-40" title="Replace the count with this exact number">Set count</button>
      </div>
    </article>
  );
}
