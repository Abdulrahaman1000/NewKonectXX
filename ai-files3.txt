/**
 * AI shopping assistant.  POST /api/ai/chat
 *
 * Body:    { messages: [{ role: "user" | "assistant", text }] }   (last one is the customer's question)
 * Returns: { data: { answer, sources: [{id,title,link}], needsHuman } }
 *
 * Separate from checkout, payments and the human chat. Nothing is stored.
 */

import { Router, Request, Response } from "express";
import { z } from "zod";
import { getKnowledge } from "../ai/knowledge";
import { retrieve, isGreeting, directAnswer, listAnswer, tokenize } from "../ai/retrieve";
import { askGemini, AIError } from "../ai/gemini";

const router = Router();

const Body = z.object({
  messages: z
    .array(
      z.object({
        role: z.enum(["user", "assistant"]),
        text: z.string().trim().min(1).max(500),
      }),
    )
    .min(1)
    .max(12),
});

// ---- limits: per visitor, plus a daily cap so the free quota is never burned by one person ----
const hits = new Map<string, number[]>();
function limited(key: string, max: number, windowMs: number): boolean {
  const now = Date.now();
  const recent = (hits.get(key) ?? []).filter((t) => now - t < windowMs);
  if (recent.length >= max) {
    hits.set(key, recent);
    return true;
  }
  recent.push(now);
  hits.set(key, recent);
  return false;
}
setInterval(() => {
  const now = Date.now();
  for (const [k, arr] of hits) if (!arr.some((t) => now - t < 3600_000)) hits.delete(k);
}, 10 * 60 * 1000).unref?.();

let day = "";
let dayCount = 0;
function overDailyCap(): boolean {
  const today = new Date().toISOString().slice(0, 10);
  if (today !== day) {
    day = today;
    dayCount = 0;
  }
  const cap = parseInt(process.env.AI_DAILY_LIMIT || "300", 10);
  if (dayCount >= cap) return true;
  dayCount++;
  return false;
}

// ---- short memory of recent answers, so popular questions don't use up the AI quota ----
const CACHE_MS = 2 * 60 * 1000; // short, because prices and stock must stay fresh
const cache = new Map<string, { at: number; data: any }>();
function cacheGet(key: string) {
  const hit = cache.get(key);
  if (hit && Date.now() - hit.at < CACHE_MS) return hit.data;
  cache.delete(key);
  return null;
}
function cacheSet(key: string, data: any) {
  if (cache.size >= 300) cache.delete(cache.keys().next().value as string);
  cache.set(key, { at: Date.now(), data });
}

const visitor = (req: Request) =>
  String(req.headers["x-forwarded-for"] ?? req.ip ?? "unknown").split(",")[0].trim();

const fail = (res: Response, status: number, code: string, message: string) =>
  res.status(status).json({ error: { code, message } });

router.post("/chat", async (req: Request, res: Response) => {
  try {
    const parsed = Body.safeParse(req.body);
    if (!parsed.success) return fail(res, 400, "BAD_REQUEST", "Please type a shorter question.");
    const { messages } = parsed.data;
    const last = messages[messages.length - 1];
    if (last.role !== "user") return fail(res, 400, "BAD_REQUEST", "Please type a question.");

    const who = visitor(req);
    if (limited(`min:${who}`, 8, 60_000) || limited(`hr:${who}`, 60, 3600_000)) {
      return fail(res, 429, "RATE_LIMIT", "You're asking too fast. Please wait a moment.");
    }

    if (isGreeting(last.text)) {
      return res.json({
        data: {
          answer:
            "Hi! 👋 I can help with our combos, prices, free gifts and delivery. What would you like to know?",
          sources: [],
          needsHuman: false,
        },
      });
    }

    const knowledge = await getKnowledge();
    const found = retrieve(last.text, knowledge);

    // Nothing relevant in the shop's data: say so, don't spend AI quota, offer a person
    if (!found.matched) {
      return res.json({
        data: {
          answer:
            "I don't have that information. I can help with our combos, prices, free gifts and delivery, or you can talk to our team.",
          sources: [],
          needsHuman: true,
        },
      });
    }

    const byId = new Map(found.docs.map((d) => [d.id, d]));
    const everything = new Map(knowledge.map((d) => [d.id, d]));
    const toSources = (ids: string[]) =>
      ids
        .filter((id) => (byId.has(id) || everything.has(id)) && !id.startsWith("list:"))
        .slice(0, 4)
        .map((id) => {
          const d = (byId.get(id) ?? everything.get(id))!;
          return { id: d.id, title: d.title, link: d.link ?? null };
        });

    // Budget / "what do you have" questions: answer completely from the live catalogue, no AI.
    // (Only when the question is about nothing else, so a mixed question still reaches the AI.)
    const otherTopic = found.docs.some(
      (d, i) =>
        !d.id.startsWith("list:") && d.id !== "policy:overview" && d.type !== "combo" && found.scores[i] >= 5,
    );
    if (!otherTopic) {
      const list = listAnswer(found);
      if (list) {
        return res.json({
          data: { answer: list.text, sources: toSources(list.refs), needsHuman: false, mode: "list" },
        });
      }
    }

    // When the AI can't answer (busy, out of quota, down), a clearly on-topic question
    // still gets our own written answer instead of an error.
    const answerDirectly = (why: string) => {
      const d = directAnswer(found);
      if (!d) return null;
      console.warn(`[ai] answering directly from knowledge (${why})`);
      return {
        data: { answer: d.text, sources: toSources(d.sourceIds), needsHuman: false, mode: "direct" },
      };
    };

    // Same single question asked recently: reuse the answer (no AI call)
    const cacheKey =
      messages.length === 1 ? `${tokenize(last.text).join(" ")}|${found.docs.map((d) => d.id).join(",")}` : "";
    if (cacheKey) {
      const cached = cacheGet(cacheKey);
      if (cached) return res.json({ data: cached });
    }

    if (overDailyCap()) {
      const direct = answerDirectly("daily cap");
      if (direct) return res.json(direct);
      return fail(res, 503, "BUSY", "Our assistant is very busy today. Please talk to our team instead.");
    }

    let result;
    try {
      result = await askGemini(
        messages.map((m) => ({ role: m.role, text: m.text })),
        found.docs,
      );
    } catch (e) {
      if (e instanceof AIError) {
        const direct = answerDirectly(e.code);
        if (direct) return res.json(direct);
      }
      throw e;
    }

    // Only return sources the AI could really have used
    const sources = toSources(result.sources);

    const data = {
      answer: result.answer || "Sorry, I couldn't answer that. Please talk to our team.",
      sources,
      needsHuman: result.needsHuman || !result.answer,
      mode: "ai",
    };
    if (cacheKey && result.answer) cacheSet(cacheKey, data);
    res.json({ data });
  } catch (e) {
    if (e instanceof AIError) {
      const status = e.code === "UNAVAILABLE" ? 503 : e.code === "BUSY" ? 503 : 502;
      return fail(res, status, e.code, "Our assistant isn't available right now. Please talk to our team.");
    }
    console.error("POST /api/ai/chat failed:", e);
    fail(res, 500, "SERVER_ERROR", "Something went wrong. Please try again.");
  }
});

export default router;

const cut = (s: unknown, max = 500) => String(s ?? "").replace(/\s+/g, " ").trim().slice(0, max);

/** Splits long page text into pieces at line breaks so each piece can be found on its own. */
function chunk(text: string, max: number): string[] {
  const out: string[] = [];
  let cur = "";
  for (const line of String(text).split(/\n+/).map((l) => l.trim()).filter(Boolean)) {
    if (cur && cur.length + line.length > max) {
      out.push(cur);
      cur = "";
    }
    cur += (cur ? " " : "") + line;
  }
  if (cur) out.push(cur);
  return out;
}

/**
 * Product cards for the AI chat (pure functions, no database here).
 *
 * Cards are built in code from the shop's own data, never written by the AI,
 * so the name, price, stock and photo are always exact.
 */

import type { KDoc } from "./retrieve";

/** A knowledge document that may carry a photo. */
export type ImgDoc = KDoc & { image?: string };

export interface AICard {
  id: string;
  title: string;
  price: number;
  inStock: boolean;
  image: string | null;
  link: string | null;
}

/**
 * Returns a safe, small photo URL or null.
 * - Rejects base64 "data:" images (far too big to send in a chat reply).
 * - Shrinks Cloudinary photos to a 240x240 thumbnail so the chat loads fast.
 */
export function thumb(url?: string | null): string | null {
  if (!url) return null;
  const u = String(url).trim();
  if (!u || u.length > 2000 || u.startsWith("data:")) return null;
  if (!/^https?:\/\//i.test(u) && !u.startsWith("/")) return null;
  const m = u.match(/^(https?:\/\/res\.cloudinary\.com\/[^/]+\/image\/upload\/)(v\d+\/.+)$/i);
  if (m) return `${m[1]}f_auto,q_auto,c_fill,w_240,h_240/${m[2]}`;
  return u;
}

export function toCard(doc: KDoc): AICard | null {
  if (doc.type !== "combo" || typeof doc.price !== "number") return null;
  return {
    id: doc.id,
    title: doc.title,
    price: doc.price,
    inStock: !!doc.inStock,
    image: (doc as ImgDoc).image ?? null,
    link: doc.link ?? null,
  };
}

/** In-stock combos that cost no more than the budget, closest to the budget first. */
export function budgetCards(all: KDoc[], budget: number, limit = 6): { cards: AICard[]; total: number } {
  const fits = all
    .filter((d) => d.type === "combo" && d.inStock && typeof d.price === "number" && (d.price as number) <= budget)
    .sort((a, b) => (b.price as number) - (a.price as number));
  const cards = fits
    .slice(0, limit)
    .map(toCard)
    .filter((c): c is AICard => c !== null);
  return { cards, total: fits.length };
}

// Questions that mention a budget but are really about something else
const OTHER_TOPIC = /\b(deliver\w*|shipping|ship|fee|gift|pay\w*|refund|return\w*|warranty|contact|track\w*)\b/i;

/** True when a budget question should simply be answered with a list of products. */
export function isBudgetListQuestion(question: string): boolean {
  return !OTHER_TOPIC.test(question);
}

  new Date(iso).toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' });

function loadAI(): AIMsg[] {
  try {
    const raw = JSON.parse(localStorage.getItem(AI_KEY) || '[]');
    return Array.isArray(raw) ? raw.slice(-MAX_KEPT) : [];
  } catch {
    return [];
  }
}
function saveAI(m: AIMsg[]) {
  try {
    localStorage.setItem(AI_KEY, JSON.stringify(m.slice(-MAX_KEPT)));
  } catch {
    /* private mode: chat still works for this visit */
  }
}

function AIChat({ onHuman, whatsappLink }: { onHuman: (q?: string) => void; whatsappLink: string }) {
  const [messages, setMessages] = useState<AIMsg[]>(loadAI);
  const [text, setText] = useState('');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  const bottomRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => saveAI(messages), [messages]);
  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
  }, [messages.length, busy, error]);
  useEffect(() => {
    if (window.matchMedia('(min-width: 768px)').matches) inputRef.current?.focus();
  }, []);

  const lastQuestion = [...messages].reverse().find((m) => m.role === 'user')?.text;

  const ask = useCallback(
    async (question: string, base: AIMsg[]) => {
      const next = [...base, { id: uid(), role: 'user' as const, text: question }];
      setMessages(next);
      setBusy(true);
      setError('');
      try {
        const turns = next.slice(-10).map((m) => ({ role: m.role, text: m.text }));
        const r = await askAI(turns);
        setMessages((cur) => [
          ...cur,
          { id: uid(), role: 'assistant', text: r.answer, sources: r.sources, needsHuman: r.needsHuman, direct: r.mode === 'direct' },
        ]);
      } catch (err) {
        const msg =
          err instanceof ApiError && err.status === 429
            ? err.message
            : "Our assistant isn't available right now. You can talk to our team instead.";
        setError(msg);
      } finally {
        setBusy(false);
      }
    },
    [],
  );

  const submit = (e: React.FormEvent) => {
    e.preventDefault();
    const q = text.trim();
    if (!q || busy) return;
    setText('');
    void ask(q, messages);
  };

  const retry = () => {
    if (!lastQuestion || busy) return;
    // drop the unanswered last question, then ask it again
    const base = messages.slice(0, messages.length - 1);
    void ask(lastQuestion, base);
  };

  const close = () => {
    if (window.innerWidth < 768) chatUI.close();
  };

  return (
    <>
      <div className="flex-1 overflow-y-auto px-3 py-4 space-y-3">
        {messages.length === 0 && (
          <div className="pt-2">
            <div className="rounded-2xl bg-muted p-4">
              <p className="font-bold">Hi, I'm the Smart Combo assistant 👋</p>
              <p className="text-sm text-muted-foreground mt-1">
                Ask me about our combos, prices, free gifts or delivery. I only answer from what's
                really in our shop.
              </p>
            </div>
            <div className="mt-3 flex flex-wrap gap-2">
              {AI_STARTERS.map((s) => (
                <button
                  key={s}
                  type="button"
                  onClick={() => void ask(s, messages)}
                  className="px-3 py-2 rounded-full border border-primary/40 text-sm text-primary text-left hover:bg-primary/10"
                >
                  {s}
                </button>
              ))}
            </div>
          </div>
        )}

        {messages.map((m) => {
          const mine = m.role === 'user';
          return (
            <div key={m.id} className={`flex flex-col ${mine ? 'items-end' : 'items-start'}`}>
              <div
                className={`max-w-[85%] rounded-2xl px-3.5 py-2.5 ${
                  mine
                    ? 'bg-primary text-primary-foreground rounded-br-md'
                    : 'bg-muted text-foreground rounded-bl-md'
                }`}
              >
                <p className="text-[15px] leading-snug whitespace-pre-wrap break-words">{m.text}</p>
              </div>

              {!mine && m.direct && (
                <p className="mt-1 text-[10px] text-muted-foreground">Quick answer from our info pages</p>
              )}

              {!mine && m.sources && m.sources.length > 0 && (
                <div className="mt-1.5 flex flex-wrap items-center gap-1.5 max-w-[92%]">
                  <span className="text-[10px] uppercase tracking-wider text-muted-foreground">Source</span>
                  {m.sources.map((s) =>
                    s.link ? (
                      <Link
                        key={s.id}
                        to={s.link}
                        onClick={close}
                        className="text-xs px-2.5 py-1 rounded-full border border-border text-foreground/80 hover:border-primary hover:text-primary truncate max-w-[200px]"
                      >
                        {s.title}
                      </Link>
                    ) : (
                      <span
                        key={s.id}
                        className="text-xs px-2.5 py-1 rounded-full border border-border text-muted-foreground truncate max-w-[200px]"
                      >
                        {s.title}
                      </span>
                    ),
                  )}
                </div>
              )}

              {!mine && m.needsHuman && (
                <div className="mt-2 flex flex-wrap gap-2">
                  <button
                    type="button"
                    onClick={() => onHuman(lastQuestion)}
                    className="px-3 h-9 rounded-full bg-primary text-primary-foreground text-sm font-semibold"
                  >
                    Talk to a person
                  </button>
                  {whatsappLink && (
                    <a
                      href={whatsappLink}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="px-3 h-9 rounded-full border border-emerald-500/50 text-emerald-500 text-sm font-semibold flex items-center"
                    >
                      WhatsApp
                    </a>
                  )}
                </div>
              )}
            </div>
          );
        })}

        {busy && (
          <div className="flex justify-start">
            <div className="rounded-2xl rounded-bl-md bg-muted px-4 py-3 flex gap-1" aria-label="Assistant is typing">
              {[0, 150, 300].map((d) => (
                <span
                  key={d}
                  className="w-2 h-2 rounded-full bg-muted-foreground/60 animate-bounce"
                  style={{ animationDelay: `${d}ms` }}
                />
              ))}
            </div>
          </div>
        )}

        {error && (
          <div className="rounded-2xl border border-destructive/40 bg-destructive/10 p-3">
            <p className="text-sm">{error}</p>
            <div className="mt-2 flex flex-wrap gap-2">
              <button
                type="button"
                onClick={retry}
                className="px-3 h-9 rounded-full border border-border text-sm font-semibold hover:bg-foreground/5"
              >
                Try again
              </button>
              <button
                type="button"
                onClick={() => onHuman(lastQuestion)}
                className="px-3 h-9 rounded-full bg-primary text-primary-foreground text-sm font-semibold"
              >
                Talk to a person
              </button>
            </div>
          </div>
        )}
        <div ref={bottomRef} />
      </div>

      <div className="shrink-0 border-t border-border bg-background">
        {messages.length > 0 && (
          <div className="px-4 pt-1.5 flex justify-end">
            <button
              type="button"
              onClick={() => {
                setMessages([]);
                setError('');
              }}
              className="text-[11px] text-muted-foreground hover:text-foreground underline"
            >
              Clear chat
            </button>
          </div>
        )}
        <form
          onSubmit={submit}
          className="flex items-center gap-2 px-3 pt-2"
          style={{ paddingBottom: 'calc(env(safe-area-inset-bottom, 0px) + 0.25rem)' }}
        >
          <input
            ref={inputRef}
            value={text}
            onChange={(e) => setText(e.target.value)}
            maxLength={500}
            enterKeyHint="send"
            placeholder="Ask about a product, price or delivery"
            className="flex-1 min-w-0 h-12 rounded-full border border-border bg-muted/50 px-4 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
          />
          <button
            type="submit"
            disabled={busy || !text.trim()}
            aria-label="Send"
            className="w-12 h-12 shrink-0 rounded-full bg-primary text-primary-foreground flex items-center justify-center active:scale-95 transition disabled:opacity-50"
          >
            {busy ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
          </button>
        </form>
        <p
          className="px-4 pb-2 text-[10px] text-muted-foreground text-center"
          style={{ paddingBottom: 'calc(env(safe-area-inset-bottom, 0px) + 0.5rem)' }}
        >
          AI can make mistakes. Always confirm the final price at checkout.
        </p>
      </div>
    </>
  );
}

function HumanChat({ prefill }: { prefill: string }) {
  const [token, setToken] = useState<string | null>(getChatToken());
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [loading, setLoading] = useState<boolean>(!!getChatToken());
  const [sending, setSending] = useState(false);
  const [error, setError] = useState('');

  const [name, setName] = useState('');
  const [phone, setPhone] = useState('');
  const [firstMessage, setFirstMessage] = useState(prefill);
  const [text, setText] = useState('');

  const bottomRef = useRef<HTMLDivElement>(null);

  const load = useCallback(async (t: string) => {
    try {
      setMessages(await fetchChat(t));
    } catch (err) {
      if (err instanceof ApiError && err.status === 404) {
        // The chat was removed on the server: start fresh
        clearChatToken();
        setToken(null);
        setMessages([]);
      }
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    if (!token) return;
    load(token);
    const id = setInterval(() => {
      if (document.visibilityState === 'visible') load(token);
    }, 4000);
    return () => clearInterval(id);
  }, [token, load]);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages.length]);

  const begin = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!name.trim() || !firstMessage.trim() || sending) return;
    setSending(true);
    setError('');
    try {
      const r = await startChat({
        name: name.trim(),
        phone: phone.trim() || undefined,
        message: firstMessage.trim(),
      });
      saveChatToken(r.token);
      setToken(r.token);
      setMessages(r.messages);
    } catch (err: any) {
      setError(err?.message || 'Could not start the chat. Please try again.');
    } finally {
      setSending(false);
    }
  };

  const send = async (e: React.FormEvent) => {
    e.preventDefault();
    const t = text.trim();
    if (!t || !token || sending) return;
    setSending(true);
    setError('');
    try {
      await sendChatMessage(token, t);
      setText('');
      await load(token);
    } catch (err: any) {
      setError(err?.message || 'Message not sent. Please try again.');
    } finally {
      setSending(false);
    }
  };

  if (!token) {
    return (
      <form onSubmit={begin} className="flex-1 overflow-y-auto px-4 py-5 space-y-4">
        <div>
          <h2 className="text-xl font-black">Talk to our team</h2>
          <p className="text-sm text-muted-foreground mt-1">
            A real person will reply here. No account needed.
          </p>
        </div>

        <label className="block">
          <span className="block text-[11px] font-bold text-muted-foreground uppercase tracking-wider mb-1.5">
            Your name *
          </span>
          <input
            value={name}
            onChange={(e) => setName(e.target.value)}
            maxLength={60}
            autoComplete="name"
            className="w-full h-12 rounded-xl border border-border bg-transparent px-3.5 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
            placeholder="e.g. Ade"
          />
        </label>

        <label className="block">
          <span className="block text-[11px] font-bold text-muted-foreground uppercase tracking-wider mb-1.5">
            Phone (optional)
          </span>
          <input
            value={phone}
            onChange={(e) => setPhone(e.target.value)}
            maxLength={30}
            inputMode="tel"
            autoComplete="tel"
            className="w-full h-12 rounded-xl border border-border bg-transparent px-3.5 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
            placeholder="So we can call you back"
          />
        </label>

        <div>
          <span className="block text-[11px] font-bold text-muted-foreground uppercase tracking-wider mb-1.5">
            Your message *
          </span>
          <div className="flex flex-wrap gap-2 mb-2">
            {QUICK.map((q) => (
              <button
                key={q}
                type="button"
                onClick={() => setFirstMessage(q)}
                className="px-3 h-9 rounded-full border border-emerald-500/50 text-emerald-500 text-sm hover:bg-emerald-500/10"
              >
                {q}
              </button>
            ))}
          </div>
          <textarea
            value={firstMessage}
            onChange={(e) => setFirstMessage(e.target.value)}
            maxLength={1000}
            rows={4}
            className="w-full rounded-xl border border-border bg-transparent px-3.5 py-3 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
            placeholder="Type your message here"
          />
        </div>

        {error && <p className="text-sm text-destructive">{error}</p>}

        <button
          type="submit"
          disabled={sending || !name.trim() || !firstMessage.trim()}
          className="btn-primary w-full h-12 font-bold text-sm flex items-center justify-center gap-2 disabled:opacity-50"
        >
          {sending ? <Loader2 className="w-4 h-4 animate-spin" /> : <Send className="w-4 h-4" />}
          Start chat
        </button>
      </form>
    );
  }

  return (
    <>
      <div className="flex-1 overflow-y-auto px-3 py-4 space-y-2.5">
        {loading ? (
          <div className="py-16 flex justify-center">
            <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
          </div>
        ) : (
          messages.map((m) => {
            const mine = m.sender === 'customer';
            return (
              <div key={m.id} className={`flex ${mine ? 'justify-end' : 'justify-start'}`}>
                <div
                  className={`max-w-[80%] rounded-2xl px-3.5 py-2 ${
                    mine
                      ? 'bg-primary text-primary-foreground rounded-br-md'
                      : 'bg-muted text-foreground rounded-bl-md'
                  }`}
                >
                  <p className="text-[15px] leading-snug whitespace-pre-wrap break-words">{m.text}</p>
                  <p className={`text-[10px] mt-1 text-right ${mine ? 'opacity-60' : 'text-muted-foreground'}`}>
                    {timeOf(m.createdAt)}
                  </p>
                </div>
              </div>
            );
          })
        )}
        <div ref={bottomRef} />
      </div>

      {error && <p className="px-4 pb-1 text-xs text-destructive">{error}</p>}

      <form
        onSubmit={send}
        className="shrink-0 flex items-center gap-2 px-3 pt-2 border-t border-border bg-background"
        style={{ paddingBottom: 'calc(env(safe-area-inset-bottom, 0px) + 0.5rem)' }}
      >
        <input
          value={text}
          onChange={(e) => setText(e.target.value)}
          maxLength={1000}
          enterKeyHint="send"
          placeholder="Type a message"
          className="flex-1 min-w-0 h-12 rounded-full border border-border bg-muted/50 px-4 text-base placeholder:text-muted-foreground outline-none focus:border-primary/60"
        />
        <button
          type="submit"
          disabled={sending || !text.trim()}
          aria-label="Send"
          className="w-12 h-12 shrink-0 rounded-full bg-primary text-primary-foreground flex items-center justify-center active:scale-95 transition disabled:opacity-50"
        >
          {sending ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
        </button>
      </form>
    </>
  );
}

import { useState } from 'react';
import { Link } from 'react-router-dom';
import { ShoppingBag } from 'lucide-react';
import type { AICard } from '@/api/ai';

const naira = (n: number) => '₦' + Math.round(n).toLocaleString('en-NG');

function Photo({ src, alt }: { src: string | null; alt: string }) {
  const [failed, setFailed] = useState(false);
  if (!src || failed) {
    return (
      <div className="w-full h-full flex items-center justify-center bg-muted text-muted-foreground">
        <ShoppingBag className="w-7 h-7" />
      </div>
    );
  }
  return (
    <img
      src={src}
      alt={alt}
      loading="lazy"
      onError={() => setFailed(true)}
      className="w-full h-full object-cover"
    />
  );
}

const clamp2: React.CSSProperties = {
  display: '-webkit-box',
  WebkitLineClamp: 2,
  WebkitBoxOrient: 'vertical',
  overflow: 'hidden',
};

export function AIProductCards({
  cards,
  onNavigate,
}: {
  cards: AICard[];
  onNavigate?: () => void;
}) {
  if (!cards.length) return null;

  return (
    <div className="mt-2 grid grid-cols-2 gap-2 w-full max-w-[92%]">
      {cards.map((c) => {
        const body = (
          <>
            <div className="aspect-square w-full overflow-hidden rounded-xl bg-background">
              <Photo src={c.image} alt={c.title} />
            </div>
            <p className="mt-2 text-sm font-semibold leading-snug break-words" style={clamp2}>
              {c.title}
            </p>
            <p className="text-sm font-bold text-primary">{naira(c.price)}</p>
            {!c.inStock && <p className="text-xs text-muted-foreground">Out of stock</p>}
          </>
        );
        const cls =
          'block min-w-0 rounded-2xl border border-border bg-muted p-2 text-left transition-colors';
        return c.link ? (
          <Link key={c.id} to={c.link} onClick={onNavigate} className={`${cls} hover:border-primary`}>
            {body}
          </Link>
        ) : (
          <div key={c.id} className={cls}>
            {body}
          </div>
        );
      })}
    </div>
  );
}

/**
 * AI assistant API. Nothing is stored on the server; the browser sends the
 * last few messages with each question.
 */

import { apiFetch } from "./client";

export interface AISource {
  id: string;
  title: string;
  link: string | null;
}

export interface AIReply {
  answer: string;
  sources: AISource[];
  needsHuman: boolean;
  /** "direct" = built from our own pages because the AI was unavailable */
  mode?: "ai" | "direct" | "list";
}

export interface AITurn {
  role: "user" | "assistant";
  text: string;
}

export async function askAI(messages: AITurn[]): Promise<AIReply> {
  const res = await apiFetch<{ data: AIReply }>("/api/ai/chat", {
    method: "POST",
    body: JSON.stringify({ messages }),
  });
  return res.data;
}
