/**
 * Order model.
 *
 * Items now snapshot the customer's variant selections so admin knows
 * exactly what to ship.
 *
 * phoneKey is the customer's phone number cleaned to its last 10 digits. It is
 * how spending is added up for the free gift (customers do not have accounts).
 */

import { Schema, model, Document, Types } from "mongoose";

export type OrderStatus =
  | "pending"
  | "paid"
  | "processing"
  | "shipped"
  | "delivered"
  | "cancelled"
  | "refunded";

export type OrderSource = "website" | "whatsapp" | "walkin" | "social" | "other";

export type PaymentMethod =
  | "paystack"
  | "flutterwave"
  | "bank_transfer"
  | "cod"
  | "whatsapp"
  | "cash"
  | "pos";

export interface OrderItemSnapshot {
  comboId: Types.ObjectId;
  slug: string;
  name: string;
  tagline?: string;
  thumbnailUrl?: string;
  unitPrice: number;
  quantity: number;
  subtotal: number;
  selectedVariants?: Record<string, string>;   // itemId → alternativeId
  variantSummary?: string;                     // human-readable
  isFreeGift?: boolean;                        // identifies free gift items
  preorder?: boolean;                          // pre-order line: nothing was taken from stock
  leadDaysMin?: number;
  leadDaysMax?: number;
}

export interface ShippingAddress {
  fullName: string;
  phone: string;
  email: string;
  state: string;
  city: string;
  street: string;
  landmark?: string;
}

export interface OrderDocument extends Document {
  orderNumber: string;
  items: OrderItemSnapshot[];
  subtotal: number;
  shippingFee: number;
  total: number;
  status: OrderStatus;
  paymentMethod: PaymentMethod;
  paymentReference?: string;
  stockRestored?: boolean;
  /** where the order came from; "website" for normal checkout */
  source?: OrderSource;
  /** true when the owner recorded it by hand (WhatsApp, walk-in...) */
  manual?: boolean;
  /** true when the order reaches a bulk price level (flat bulk delivery, paid online) */
  isBulk?: boolean;
  /** the order has pre-order items */
  hasPreorder?: boolean;
  /** paid pre-order whose goods you have not got yet: kept out of the packing queue */
  awaitingGoods?: boolean;
  /** "now" = the ready pieces are sent first (in parts); "together" = everything in one go; empty = not a mixed order */
  partialChoice?: "now" | "together" | "";
  /** only when partialChoice is "now": the parts and where each one is */
  shipments?: Array<{
    n: number;
    itemIdx: number[];
    status: "waiting" | "to_pack" | "packed" | "shipped" | "delivered";
    trackingNumber?: string;
    trackingProviderUrl?: string;
    readyAt?: Date;
    packedAt?: Date;
    shippedAt?: Date;
    deliveredAt?: Date;
  }>;
  paystackRefs?: string[];
  shipping: ShippingAddress;
  phoneKey?: string;
  notes?: string;
  trackingNumber?: string;
  trackingProviderUrl?: string;
  adminNotes?: string;
  paidAt?: Date;
  shippedAt?: Date;
  deliveredAt?: Date;
  createdAt: Date;
  updatedAt: Date;
}

    preorder: { type: Boolean, default: false },
    leadDaysMin: { type: Number },

      type: String,
      enum: ["pending", "paid", "processing", "shipped", "delivered", "cancelled", "refunded"],
      default: "pending",
      index: true,
    },
    paymentMethod: {
      type: String,
      enum: ["paystack", "flutterwave", "bank_transfer", "cod", "whatsapp", "cash", "pos"],
      required: true,
    },
    paymentReference: { type: String, index: true },
    // true once this order's stock has been given back (cancel / refund / expiry)
    stockRestored: { type: Boolean, default: false },
    // every Paystack reference started for this order (a customer can retry), used to re-check unpaid orders
    paystackRefs: { type: [String], default: [] },
    shipping: { type: ShippingAddressSchema, required: true },
    phoneKey: { type: String },
    source: { type: String, enum: ["website", "whatsapp", "walkin", "social", "other"], default: "website", index: true },
    manual: { type: Boolean, default: false },
    isBulk: { type: Boolean, default: false },
    hasPreorder: { type: Boolean, default: false },
    awaitingGoods: { type: Boolean, default: false, index: true },
    partialChoice: { type: String, enum: ["now", "together", ""], default: "" },
    shipments: { type: [ShipmentSchema], default: undefined },
    notes: { type: String },
    trackingNumber: { type: String },
    trackingProviderUrl: { type: String },
    adminNotes: { type: String },
    paidAt: { type: Date },
    shippedAt: { type: Date },
    deliveredAt: { type: Date },
  },
  { timestamps: true },
);

OrderSchema.index({ createdAt: -1 });
OrderSchema.index({ "shipping.phone": 1, orderNumber: 1 });
OrderSchema.index({ phoneKey: 1, createdAt: -1 });

export const Order = model<OrderDocument>("Order", OrderSchema);

/**

 *
 * NEW: buildItemSnapshots reads the customer's selectedVariants per cart

async function generateOrderNumber(): Promise<string> {
  const year = new Date().getFullYear();
  const seq = await nextSeq(`order:${year}`);
  return `SC-${year}-${String(seq).padStart(6, "0")}`;
}

/** Flat delivery fee for customers in Ilorin. Everywhere else Jumia sets the price and the customer pays Jumia directly. */
export const LOCAL_DELIVERY_FEE = 1000;

async function calculateShippingFee(
  _paymentMethod: PaymentMethod,
  city: string,
): Promise<number> {
  const settings = await SiteSettings.findOne().lean();
  const localCities = (settings?.shipping?.codCities ?? ["ilorin"]).map((c: string) => c.toLowerCase());
  const cityLower = city.trim().toLowerCase();
  return localCities.includes(cityLower) ? LOCAL_DELIVERY_FEE : 0;
}

/**
 * Builds a human-readable summary of what the customer picked for each slot.
 * Example: "Casio Watch · Bluetooth Audio Glasses · Premium Bracelet"
 */
function buildVariantSummary(
  combo: any,
  selectedVariants: Record<string, string> | undefined,
): string {
  return (combo.items ?? [])
    .map((item: any) => {
      const chosenId = selectedVariants?.[item.id];
      const alt = chosenId ? (item.alternatives ?? []).find((a: any) => a.id === chosenId) : undefined;
      const name = alt ? alt.name : item.name; // default when nothing was chosen
      // include the chosen color so you know exactly what to pack
      const colorId = selectedVariants?.[`${item.id}__color`];
      const color = colorId ? (item.colors ?? []).find((c: any) => c.id === colorId) : undefined;
      return color ? `${name} (${color.name})` : name;
    })
    .join(" · ");
}

async function buildItemSnapshots(
  items: CartItem[],
  priorSpend = 0,
): Promise<OrderItemSnapshot[]> {
  if (!items?.length) {
    throw new OrderError(400, "EMPTY_CART", "Order has no items");
  }

  // Pass 1: Calculate subtotal from paid items only
  let paidSubtotal = 0;
  const freeGiftItems = items.filter((it) => it.isFreeGift);

  if (freeGiftItems.length > 1) {
    throw new OrderError(400, "INVALID_GIFT", "Only 1 free gift is allowed per order");
  }

  for (const cartItem of items) {
    if (!cartItem.isFreeGift) {
      if (!cartItem.comboId) {
        throw new OrderError(400, "INVALID_ITEM", "Item missing combo reference");
      }

      const combo = await Combo.findById(cartItem.comboId);
      if (combo && combo.isActive) {
        paidSubtotal += unitPriceFor(combo, cartItem.quantity) * cartItem.quantity;
      }
    }
  }

  // Validate threshold if a free gift is attached
  // Earlier paid spending by the same phone number counts too.
  const giftUnlocked = paidSubtotal + priorSpend >= GIFT_THRESHOLD;
  if (freeGiftItems.length === 1 && !giftUnlocked) {
    throw new OrderError(
      400,
      "GIFT_THRESHOLD_NOT_MET",
      "Your spending (this order plus the last 30 days) must reach ₦40,000 to qualify for a free gift",
    );
  }

  const snapshots: OrderItemSnapshot[] = [];

  // Pass 2: Build snapshot objects
  for (const cartItem of items) {
    if (!cartItem.comboId) {
      throw new OrderError(400, "INVALID_ITEM", "Item missing combo reference");
    }
    if (!Number.isInteger(cartItem.quantity) || cartItem.quantity < 1 || cartItem.quantity > BULK_MAX_QTY) {
      throw new OrderError(400, "INVALID_ITEM", "Item quantity must be a whole number of 1 or more");
    }

    // Free gifts normally come from the Gift Bank
    if (cartItem.isFreeGift) {
      const gift: any = await Gift.findById(cartItem.comboId).catch(() => null);
      if (gift) {
        if (gift.isActive === false || (gift.stock ?? 0) < 1) {
          throw new OrderError(409, "INSUFFICIENT_STOCK", `${gift.name} is no longer available`);
        }
        snapshots.push({
          comboId: gift._id,
          slug: `gift-${gift._id}`,
          name: `[FREE GIFT] ${gift.name}`,
          tagline: gift.description ?? "",
          thumbnailUrl: gift.images?.[0] || gift.image || "",
          unitPrice: 0,
          quantity: 1,
          subtotal: 0,
          selectedVariants: {},
          variantSummary: "",
          isFreeGift: true,
        });
        continue;
      }
    }

    const combo = await Combo.findById(cartItem.comboId);
    if (!combo) {
      throw new OrderError(404, "COMBO_NOT_FOUND", `Combo not found: ${cartItem.comboId}`);
    }
    if (!combo.isActive) {
      throw new OrderError(400, "COMBO_INACTIVE", `${combo.name} is no longer available`);
    }
    if (cartItem.quantity > maxQtyFor(combo)) {
      throw new OrderError(
        400,
        "INVALID_ITEM",
        `You can order up to ${maxQtyFor(combo)} of ${combo.name} at a time`,
      );
    }
    // Pre-order switch on: pieces in stock come from stock, the rest are bought after payment.
    const plan = planQty(combo.stockLeft, cartItem.quantity, Boolean((combo as any).preorder));
    if (!plan) {
      throw new OrderError(
        409,
        "INSUFFICIENT_STOCK",
        `Only ${combo.stockLeft} of ${combo.name} available`,
      );
    }

    const isFree = Boolean(cartItem.isFreeGift && giftUnlocked);

    // Ensure the combo itself is eligible to be a free gift candidate
    if (isFree && !combo.isFreeGiftCandidate && combo.totalPrice > 5000) {
      throw new OrderError(
        400,
        "INELIGIBLE_FREE_GIFT",
        `${combo.name} is not eligible as a free gift`,
      );
    }

    // The price level follows the TOTAL quantity, even when the line is split in two below.
    const unitPrice = isFree ? 0 : unitPriceFor(combo, cartItem.quantity);

    const thumbnailUrl = combo.items?.[0]?.images?.[0]?.url ?? "";
    const variantSummary = buildVariantSummary(combo, cartItem.selectedVariants);
    const base = {
      comboId: combo._id,
      slug: combo.slug,
      name: isFree ? `[FREE GIFT] ${combo.name}` : combo.name,
      tagline: combo.tagline,
      thumbnailUrl,
      unitPrice,
      selectedVariants: cartItem.selectedVariants ?? {},
      variantSummary,
      isFreeGift: isFree,
    };
    const lead = {
      preorder: true,
      leadDaysMin: Number((combo as any).leadDaysMin) || 5,
      leadDaysMax: Number((combo as any).leadDaysMax) || 10,
    };

    if (isFree) {
      snapshots.push({ ...base, quantity: 1, subtotal: 0, ...(plan.ready === 0 ? lead : {}) });
    } else {
      // Ready pieces and pieces we still have to get become two lines of the same product.
      if (plan.ready > 0) snapshots.push({ ...base, quantity: plan.ready, subtotal: unitPrice * plan.ready });
      if (plan.waiting > 0) snapshots.push({ ...base, quantity: plan.waiting, subtotal: unitPrice * plan.waiting, ...lead });
    }
  }

  return snapshots;
}

/** Takes one item's stock. Says where it came from, or throws if sold out. */
async function takeStockForItem(item: OrderItemSnapshot): Promise<"gift" | "combo"> {
  // Pre-order line: there is no stock to take
  if (item.preorder) return "combo";
  // Gift Bank gifts have their own stock
  if (item.isFreeGift) {
    const gift = await Gift.findOneAndUpdate(
      { _id: item.comboId, stock: { $gte: 1 } },
      { $inc: { stock: -1 } },
    );
    if (gift) return "gift";
  }
  const updated = await Combo.findOneAndUpdate(
    { _id: item.comboId, stockLeft: { $gte: item.quantity } },
    { $inc: { stockLeft: -item.quantity } },
  );
  if (!updated) {
    throw new OrderError(409, "INSUFFICIENT_STOCK", `${item.name} sold out before checkout`);
  }
  return "combo";
}

async function giveStockBackForItem(item: OrderItemSnapshot, kind?: "gift" | "combo"): Promise<void> {
  // Pre-order line: nothing was taken, so nothing goes back
  if (item.preorder) return;
  if (kind === "combo") {
    await Combo.updateOne({ _id: item.comboId }, { $inc: { stockLeft: item.quantity } });
    return;
  }
  if (kind === "gift" || item.isFreeGift) {
    const r = await Gift.updateOne({ _id: item.comboId }, { $inc: { stock: 1 } });
    if (r.matchedCount > 0) return;
  }
  await Combo.updateOne({ _id: item.comboId }, { $inc: { stockLeft: item.quantity } });
}

/** All-or-nothing: if any item is sold out, everything already taken is put back. */
async function decrementStock(items: OrderItemSnapshot[]): Promise<void> {
  const taken: Array<{ item: OrderItemSnapshot; kind: "gift" | "combo" }> = [];
  try {
    for (const item of items) {
      const kind = await takeStockForItem(item);
      taken.push({ item, kind });
    }
  } catch (err) {
    for (const t of taken) {
      await giveStockBackForItem(t.item, t.kind).catch((e) =>
        console.error("Stock rollback failed:", e),
      );
    }
    throw err;
  }
}

/** Pre-order details for the emails: undefined when the order has no pre-order pieces. */
function preorderInfoFor(order: any): { min: number; max: number; inParts: boolean } | undefined {
  const pre = (order.items ?? []).filter((it: any) => it.preorder);
  if (pre.length === 0) return undefined;
  return {
    min: Math.min(...pre.map((it: any) => Number(it.leadDaysMin) || 5)),
    max: Math.max(...pre.map((it: any) => Number(it.leadDaysMax) || 10)),
    inParts: order.partialChoice === "now",
  };
}

function deliveryRouteOf(order: any): "local" | "jumia" | "pickup" {
  if (String(order.shipping?.street ?? "").startsWith("Pickup")) return "pickup";
  return String(order.shipping?.city ?? "").trim().toLowerCase() === "ilorin" ? "local" : "jumia";
}

function buildEmailContext(order: any) {
  return {
    orderNumber: order.orderNumber,
    customerName: order.shipping.fullName,
    customerEmail: order.shipping.email,
    total: order.total,
    items: order.items.map((it: any) => ({
      name: (it.variantSummary || it.name) + (it.preorder ? " (pre-order)" : ""),
      quantity: it.quantity,
      subtotal: it.subtotal,
    })),
    shippingAddress: {
      fullName: order.shipping.fullName,
      street: order.shipping.street,
      city: order.shipping.city,
      state: order.shipping.state,
      phone: order.shipping.phone,
    },
    paymentMethod: order.paymentMethod,
    trackingUrl: `${config.siteUrl}/order-tracking?order=${encodeURIComponent(order.orderNumber)}`,
    preorder: preorderInfoFor(order),
    deliveryRoute: deliveryRouteOf(order),
  };
}

/** Extra options when the owner records a sale by hand (WhatsApp, walk-in...). */
export interface ManualSaleOptions {
  source: "whatsapp" | "walkin" | "social" | "other";
  /** payment already received */
  paid: boolean;
  /** customer collects the goods: no delivery fee */
  pickup: boolean;
}

export async function createOrder(input: CreateOrderInput, manual?: ManualSaleOptions) {
  const phoneKey = normalizePhone(input.shipping.phone);
  const priorSpend = await getPriorSpend(phoneKey);
  const items = await buildItemSnapshots(input.items, priorSpend);

  const subtotal = items.reduce((sum, it) => sum + it.subtotal, 0);
  // Bulk orders: one flat Ilorin delivery fee for the whole order, and they must be paid online.
  const bulk = await orderHasBulk(input.items);
  if (bulk && !manual && input.paymentMethod !== "paystack") {
    throw new OrderError(400, "BULK_PAY_ONLINE", "Bulk orders must be paid online. Please choose Pay online.");
  }
  const baseFee = manual?.pickup
    ? 0
    : await calculateShippingFee(
        input.paymentMethod,
        input.shipping.city,
      );
  const shippingFee = bulk && baseFee > 0 ? BULK_DELIVERY_FEE : baseFee;

  // Pre-order items: the customer must tick the box and pay online (you buy the goods after payment).
  const hasPre = items.some((i) => i.preorder);
  if (hasPre && !manual) {
    if (!input.ackPreorder) {
      throw new OrderError(400, "PREORDER_ACK", "Please tick the box to say you understand some items are pre-order.");
    }
    if (input.paymentMethod !== "paystack") {
      throw new OrderError(400, "PREORDER_PAY_ONLINE", "Pre-order items must be paid online. Please choose Pay online.");
    }
  }
  const total = subtotal + shippingFee;

  // Some pieces ready and some to get: in Ilorin the customer may choose to have the ready pieces sent first.
  // Everywhere else (Jumia) and for hand-recorded sales, everything is sent together.
  const mixed = isMixed(items);
  const partialChoice: "now" | "together" | "" = !mixed
    ? ""
    : !manual && input.partialChoice === "now" && baseFee > 0
      ? "now"
      : "together";

  const orderNumber = await generateOrderNumber();

  await decrementStock(items);

  const initialStatus: OrderStatus = manual
    ? manual.paid
      ? "paid"
      : "pending"
    : input.paymentMethod === "cod"
      ? "processing"
      : "pending";

  let order;
  try {
    order = await Order.create({
      orderNumber,
      items,
      subtotal,
      shippingFee,
      total,
      isBulk: bulk,
      hasPreorder: hasPre,
      awaitingGoods: hasPre,
      partialChoice,
      ...(partialChoice === "now" ? { shipments: buildShipments(items) } : {}),
      status: initialStatus,
      paymentMethod: input.paymentMethod,
      shipping: input.shipping,
      phoneKey,
      notes: input.notes,
      ...(manual
        ? {
            source: manual.source,
            manual: true,
            ...(manual.paid ? { paidAt: new Date(), paymentReference: `manual-${input.paymentMethod}` } : {}),
          }
        : {}),
    });
  } catch (err) {
    // The order could not be saved: put the stock back so nothing is lost.
    for (const item of items) {
      await giveStockBackForItem(item).catch((e) => console.error("Stock rollback failed:", e));
    }
    throw err;
  }

  // Remember what these goods cost you right now, so profit stays correct if prices change later.
  await recordOrderCost(order);

  // A sale recorded by hand needs no "order placed" email: you are already talking to the customer.
  if (!manual) {
    const ctx = buildEmailContext(order);
    sendOrderPlaced(ctx).catch(() => {});
    sendNewOrderAdminAlert(ctx).catch(() => {});
  }

  return order;
}

/**
 * Checkout asks this before the customer pays: which pieces are ready, which must be fetched,
 * and whether the customer can choose to have the ready pieces sent first. Changes nothing.
 */
export async function previewOrder(items: CartItem[], city: string) {
  const lines: Array<{
    comboId: string;
    name: string;
    ready: number;
    waiting: number;
    leadDaysMin: number;
    leadDaysMax: number;
    /** the customer asks for more than we have and the product is not a pre-order product */
    short: boolean;
    want: number;
  }> = [];
  let subtotal = 0;
  for (const it of items) {
    if (it.isFreeGift || !it.comboId) continue;
    const q = Math.floor(Number(it.quantity));
    if (!Number.isInteger(q) || q < 1 || q > BULK_MAX_QTY) continue;
    const combo: any = await Combo.findById(it.comboId).lean().catch(() => null);
    if (!combo || !combo.isActive) continue;
    const plan = planQty(combo.stockLeft, q, Boolean(combo.preorder));
    subtotal += unitPriceFor(combo, q) * q;
    lines.push({
      comboId: String(combo._id),
      name: combo.name,
      ready: plan ? plan.ready : Math.max(0, combo.stockLeft),
      waiting: plan ? plan.waiting : 0,
      leadDaysMin: Number(combo.leadDaysMin) || 5,
      leadDaysMax: Number(combo.leadDaysMax) || 10,
      short: !plan,
      want: q,
    });
  }
  const waitingLines = lines.filter((l) => l.waiting > 0);
  const hasPre = waitingLines.length > 0;
  const mixed = hasPre && lines.some((l) => l.ready > 0);
  const local = city.trim() !== "" && (await calculateShippingFee("paystack", city)) > 0;
  return {
    lines,
    hasPre,
    mixed,
    local,
    subtotal,
    leadDaysMin: hasPre ? Math.min(...waitingLines.map((l) => l.leadDaysMin)) : 0,
    leadDaysMax: hasPre ? Math.max(...waitingLines.map((l) => l.leadDaysMax)) : 0,
  };
}

/** The goods arrived (To get > Got it). A split order also opens its waiting part for packing. */
export async function gotItOrder(id: any, waiting: Record<string, any>): Promise<any> {
  const cur: any = await Order.findOne({ _id: id, ...waiting }).lean();
  if (!cur) return null;
  const set: Record<string, any> = { awaitingGoods: false };
  if (cur.partialChoice === "now" && cur.shipments?.length) set.shipments = openGotIt(cur.shipments);
  return Order.findOneAndUpdate({ _id: cur._id, ...waiting }, { $set: set }, { new: true }).lean();
}

/**
 * Moves ONE part of a split order: pack, ship or deliver.
 * The order stays "processing" until every part is delivered, then it becomes "delivered".
 * Returns { error } when the step is not allowed.
 */
export async function stepPart(
  orderNumber: string,
  n: number,
  step: Step,
  opts: { trackingNumber?: string; trackingProviderUrl?: string } = {},
): Promise<{ order?: any; error?: string }> {
  for (let attempt = 0; attempt < 3; attempt++) {
    const cur: any = await Order.findOne({ orderNumber: orderNumber.trim().toUpperCase() }).lean();
    if (!cur) return { error: "Order not found" };
    if (cur.partialChoice !== "now" || !cur.shipments?.length) return { error: "This order is not sent in parts." };
    if (cur.status !== "paid" && cur.status !== "processing") {
      return { error: `An order that is ${cur.status} cannot be changed here.` };
    }
    const now = new Date();
    const r = stepShipments(cur.shipments, n, step, { ...opts, now });
    if (r.error) return { error: r.error };

    const set: Record<string, any> = { shipments: r.shipments, status: "processing" };
    if (step === "ship" && !cur.shippedAt) set.shippedAt = now;
    if (r.allDelivered) {
      set.status = "delivered";
      set.deliveredAt = now;
    }
    // Only write if nobody changed the order since we read it
    const saved = await Order.findOneAndUpdate(
      { _id: cur._id, updatedAt: cur.updatedAt },
      { $set: set },
      { new: true },
    );
    if (!saved) continue;

    const ctx = buildEmailContext(saved);
    const part = (r.shipments.find((s) => s.n === n) ?? { itemIdx: [] as number[] }) as { itemIdx: number[] };
    const partItems = ctx.items.filter((_: any, i: number) => part.itemIdx.includes(i));
    if (step === "ship") {
      sendOrderShipped({ ...ctx, items: partItems, partLabel: `Part ${n} of ${r.shipments.length}`, trackingNumber: opts.trackingNumber, trackingProviderUrl: opts.trackingProviderUrl }).catch(() => {});
    }
    if (r.allDelivered) sendOrderDelivered(ctx).catch(() => {});
    return { order: saved };
  }
  return { error: "The order was changed at the same time. Please refresh and try again." };
}

/**
 * A Paystack payment only counts for an order when the reference was started for THAT order,
 * the payment succeeded, and the amount paid equals the order total (in kobo).
 * Without this check a customer could pay a small amount and have a big order marked paid.
 */
export function paymentMatchesOrder(
  order: { orderNumber: string; total: number },
  v: { status: string; amount: number; currency?: string },
  reference: string,
): boolean {
  return (
    v.status === "success" &&
    v.amount === Math.round(order.total * 100) &&
    (!v.currency || v.currency === "NGN") &&
    reference.startsWith(`${order.orderNumber}-`)
  );
}

export async function markOrderPaid(
  orderNumber: string,
  paymentReference: string,
): Promise<any> {
  const number = orderNumber.trim().toUpperCase();

  // Atomic claim: only ONE caller (webhook, return-redirect or admin) can flip
  // pending -> paid, so emails are never doubled.
  const claimed = await Order.findOneAndUpdate(
    { orderNumber: number, status: "pending" },
    { $set: { status: "paid", paidAt: new Date(), paymentReference } },
    { new: true },
  );
  if (claimed) {
    sendPaymentConfirmed(buildEmailContext(claimed)).catch(() => {});
    return claimed;
  }

  const order = await Order.findOne({ orderNumber: number });
  if (!order) return null;

  // Customer paid AFTER the order was auto-cancelled: take the stock again if it is still there.
  if (order.status === "cancelled" && order.stockRestored) {
    const revived = await Order.findOneAndUpdate(
      { orderNumber: number, status: "cancelled", stockRestored: true },
      { $set: { status: "paid", paidAt: new Date(), paymentReference, stockRestored: false } },
      { new: true },
    );
    if (revived) {
      try {
        await decrementStock(revived.items as any);
      } catch {
        revived.stockRestored = true;
        revived.adminNotes =
          `${revived.adminNotes ? revived.adminNotes + "\n" : ""}LATE PAYMENT: customer paid after the order was cancelled and an item is now out of stock. Ship when restocked or refund on Paystack.`;
        await revived.save();
      }
      sendPaymentConfirmed(buildEmailContext(revived)).catch(() => {});
      return revived;
    }
  }

  // Already paid / processing / shipped / delivered / refunded: nothing to redo
  if (!order.paymentReference && order.status !== "cancelled") {
    order.paymentReference = paymentReference;
    await order.save();
  }
  return order;
}

/**
 * Cancels or refunds an order and gives its stock back, exactly once.
 *  - cancelled: always puts stock back (if not already).
 *  - refunded : puts stock back only if the order had not shipped yet
 *               (shipped/delivered refunds mean goods come back - adjust stock by hand after checking them).
 */
export async function cancelOrRefundOrder(
  orderNumber: string,
  newStatus: "cancelled" | "refunded",
  note?: string,
): Promise<any> {
  const number = orderNumber.trim().toUpperCase();
  const current = await Order.findOne({ orderNumber: number });
  if (!current) return null;

  const wasShipped = current.status === "shipped" || current.status === "delivered" || anyPartSent((current as any).shipments);
  const shouldRestock = newStatus === "cancelled" || !wasShipped;
  const noteLine = note ? `${current.adminNotes ? current.adminNotes + "\n" : ""}${note}` : undefined;

  if (shouldRestock) {
    const claimed = await Order.findOneAndUpdate(
      { orderNumber: number, stockRestored: { $ne: true }, status: { $nin: ["cancelled", "refunded"] } },
      { $set: { status: newStatus, stockRestored: true, ...(noteLine ? { adminNotes: noteLine } : {}) } },
      { new: true },
    );
    if (claimed) {
      for (const item of claimed.items as any[]) {
        await giveStockBackForItem(item).catch((e) => console.error("Restock failed:", e));
      }
      return claimed;
    }
  }

  // Already cancelled/refunded (or no restock wanted): only the label changes
  return Order.findOneAndUpdate(
    { orderNumber: number },
    { $set: { status: newStatus, ...(noteLine ? { adminNotes: noteLine } : {}) } },
    { new: true },
  );
}

/**
 * Cancels Paystack orders that were never paid so abandoned checkouts do not
 * lock stock. Before cancelling, it asks Paystack whether the customer actually paid.
 */
export async function expireStalePendingOrders(maxAgeMinutes = 180): Promise<number> {
  const cutoff = new Date(Date.now() - maxAgeMinutes * 60 * 1000);
  const stale = await Order.find({
    status: "pending",
    paymentMethod: { $in: ["paystack", "flutterwave"] },
    createdAt: { $lt: cutoff },
  })
    .sort({ createdAt: 1 })
    .limit(50);

  let cancelled = 0;
  for (const order of stale) {
    try {
      const refs: string[] = (order as any).paystackRefs ?? [];
      let verdict: "paid" | "unpaid" | "unknown" = "unknown";
      if (refs.length === 0) {
        verdict = "unpaid"; // customer never reached the Paystack page
      } else {
        let paid = false;
        let unsure = false;
        for (const ref of refs) {
          try {
            const v = await verifyTransaction(ref);
            if (paymentMatchesOrder(order, v, ref)) {
              await markOrderPaid(order.orderNumber, ref);
              paid = true;
              break;
            }
            if (v.status === "success") {
              console.error(`[expire] ${order.orderNumber}: paid amount ${v.amount} does not match total, left pending for review`);
              unsure = true;
            }
          } catch (err) {
            // 404 = Paystack has no such payment (never started); anything else = could not check, so do not cancel
            if (!(err instanceof PaystackError && err.status === 404)) unsure = true;
          }
        }
        verdict = paid ? "paid" : unsure ? "unknown" : "unpaid";
      }

      if (verdict === "unpaid") {
        await cancelOrRefundOrder(
          order.orderNumber,
          "cancelled",
          "Auto-cancelled: payment was not completed in time. Stock returned.",
        );
        cancelled++;
      }
    } catch (err) {
      console.error(`[expire] ${order.orderNumber} failed:`, err);
    }
  }
  return cancelled;
}

// Order creation limiter: 10 orders per minute per visitor, so nobody can drain stock by spamming.
const orderHits = new Map<string, { count: number; resetAt: number }>();
function orderLimited(ip: string): boolean {
  const now = Date.now();
  const hit = orderHits.get(ip);
  if (!hit || hit.resetAt < now) {
    orderHits.set(ip, { count: 1, resetAt: now + 60_000 });
    return false;
  }
  hit.count += 1;
  return hit.count > 10;
}

// Small in-memory limiter for gift-progress: 30 lookups per minute per IP.
const progressHits = new Map<string, { count: number; resetAt: number }>();

  const hit = progressHits.get(ip);
  if (!hit || hit.resetAt < now) {
    progressHits.set(ip, { count: 1, resetAt: now + 60_000 });
    return false;
  }
  hit.count += 1;
  return hit.count > 30;
}

  try {
    const { orderNumber } = req.body || {};
    if (!orderNumber || typeof orderNumber !== "string") {
      return res.status(400).json({
        error: { code: "BAD_REQUEST", message: "orderNumber is required" },
      });
    }

    const order = await findByOrderNumber(orderNumber);
    if (!order) {
      return res.status(404).json({
        error: { code: "NOT_FOUND", message: "Order not found" },
      });
    }

    if (
      order.status === "paid" ||
      order.status === "processing" ||
      order.status === "shipped" ||
      order.status === "delivered"
    ) {
      return res.status(409).json({
        error: { code: "ALREADY_PAID", message: "This order is already paid" },
      });
    }

    if (order.status === "cancelled" || order.status === "refunded") {
      return res.status(409).json({
        error: {
          code: "ORDER_CLOSED",
          message: "This order has expired or been cancelled. Please place a new order.",
        },
      });
    }

    if (order.paymentMethod !== "paystack") {
      return res.status(400).json({
        error: {
          code: "WRONG_METHOD",
          message: `Order payment method is "${order.paymentMethod}", not paystack`,
        },
      });
    }

    const reference = `${order.orderNumber}-${Date.now()}`;
    const callbackUrl = `${config.siteUrl}/order-confirmation/${order.orderNumber}?ref=${reference}`;

    // Remember the reference BEFORE the customer pays, so an unpaid order can be re-checked later
    await Order.updateOne({ _id: order._id }, { $addToSet: { paystackRefs: reference } });

    const init = await initializeTransaction({
      email: order.shipping.email,
      amountKobo: order.total * 100,
      reference,
      callbackUrl,
      metadata: {
        orderNumber: order.orderNumber,
        customerName: order.shipping.fullName,
      },
    });

    return res.status(200).json({
      data: {
        authorizationUrl: init.authorization_url,
        reference,
      },
    });
  } catch (err: any) {
    if (err instanceof PaystackError) {
      return res.status(err.status).json({
        error: { code: "PAYSTACK_ERROR", message: err.message },
      });
    }
    console.error("POST /api/payments/initialize failed:", err);
    return res.status(500).json({
      error: { code: "SERVER_ERROR", message: "Failed to initialize payment" },
    });
  }
});

      );

/**
 * POST /api/payments/webhook/paystack
 *
 * Uses express.raw inline so this endpoint receives unparsed body bytes
 * for HMAC SHA512 verification.
 */
router.post(
  "/webhook/paystack",
  express.raw({ type: "application/json" }),
  async (req: Request, res: Response) => {
    try {
      const signature = req.headers["x-paystack-signature"] as string | undefined;
      const rawBody = (req.body as Buffer).toString("utf-8");

      if (!verifyWebhookSignature(rawBody, signature)) {
        console.warn("⚠️  Paystack webhook with invalid signature");
        return res.status(401).send("Invalid signature");
      }

      const payload = JSON.parse(rawBody);
      const eventType = payload.event;
      const reference = payload?.data?.reference;

      console.log(`📥 Paystack webhook: ${eventType} (ref: ${reference})`);

      if (eventType === "charge.success" && reference) {
        const verification = await verifyTransaction(reference);
        if (verification.status === "success") {
          const orderNumber = String(verification.metadata?.orderNumber ?? "");
          const existing = orderNumber ? await findByOrderNumber(orderNumber) : null;
          if (existing && paymentMatchesOrder(existing, verification, reference)) {
            await markOrderPaid(orderNumber, reference);
            console.log(`✅ Marked order paid via webhook: ${orderNumber}`);
          } else {
            console.error(
              `⚠️  Webhook payment did not match an order (order ${orderNumber || "?"}, ref ${reference}, paid ${verification.amount} kobo)`,
            );
          }
        }
      }

      return res.status(200).send("OK");
    } catch (err) {
      // 500 (not 200) so Paystack tries again later instead of the payment being lost
      console.error("Paystack webhook error:", err);
      return res.status(500).send("Error");
    }
  },
);

/**
 * Paystack service.
 *
 * Wraps Paystack's REST API for the two operations we need:
 *  1. initialize() — create a payment session, get checkout URL
 *  2. verify()     — confirm a payment was successful (called both on
 *                     return-redirect AND on webhook)
 *
 * Plus webhook signature verification using HMAC SHA512.
 *
 * Paystack docs: https://paystack.com/docs/api/transaction
 */

import crypto from "crypto";
import { config } from "../config";

const PAYSTACK_BASE = "https://api.paystack.co";

interface InitializeArgs {
  email: string;
  amountKobo: number;     // amount in KOBO (1 NGN = 100 kobo)
  reference: string;      // your unique reference (we use orderNumber)
  callbackUrl: string;    // where to send customer after payment
  metadata?: Record<string, any>;
}

export interface PaystackInitResponse {
  authorization_url: string;  // checkout URL to redirect customer to
  access_code: string;
  reference: string;
}

export interface PaystackVerifyResponse {
  status: "success" | "failed" | "abandoned";
  amount: number;        // in kobo
  currency?: string;
  reference: string;
  paid_at: string | null;
  channel: string;       // 'card', 'bank_transfer', 'ussd', etc.
  customer: { email: string };
  metadata?: Record<string, any>;
}

export class PaystackError extends Error {
  status: number;
  constructor(message: string, status = 500) {
    super(message);
    this.status = status;
  }
}

function ensureKey() {
  if (!config.paystack.secretKey) {
    throw new PaystackError(
      "Paystack secret key not configured. Set PAYSTACK_SECRET_KEY in .env",
      500,
    );
  }
}

/**
 * Initializes a transaction with Paystack.
 * Returns a checkout URL the frontend redirects the customer to.
 */
export async function initializeTransaction(
  args: InitializeArgs,
): Promise<PaystackInitResponse> {
  ensureKey();

  const res = await fetch(`${PAYSTACK_BASE}/transaction/initialize`, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${config.paystack.secretKey}`,
      "Content-Type": "application/json",
    },
    signal: AbortSignal.timeout(20000),
    body: JSON.stringify({
      email: args.email,
      amount: args.amountKobo,
      reference: args.reference,
      callback_url: args.callbackUrl,
      metadata: args.metadata,
    }),
  });

  const json = await res.json();
  if (!res.ok || !json.status) {
    throw new PaystackError(
      json?.message || "Paystack initialization failed",
      res.status,
    );
  }

  return json.data as PaystackInitResponse;
}

/**
 * Verifies a transaction status with Paystack.
 * Called when:
 *  - Customer returns from payment page (we re-verify to be safe)
 *  - Webhook fires (we verify the reference, never trust the body alone)
 */
export async function verifyTransaction(
  reference: string,
): Promise<PaystackVerifyResponse> {
  ensureKey();

  const res = await fetch(
    `${PAYSTACK_BASE}/transaction/verify/${encodeURIComponent(reference)}`,
    {
      headers: {
        Authorization: `Bearer ${config.paystack.secretKey}`,
      },
      signal: AbortSignal.timeout(20000),
    },
  );

  const json = await res.json();
  if (!res.ok || !json.status) {
    throw new PaystackError(
      json?.message || "Paystack verification failed",
      res.status,
    );
  }

  return json.data as PaystackVerifyResponse;
}

/**
 * Verifies a webhook came from Paystack by checking the HMAC SHA512 signature.
 * Without this, anyone could send fake "payment success" requests.
 *
 * Paystack signs the entire raw body with our secret key.
 */
export function verifyWebhookSignature(rawBody: string, signature: string | undefined): boolean {
  if (!signature || !config.paystack.secretKey) return false;

  const expected = crypto
    .createHmac("sha512", config.paystack.secretKey)
    .update(rawBody)
    .digest("hex");

  const a = Buffer.from(signature);
  const b = Buffer.from(expected);
  // timingSafeEqual throws when lengths differ; a wrong-length signature is just invalid
  if (a.length !== b.length) return false;
  return crypto.timingSafeEqual(a, b);
}

/**

const LOW_AT = 3;
const COUNTED = ["paid", "processing", "shipped", "delivered"];

/** Lagos is UTC+1 all year (no daylight saving). */
function lagosYmd(offsetDays = 0): string {
  const d = new Date(Date.now() + 3600_000 + offsetDays * 86400_000);
  return d.toISOString().slice(0, 10);
}
function lagosHour(): number {
  return new Date(Date.now() + 3600_000).getUTCHours();
}

export type SummaryDay = "today" | "yesterday";

export async function buildSummary(day: SummaryDay = "yesterday") {
  const ymd = lagosYmd(day === "today" ? 0 : -1);
  const start = new Date(`${ymd}T00:00:00+01:00`);
  const end = new Date(start.getTime() + 86400_000 - 1);

  const [orders, toPackCount, toGetCount, oldestToPack, unpaidCount, stock, expenses] = await Promise.all([
    Order.find({ status: { $in: COUNTED }, createdAt: { $gte: start, $lte: end } })
      .select("subtotal shippingFee total source items.name items.quantity items.isFreeGift")
      .lean(),
    Order.countDocuments(PACK_NOW),
    Order.countDocuments(WAITING_GOODS),
    Order.findOne(PACK_NOW)
      .sort({ createdAt: 1 })
      .select("createdAt")
      .lean(),
    Order.countDocuments({ status: "pending", paymentMethod: { $ne: "cod" } }),
    Combo.find({ isActive: { $ne: false }, stockLeft: { $lte: LOW_AT } }).select("name stockLeft").sort({ stockLeft: 1, name: 1 }).limit(30).lean(),
    Expense.find({ date: { $gte: start, $lte: end } }).select("amount category").lean(),
  ]);

  const costs = orders.length ? await OrderCost.find({ orderId: { $in: orders.map((o: any) => o._id) } }).lean() : [];
  const report = buildReport(orders as any, costs as any, expenses as any);

  // Partner commission on the orders that are counted in profit
  const costedIds = (costs as any[]).filter((c) => c.complete).map((c) => c.orderId);
  const commissionCost = costedIds.length
    ? ((await Commission.find({ orderId: { $in: costedIds } }).select("amount").lean()) as any[]).reduce((t, c) => t + (c.amount ?? 0), 0)
    : 0;

  // best sellers by units
  const units = new Map<string, number>();
  for (const o of orders as any[]) for (const it of o.items ?? []) if (!it.isFreeGift) units.set(it.name, (units.get(it.name) ?? 0) + it.quantity);
  const bestSellers = [...units.entries()].sort((a, b) => b[1] - a[1]).slice(0, 3).map(([name, n]) => ({ name, units: n }));

  // partner commissions: owed now vs still waiting out the hold period
  let commissionOwed = 0;
  let commissionPending = 0;
  try {
    const open: any[] = await Commission.find({ paidAt: { $exists: false } }).select("resellerId amount").lean();
    const valid = open.filter((c) => c.resellerId);
    if (valid.length) {
      const { statusById } = await resolveCommissionStatuses(valid.map((c) => c.resellerId));
      for (const c of valid) {
        const s = statusById.get(String(c._id));
        if (s === "approved") commissionOwed += c.amount;
        else if (s === "pending") commissionPending += c.amount;
      }
    }
  } catch (e) {
    console.error("[summary] commissions failed:", e);
  }

  const revenue = (orders as any[]).reduce((s, o) => s + (o.total ?? 0), 0);
  return {
    day,
    date: ymd,
    orders: orders.length,
    manualOrders: (orders as any[]).filter((o) => (o.source ?? "website") !== "website").length,
    revenue,
    profit: report.orders.costed > 0 ? report.netProfit - commissionCost : null,
    profitCoverage: { costed: report.orders.costed, total: report.orders.total },
    expenses: report.expensesTotal,
    bestSellers,
    toPack: toPackCount,
    toGet: toGetCount,
    oldestToPackHours: oldestToPack ? Math.floor((Date.now() - new Date((oldestToPack as any).createdAt).getTime()) / 3600_000) : 0,
    unpaid: unpaidCount,
    lowStock: (stock as any[]).map((c) => ({ name: c.name, stockLeft: c.stockLeft ?? 0 })),
    commissionOwed,
    commissionPending,
  };
}

export async function sendSummaryEmail(day: SummaryDay = "yesterday"): Promise<{ sent: boolean; message: string; to?: string }> {
  const to = config.resend.adminNotify;
  if (!to) return { sent: false, message: "ADMIN_NOTIFY_EMAIL is not set, so there is nowhere to send the summary." };
  if (!config.resend.apiKey) return { sent: false, message: "RESEND_API_KEY is not set, so emails are switched off." };
  const s = await buildSummary(day);
  const resend = new Resend(config.resend.apiKey);
  try {
    const r = await resend.emails.send({
      from: config.resend.from,
      to,
      subject: `Smart Combo ${s.day === "today" ? "so far today" : "yesterday"}: ${naira(s.revenue)} from ${s.orders} order${s.orders === 1 ? "" : "s"}`,
      html: summaryHtml(s),
      replyTo: config.resend.replyTo || undefined,
    });
    if (r.error) return { sent: false, message: `Email failed: ${r.error.message}` };
    return { sent: true, message: "Summary sent.", to };
  } catch (e: any) {
    return { sent: false, message: `Email failed: ${e?.message ?? e}` };
  }
}

/** Live site only. Checks every 10 minutes; sends once per Lagos day after SUMMARY_HOUR. */
export function startDailySummary() {
  const g = globalThis as any;
  if (g.__scSummaryTimer) return;
  const tick = async () => {
    try {
      if (lagosHour() < SUMMARY_HOUR) return;
      const ymd = lagosYmd(0);
      try {
        await SummaryLog.create({ _id: ymd }); // fails if today's was already claimed
      } catch {
        return;
      }
      const r = await sendSummaryEmail("yesterday");
      console.log(`[summary] ${ymd}: ${r.message}`);
    } catch (e) {
      console.error("[summary] job failed:", e);
    }
  };
  g.__scSummaryTimer = setInterval(tick, 10 * 60 * 1000);
  g.__scSummaryTimer.unref?.();
  setTimeout(tick, 30_000).unref?.();
}

/**
 * Admin dashboard stats — aggregates for the homepage of /admin.
 *
 * GET /api/admin/dashboard/stats
 *
 * Returns:
 *   - ordersToday: count of orders created today
 *   - revenueToday: sum of totals from PAID orders today (excludes pending)
 *   - pendingOrders: count of orders with status "pending"
 *   - customers: distinct unique customers (by phone) ever
 *   - recentOrders: last 5 orders for quick preview
 */

import { Router, Request, Response } from "express";
import { Order } from "../../models/Order";
import { PACK_NOW, WAITING_GOODS } from "../../services/packFilters";

const router = Router();

router.get("/stats", async (_req: Request, res: Response) => {
  try {
    const startOfToday = new Date();
    startOfToday.setHours(0, 0, 0, 0);

    const [ordersToday, revenueAggResult, pendingOrders, customerAgg, recentOrders] =
      await Promise.all([
        Order.countDocuments({ createdAt: { $gte: startOfToday } }),

        // Revenue today = sum of totals for orders that are 'paid' OR beyond
        // (paid, processing, shipped, delivered) created today.
        // We exclude 'pending' so revenue reflects actual confirmed money.
        Order.aggregate([
          {
            $match: {
              createdAt: { $gte: startOfToday },
              status: { $in: ["paid", "processing", "shipped", "delivered"] },
            },
          },
          { $group: { _id: null, total: { $sum: "$total" } } },
        ]),

        Order.countDocuments({ status: "pending" }),

        // Unique customers ever (by phone)
        Order.aggregate([
          { $group: { _id: "$shipping.phone" } },
          { $count: "uniqueCustomers" },
        ]),

        // Recent 5 orders, lightweight
        Order.find({})
          .sort({ createdAt: -1 })
          .limit(5)
          .select("orderNumber status total shipping.fullName createdAt")
          .lean(),
      ]);

    // to pack = ready to pack now (waiting-for-goods orders are not); to get = paid pre-order orders still waiting for goods
    const [toPack, toGet] = await Promise.all([Order.countDocuments(PACK_NOW), Order.countDocuments(WAITING_GOODS)]);
    const revenueToday = revenueAggResult[0]?.total ?? 0;
    const customers = customerAgg[0]?.uniqueCustomers ?? 0;

    res.json({
      data: {
        ordersToday,

/**
 * Orders API - frontend client.
 *
 * Cart items include selectedVariants and the free-gift flag when sent to the backend.
 */

import { apiFetch } from "./client";
import type { Order, ShippingAddress, PaymentMethod, CartItem } from "@/types/order";
import { getReferral } from "@/lib/referral";

interface ApiResponse<T> {
  data: T;
}

interface BackendOrder extends Omit<Order, "id"> {
  _id: string;
}

function normalize(raw: BackendOrder): Order {
  const { _id, ...rest } = raw;
  return { ...rest, id: _id };
}

export interface CreateOrderInput {
  items: CartItem[];
  shipping: ShippingAddress;
  paymentMethod: PaymentMethod;
  notes?: string;
  ackPreorder?: boolean;
  /** "now": send the ready pieces first (Ilorin only) */
  partialChoice?: "now" | "together";
}

export async function createOrder(input: CreateOrderInput): Promise<Order> {
  // Send each cart item with its selectedVariants and free-gift flag.
  // The server needs isFreeGift to know the item comes from the Gift Bank.
  const payload = {
    items: input.items.map((it) => ({
      comboId: it.comboId,
      quantity: it.quantity,
      selectedVariants: it.selectedVariants ?? {},
      isFreeGift: Boolean(it.isFreeGift),
    })),
    shipping: input.shipping,
    paymentMethod: input.paymentMethod,
    ref: getReferral(),
    notes: input.notes,
    ackPreorder: Boolean(input.ackPreorder),
    partialChoice: input.partialChoice === "now" ? "now" : "together",
  };

  const res = await apiFetch<ApiResponse<BackendOrder>>("/api/orders", {
    method: "POST",
    body: JSON.stringify(payload),
  });
  return normalize(res.data);
}

export interface OrderPreview {
  lines: Array<{ comboId: string; name: string; ready: number; waiting: number; leadDaysMin: number; leadDaysMax: number; short?: boolean; want?: number }>;
  hasPre: boolean;
  /** some pieces are ready and some must be fetched */
  mixed: boolean;
  /** the city is one we deliver in ourselves (Ilorin) */
  local: boolean;
  subtotal: number;
  leadDaysMin: number;
  leadDaysMax: number;
}

/** Asks the server which pieces are ready and which must be fetched. Returns null if it cannot tell. */
export async function previewOrder(items: CartItem[], city: string): Promise<OrderPreview | null> {
  try {
    const res = await apiFetch<ApiResponse<OrderPreview>>("/api/orders/preview", {
      method: "POST",
      body: JSON.stringify({
        items: items.filter((i) => !i.isFreeGift).map((i) => ({ comboId: i.comboId, quantity: i.quantity })),
        city,
      }),
    });
    return res.data;
  } catch {
    return null;
  }
}

export async function trackOrder(orderNumber: string, phone: string): Promise<Order | null> {
  try {
    const res = await apiFetch<ApiResponse<BackendOrder>>(
      `/api/orders/track?orderNumber=${encodeURIComponent(orderNumber)}&phone=${encodeURIComponent(phone)}`,
    );
    return normalize(res.data);
  } catch (err: any) {
    if (err.status === 404) return null;
    throw err;
  }
}

function formatDate(dateString?: string) {
  if (!dateString) return '';
  try {
    return new Date(dateString).toLocaleDateString('en-NG', {
      day: 'numeric',
      month: 'long',
      year: 'numeric',
    });
  } catch {
    return '';
  }
}

export default function OrderTracking() {
  const [searchParams] = useSearchParams();
  const [orderNumber, setOrderNumber] = useState(searchParams.get('order') ?? '');
  const [phone, setPhone] = useState(searchParams.get('phone') ?? '');
  const [order, setOrder] = useState<Order | null>(null);
  const [loading, setLoading] = useState(false);
  const [notFound, setNotFound] = useState(false);

  // Auto-lookup if both query params provided
  useEffect(() => {
    if (orderNumber && phone) {
      handleLookup(orderNumber, phone);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const handleLookup = async (orderNo: string, ph: string) => {
    setLoading(true);
    setNotFound(false);
    setOrder(null);
    try {
      const result = await trackOrder(orderNo.trim(), ph.trim());
      if (result) {
        setOrder(result);
      } else {
        setNotFound(true);
      }
    } catch (err: any) {
      toast.error(err?.message || 'Failed to look up order');
    } finally {
      setLoading(false);
    }
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!orderNumber.trim() || !phone.trim()) {
      toast.error('Enter both order number and phone');
      return;
    }
    handleLookup(orderNumber, phone);
  };

  const reset = () => {
    setOrder(null);
    setNotFound(false);
  };

  return (
    <div className="min-h-screen bg-background text-foreground flex flex-col">
      <SEO title="Track your order" description="Check the status of your Smart Combo order." />
      <Header />
      <CartDrawer />

      <main className="flex-1 section-padding py-10 md:py-14">
        <div className="container-premium max-w-3xl">
          <Link to="/" className="inline-flex items-center gap-1 text-xs text-white/50 hover:text-primary mb-6">
            <ArrowLeft className="w-3 h-3" />
            Back to home
          </Link>

          <div className="text-center mb-8">
            <p className="text-xs uppercase tracking-[0.28em] text-primary/70 font-bold mb-2">Order status</p>
            <h1 className="text-2xl md:text-3xl font-black text-white mb-2">Track Your Order</h1>
            <p className="text-sm text-white/50 max-w-md mx-auto">
              Enter your order number and phone number to check the status.
            </p>
          </div>

          {/* Form (always visible) */}
          <form
            onSubmit={handleSubmit}
            className="rounded-2xl border border-white/10 p-5 md:p-6 mb-6"
            style={{ background: 'rgba(255,255,255,0.02)' }}
          >
            <div className="space-y-3">
              <label className="block">
                <span className="block text-[11px] font-bold text-white/60 uppercase tracking-wider mb-1.5">
                  Order Number

                <div className="relative">
                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/30" />
                  <input

                    value={orderNumber}

                    className="w-full pl-9 pr-3 py-2.5 rounded-lg bg-black/30 border border-white/10 focus:border-primary/40 focus:outline-none text-sm text-white placeholder-white/30 transition-colors"
                    placeholder="SC-2026-000001"
                    autoComplete="off"
                  />
                </div>

                <span className="block text-[11px] font-bold text-white/60 uppercase tracking-wider mb-1.5">

/**
 * Order Confirmation page.
 *
 * Reached three ways:
 *  1. From Checkout via state navigation (we have the order data already)
 *  2. From Paystack callback (URL has ?ref=... query — we verify, then show)
 *  3. Direct URL with :orderNumber (we ask for phone to look up)
 *
 * For Paystack, the URL looks like:
 *   /order-confirmation/SC-2026-000010?ref=SC-2026-000010-1715175263000
 *
 * On detecting `ref`, we:
 *  - Call /api/payments/verify/:reference
 *  - If paid → load the now-paid order
 *  - If not → show "payment incomplete" with retry option
 */

import { useEffect, useState } from 'react';
import { Link, useLocation, useParams, useSearchParams } from 'react-router-dom';
import {
  AlertCircle,
  Check,
  Copy,
  CreditCard,
  Loader2,
  MessageCircle,
  Package,
} from 'lucide-react';
import { Header } from '@/components/Header';
import { Footer } from '@/components/Footer';
import { CartDrawer } from '@/components/shared/CartDrawer';
import { useSettings } from '@/contexts/SettingsContext';
import { formatNaira } from '@/lib/format';
import { trackOrder } from '@/api/orders';
import { initializePaystack, verifyPaystackPayment } from '@/api/payments';
import type { Order } from '@/types/order';
import { OrderParts } from '@/components/OrderParts';
import { toast } from 'sonner';

export default function OrderConfirmation() {
  const { orderNumber } = useParams<{ orderNumber: string }>();
  const location = useLocation();
  const [searchParams] = useSearchParams();
  const { settings } = useSettings();

  const stateOrder = (location.state as { order?: Order; phone?: string } | null)?.order;
  const statePhone = (location.state as { order?: Order; phone?: string } | null)?.phone;
  const paystackRef = searchParams.get('ref');

  const [order, setOrder] = useState<Order | null>(stateOrder ?? null);
  const [loading, setLoading] = useState(!stateOrder || !!paystackRef);
  const [copied, setCopied] = useState(false);
  const [paymentStatus, setPaymentStatus] = useState<
    'idle' | 'verifying' | 'paid' | 'failed' | 'abandoned'
  >('idle');
  const [retryingPayment, setRetryingPayment] = useState(false);

  const [phoneInput, setPhoneInput] = useState(
    statePhone ?? (orderNumber ? sessionStorage.getItem(`order:${orderNumber}:phone`) ?? '' : ''),
  );
  const [recoveryError, setRecoveryError] = useState<string | null>(null);

  const fetchOrder = async (phone: string) => {
    if (!orderNumber || !phone) return;
    try {
      const fetched = await trackOrder(orderNumber, phone.trim());
      if (fetched) setOrder(fetched);
      else setRecoveryError("We couldn't find an order with those details.");
    } catch {
      setRecoveryError('Something went wrong. Try again.');
    }
  };

  // Step 1: handle Paystack return
  useEffect(() => {
    let cancelled = false;
    async function handle() {
      if (!paystackRef) {
        if (!order && statePhone) await fetchOrder(statePhone);
        if (!cancelled) setLoading(false);
        return;
      }
      setPaymentStatus('verifying');
      try {
        const result = await verifyPaystackPayment(paystackRef);
        if (cancelled) return;
        if (result.paid) {
          setPaymentStatus('paid');
          // Load the now-paid order
          const phone = statePhone || sessionStorage.getItem(`order:${orderNumber}:phone`) || '';
          if (phone) await fetchOrder(phone);
          toast.success('Payment confirmed! 🎉');
        } else {
          setPaymentStatus(
            result.status === 'abandoned' ? 'abandoned' : 'failed',
          );
          // Try to load order anyway (status will still be 'pending')
          const phone = statePhone || sessionStorage.getItem(`order:${orderNumber}:phone`) || '';
          if (phone) await fetchOrder(phone);
        }
      } catch {
        if (!cancelled) {
          setPaymentStatus('failed');
          toast.error('Payment verification failed. Please try again.');
        }
      } finally {
        if (!cancelled) setLoading(false);
      }
    }
    handle();
    return () => { cancelled = true; };
  }, []); // eslint-disable-line

  const handleRecoveryFetch = async (e?: React.FormEvent) => {
    e?.preventDefault();
    setRecoveryError(null);
    setLoading(true);
    await fetchOrder(phoneInput.trim());
    setLoading(false);
  };

  const copyOrderNumber = () => {
    if (!order) return;
    navigator.clipboard.writeText(order.orderNumber).then(() => {
      setCopied(true);
      toast.success('Order number copied');
      setTimeout(() => setCopied(false), 2000);
    });
  };

  const retryPaystack = async () => {
    if (!order) return;
    setRetryingPayment(true);
    try {
      const init = await initializePaystack(order.orderNumber);
      window.location.href = init.authorizationUrl;
    } catch (err: any) {
      toast.error(err?.message || 'Could not start payment');
    } finally {
      setRetryingPayment(false);
    }
  };

  return (
    <div className="min-h-screen bg-background text-foreground flex flex-col">
      <Header />
      <CartDrawer />

      <main className="flex-1 section-padding py-10">
        <div className="container-premium max-w-3xl">
          {/* Need phone — direct URL access without state */}
          {!order && !loading && (
            <div className="rounded-2xl border border-white/10 p-6" style={{ background: 'rgba(255,255,255,0.02)' }}>
              <h1 className="text-xl font-black text-white mb-2">
                Confirm your order details
              </h1>
              <p className="text-sm text-white/50 mb-4">
                Enter the phone number used for order <span className="font-bold text-white/80">{orderNumber}</span>.
              </p>
              <form onSubmit={handleRecoveryFetch} className="flex flex-col sm:flex-row gap-2">
                <input
                  type="tel"
                  value={phoneInput}
                  onChange={(e) => setPhoneInput(e.target.value)}
                  placeholder="+234 800 000 0000"
                  className="flex-1 px-3 py-2.5 rounded-lg bg-black/30 border border-white/10 focus:border-primary/40 focus:outline-none text-sm text-white"
                />
                <button type="submit" className="btn-primary px-5 py-2.5 text-sm font-bold">
                  Continue
                </button>
              </form>
              {recoveryError && <p className="mt-3 text-xs text-red-400">{recoveryError}</p>}
            </div>
          )}

          {loading && (
            <div className="text-center py-16">
              <Loader2 className="w-10 h-10 animate-spin text-primary mx-auto" />
              <p className="text-white/50 text-sm mt-4">
                {paymentStatus === 'verifying'
                  ? 'Verifying your payment...'
                  : 'Loading your order...'}
              </p>
            </div>
          )}

          {order && (
            <>
              {/* Hero — varies by payment status */}
              {paymentStatus === 'failed' || paymentStatus === 'abandoned' ? (
                <div className="text-center mb-8">
                  <div className="w-16 h-16 rounded-full bg-amber-500/15 border border-amber-500/30 flex items-center justify-center mx-auto mb-4">
                    <AlertCircle className="w-8 h-8 text-amber-400" />
                  </div>
                  <p className="text-xs uppercase tracking-[0.28em] text-amber-400 font-bold mb-2">
                    Payment {paymentStatus === 'abandoned' ? 'cancelled' : 'incomplete'}
                  </p>
                  <h1 className="text-2xl md:text-3xl font-black text-white mb-3">
                    Your order is held for {order.shipping.fullName.split(' ')[0]}
                  </h1>
                  <p className="text-white/50 text-sm max-w-md mx-auto mb-6">
                    Your order is created and stock is reserved. Complete payment below to ship it out.
                  </p>
                  <button
                    type="button"
                    onClick={retryPaystack}
                    disabled={retryingPayment}
                    className="btn-primary inline-flex items-center gap-2 px-6 py-3 text-sm font-bold disabled:opacity-50"
                  >
                    {retryingPayment ? (
                      <><Loader2 className="w-4 h-4 animate-spin" />Loading...</>
                    ) : (
                      <><CreditCard className="w-4 h-4" />Pay {formatNaira(order.total)} now</>
                    )}
                  </button>
                </div>
              ) : (
                <div className="text-center mb-8">
                  <div className="w-16 h-16 rounded-full bg-emerald-500/15 border border-emerald-500/30 flex items-center justify-center mx-auto mb-4">
                    <Check className="w-8 h-8 text-emerald-400" />
                  </div>
                  <p className="text-xs uppercase tracking-[0.28em] text-emerald-400 font-bold mb-2">
                    {paymentStatus === 'paid' ? 'Payment confirmed' : 'Order received'}
                  </p>
                  <h1 className="text-2xl md:text-3xl font-black text-white mb-3">
                    Thank you, {order.shipping.fullName.split(' ')[0]} 🎉
                  </h1>
                  <p className="text-white/50 text-sm max-w-md mx-auto">
                    {paymentStatus === 'paid'
                      ? "We've received your payment. Your combo is being prepared for shipment."
                      : "We've received your order and will be in touch shortly."}
                  </p>
                </div>
              )}

              {/* Order number */}
              <div className="text-center mb-8">
                <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full border border-primary/30 bg-primary/5">
                  <span className="text-xs text-white/50">Order #</span>
                  <span className="font-mono font-bold text-primary">{order.orderNumber}</span>
                  <button
                    type="button"
                    onClick={copyOrderNumber}
                    className="ml-1 text-white/40 hover:text-primary transition-colors"
                    aria-label="Copy"
                  >
                    {copied ? <Check className="w-4 h-4" /> : <Copy className="w-3.5 h-3.5" />}
                  </button>
                </div>
              </div>

              {/* Payment instructions (only for non-Paystack methods OR if Paystack failed) */}
              {paymentStatus !== 'paid' && order.status !== 'paid' && order.status !== 'processing' && order.status !== 'shipped' && order.status !== 'delivered' && (

function PaymentInstructions({ order, settings }: { order: Order; settings: any }) {
  if (order.paymentMethod === 'cod') {
    return (
      <div className="rounded-2xl border border-emerald-500/25 bg-emerald-500/5 p-5 mb-5">
        <p className="text-xs uppercase tracking-widest text-emerald-300 font-bold mb-2">Cash on delivery</p>
        <p className="text-sm text-white/80 mb-1">
          You'll pay <span className="font-black text-emerald-300">{formatNaira(order.total)}</span> when our courier delivers your combo.
        </p>
        <p className="text-xs text-white/50">Estimated delivery: 1-2 business days.</p>
      </div>
    );
  }

  if (order.paymentMethod === 'bank_transfer') {
    const bank = settings?.bankAccount;
    return (
      <div className="rounded-2xl border border-primary/25 bg-primary/5 p-5 mb-5">
        <p className="text-xs uppercase tracking-widest text-primary font-bold mb-3">Bank transfer</p>
        <p className="text-sm text-white/80 mb-4">
          Transfer <span className="font-black text-primary">{formatNaira(order.total)}</span> to the account below.
        </p>
        {bank?.accountNumber ? (
          <div className="rounded-lg bg-black/30 border border-white/10 p-4 space-y-2">
            <BankRow label="Bank" value={bank.bankName} />
            <BankRow label="Account name" value={bank.accountName} />
            <BankRow label="Account number" value={bank.accountNumber} copyable />
            <div className="border-t border-white/10 pt-2 mt-3">
              <p className="text-[11px] text-white/40">
                Use your order number <span className="font-mono text-primary">{order.orderNumber}</span> as the reference.
              </p>
            </div>
          </div>
        ) : (
          <p className="text-xs text-amber-300">Bank details not configured. We'll send them to you shortly.</p>
        )}
      </div>
    );
  }

  if (order.paymentMethod === 'whatsapp') {
    return (
      <div className="rounded-2xl border border-emerald-500/25 bg-emerald-500/5 p-5 mb-5">
        <p className="text-xs uppercase tracking-widest text-emerald-300 font-bold mb-2">WhatsApp payment</p>
        <p className="text-sm text-white/80 mb-3">
          We'll message you on WhatsApp at <span className="font-bold text-white">{order.shipping.phone}</span>.
        </p>
      </div>
    );
  }

  if (order.paymentMethod === 'paystack') {
    return null; // Handled in main hero
  }

  return null;
}

function BankRow({ label, value, copyable }: { label: string; value: string; copyable?: boolean }) {
  const handleCopy = () => {
    navigator.clipboard.writeText(value);
    toast.success(`${label} copied`);
  };
  return (
    <div className="flex items-center justify-between gap-3">
      <span className="text-[11px] uppercase tracking-wider text-white/40">{label}</span>
      <div className="flex items-center gap-2">
        <span className="text-sm font-mono font-bold text-white">{value}</span>
        {copyable && (
          <button type="button" onClick={handleCopy} className="text-white/40 hover:text-primary" aria-label={`Copy ${label}`}>
            <Copy className="w-3.5 h-3.5" />
          </button>
        )}
      </div>
    </div>
  );
}

function Row({ label, value }: { label: string; value: string }) {
  return (
    <div className="flex items-center justify-between">
      <span className="text-white/50">{label}</span>
      <span className="text-white/80 tabular-nums font-medium">{value}</span>
    </div>
  );
}
